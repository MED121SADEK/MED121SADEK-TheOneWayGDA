    73→    73→export function generateActionToken(userId: string, action: 'approve' | 'reject' | 'reset' | 'sub_approve' | 'sub_reject'): string {
    74→    74→  const payload = `${userId}:${action}:${Date.now()}`

    79→    79→export function verifyActionToken(token: string): { userId: string; action: 'approve' | 'reject' | 'reset'; valid: boolean } {
    80→    80→  try {
    81→    81→    const decoded = Buffer.from(token, 'base64url').toString('utf-8')
    82→    82→    const [payload, signature] = decoded.split(':')
    83→    83→
    84→    84→    // Reconstruct and verify signature
    85→    85→    const parts = payload.split(':')
    86→    86→    if (parts.length < 3) return { userId: '', action: 'approve', valid: false }
    87→    87→
    88→    88→    const userId = parts[0]
    89→    89→    const action = parts[1] as 'approve' | 'reject' | 'reset'
    90→    90→    const timestamp = parseInt(parts[2])
    91→    91→
    92→    92→    // Check token expiry (24 hours)
    93→    93→    if (Date.now() - timestamp > 24 * 60 * 60 * 1000) {
    94→    94→      return { userId, action, valid: false }
    95→    95→    }
    96→    96→
    97→    97→    // Verify HMAC signature
    98→    98→    const expectedPayload = `${userId}:${action}:${timestamp}`
    99→    99→    const expectedSig = createHmac('sha256', EMAIL_SECRET).update(expectedPayload).digest('hex')
   100→   100→
   101→   101→    // Find the signature part (everything after the last : of payload)
   102→   102→    const sigIndex = decoded.lastIndexOf(':') + 1
   103→   103→    const actualSig = decoded.slice(sigIndex)
   104→   104→
   105→   105→    if (actualSig === expectedSig) {
   106→   106→      return { userId, action, valid: true }
   107→   107→    }
   108→   108→    return { userId, action, valid: false }
   109→   109→  } catch {
   110→   110→    return { userId: '', action: 'approve', valid: false }
   111→   111→  }
