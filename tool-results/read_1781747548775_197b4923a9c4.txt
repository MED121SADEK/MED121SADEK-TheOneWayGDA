   128→
   129→function makeRequest(url: string, options: RequestInit = {}): NextRequest {
   130→  return new NextRequest(new URL(url, 'http://localhost:3000'), options)
   131→}
   132→function makeJsonRequest(url: string, body: unknown, options: RequestInit = {}): NextRequest {
   133→  return new NextRequest(new URL(url, 'http://localhost:3000'), {
   134→    ...options, method: 'POST',
   135→    headers: { 'Content-Type': 'application/json', ...options.headers },
   136→    body: JSON.stringify(body),
   137→  })
   138→}
   139→function authedRequest(url: string, token = 'tok', options: RequestInit = {}) {
   140→  return makeRequest(url, { ...options, headers: { Authorization: `Bearer ${token}`, ...options.headers } })
   141→}
