   462→   462→function validateStepConfig(step: ChainStep, index: number): ValidationError[] {
   463→   463→  const errors: ValidationError[] = []
   464→   464→  const prefix = step.id || `steps[${index}]`
   465→   465→  const cfg = step.config as unknown as Record<string, unknown>
   466→   466→
   467→   467→  switch (step.action) {
   468→   468→    case 'clean_data': {
   469→   469→      const strat = cfg.strategy as string | undefined
   470→   470→      if (!strat || !['auto', 'standard', 'aggressive'].includes(strat)) {
   471→   471→        errors.push({ stepId: prefix, field: 'config.strategy', message: `Invalid strategy "${strat}". Expected: auto | standard | aggressive` })
   472→   472→      }
   473→   473→      if (cfg.columns !== undefined && !Array.isArray(cfg.columns)) {
   474→   474→        errors.push({ stepId: prefix, field: 'config.columns', message: 'columns must be a string array' })
   475→   475→      }
   476→   476→      break
   477→   477→    }
   478→   478→
   479→   479→    case 'run_model': {
   480→   480→      const model = cfg.model as string | undefined
   481→   481→      const validModels = ['descriptive-stats', 'regression', 'random-forest', 'anomaly-detection', 'classification', 'clustering', 'time-series', 'neural-net']
   482→   482→      if (!model || !validModels.includes(model)) {
   483→   483→        errors.push({ stepId: prefix, field: 'config.model', message: `Invalid model "${model}". Expected one of: ${validModels.join(', ')}` })
   484→   484→      }
   485→   485→      break
   486→   486→    }
   487→   487→
   488→   488→    case 'generate_report': {
   489→   489→      const fmt = cfg.format as string | undefined
   490→   490→      if (!fmt || !['pdf', 'html', 'csv', 'json'].includes(fmt)) {
   491→   491→        errors.push({ stepId: prefix, field: 'config.format', message: `Invalid format "${fmt}". Expected: pdf | html | csv | json` })
   492→   492→      }
   493→   493→      break
   494→   494→    }
   495→   495→
   496→   496→    case 'send_notification': {
   497→   497→      const ch = cfg.channel as string | undefined
   498→   498→      if (!ch || !['email', 'push', 'webhook', 'slack', 'teams'].includes(ch)) {

   628→   628→function validateChainRequest(body: Record<string, unknown>): { valid: boolean; errors: ValidationError[] } {
   629→   629→  const errors: ValidationError[] = []
   630→   630→
   631→   631→  // name
   632→   632→  const name = body.name as string | undefined
   633→   633→  if (!name || typeof name !== 'string' || name.trim().length === 0) {
   634→   634→    errors.push({ field: 'name', message: 'name is required' })
   635→   635→  } else if (name.length > 200) {
   636→   636→    errors.push({ field: 'name', message: 'name must be 200 characters or fewer' })
   637→   637→  }
   638→   638→
   639→   639→  // steps
   640→   640→  const steps = body.steps as ChainStep[] | undefined
   641→   641→  if (!Array.isArray(steps) || steps.length === 0) {
   642→   642→    errors.push({ field: 'steps', message: 'steps must be a non-empty array' })
   643→   643→  } else if (steps.length > 100) {
   644→   644→    errors.push({ field: 'steps', message: 'A chain may contain at most 100 steps' })
   645→   645→  } else {
   646→   646→    // Check for duplicate step IDs
   647→   647→    const seenIds = new Set<string>()
   648→   648→    for (let i = 0; i < steps.length; i++) {
   649→   649→      const step = steps[i]
   650→   650→      if (!step.id || typeof step.id !== 'string' || step.id.trim().length === 0) {
   651→   651→        errors.push({ stepId: step.id || undefined, field: `steps[${i}].id`, message: 'Each step must have a non-empty id' })
   652→   652→      } else if (seenIds.has(step.id)) {
   653→   653→        errors.push({ stepId: step.id, field: `steps[${i}].id`, message: `Duplicate step id: "${step.id}"` })
   654→   654→      } else {
   655→   655→        seenIds.add(step.id)
   656→   656→      }
   657→   657→
   658→   658→      // Validate action type
   659→   659→      if (!step.action || !validateActionType(step.action)) {
   660→   660→        errors.push({ stepId: step.id, field: `steps[${i}].action`, message: `Invalid action type "${step.action}". Valid types: ${ALL_ACTION_TYPES.join(', ')}` })
   661→   661→      } else {
   662→   662→        // Validate step config
   663→   663→        errors.push(...validateStepConfig(step, i))
   664→   664→      }
   665→   665→
   666→   666→      // Validate condition if provided
   667→   667→      if (step.condition !== undefined) {
   668→   668→        const cond = step.condition as string
   669→   669→        const validBaseConditions = ['always', 'on_success', 'on_failure']
   670→   670→        if (!validBaseConditions.includes(cond) && !cond.startsWith('expression:')) {
   671→   671→          errors.push({ stepId: step.id, field: `steps[${i}].condition`, message: `Invalid condition "${cond}". Use: always | on_success | on_failure | expression:<expr>` })
   672→   672→        }
   673→   673→      }
   674→   674→    }
   675→   675→  }
   676→   676→
   677→   677→  // trigger
   678→   678→  const trigger = body.trigger as string | undefined
   679→   679→  if (!trigger || !validateTriggerType(trigger)) {
   680→   680→    errors.push({ field: 'trigger', message: `Invalid trigger type "${trigger}". Expected: ${ALL_TRIGGER_TYPES.join(', ')}` })
   681→   681→  }
   682→   682→
   683→   683→  // executeImmediately
   684→   684→  if (body.executeImmediately !== undefined && typeof body.executeImmediately !== 'boolean') {
   685→   685→    errors.push({ field: 'executeImmediately', message: 'executeImmediately must be a boolean' })
   686→   686→  }
   687→   687→
   688→   688→  return { valid: errors.length === 0, errors }
   689→   689→}

   691→   691→// ─── Condition evaluation ───────────────────────────────────────────────
   692→   692→
   693→   693→function shouldExecuteStep(
   694→   694→  step: ChainStep,
   695→   695→  previousStatus: StepStatus | null,
   696→   696→): boolean {
   697→   697→  const condition: string = step.condition ?? 'always'
   698→   698→
   699→   699→  // First step always runs (unless condition explicitly says otherwise)
   700→   700→  if (previousStatus === null) {
   701→   701→    return condition === 'always' || condition === 'on_success'
   702→   702→  }
   703→   703→
   704→   704→  switch (condition) {
   705→   705→    case 'always':
   706→   706→      return true
   707→   707→    case 'on_success':
   708→   708→      return previousStatus === 'success'
   709→   709→    case 'on_failure':
   710→   710→      return previousStatus === 'error'
   711→   711→    default:
   712→   712→      // Expression-based condition — evaluate against step results context
   713→   713→      // Format: "expression:<javascript-like condition>"
   714→   714→      if (condition.startsWith('expression:')) {
   715→   715→        // In a production system, this would use a safe expression evaluator.
   716→   716→        // For now, expression conditions always evaluate to true so the step
   717→   717→        // runs and the output can be inspected.  The expression is recorded
   718→   718→        // for observability.
   719→   719→        return true
   720→   720→      }
   721→   721→      return true
   722→   722→  }
   723→   723→}

   945→   945→            message: `Webhook call to ${wCfg.url} returned ${res.status}.`,
   946→   946→          }
   947→   947→        } catch (fetchErr: unknown) {
   948→   948→          const fetchMsg = fetchErr instanceof Error ? fetchErr.message : 'Fetch error'
   949→   949→          output = {
   950→   950→            status: 0,
   951→   951→            ok: false,
   952→   952→            error: fetchMsg,
   953→   953→            message: `Webhook call failed: ${fetchMsg}`,
   954→   954→          }
   955→   955→        }
   956→   956→        break
   957→   957→      }
   958→   958→
   959→   959→      case 'conditional_logic': {
   960→   960→        const clCfg = step.config as ConditionalLogicConfig
   961→   961→        // Evaluate condition against context of previous step outputs
   962→   962→        const previousOutputs: Record<string, unknown> = {}
   963→   963→        for (const [id, result] of Array.from(ctx.stepResults.entries())) {
   964→   964→          previousOutputs[id] = {
   965→   965→            status: result.status,
   966→   966→            output: result.output,
   967→   967→          }
   968→   968→        }
   969→   969→        output = {
   970→   970→          condition: clCfg.condition,
   971→   971→          evaluatedWith: previousOutputs,
   972→   972→          onTrue: clCfg.onTrue,
   973→   973→          onFalse: clCfg.onFalse,
   974→   974→          message: 'Conditional branch evaluated. Referenced steps should use on_success/on_failure conditions.',
   975→   975→        }
   976→   976→        break
   977→   977→      }
   978→   978→
   979→   979→      default: {
   980→   980→        // Exhaustiveness check — should never reach here due to validation
   981→   981→        const _exhaustive: never = step.action
   982→   982→        output = { error: `Unknown action: ${_exhaustive}` }
   983→   983→      }
   984→   984→    }
   985→   985→
   986→   986→    return {
   987→   987→      stepId: step.id,
   988→   988→      action: step.action,
   989→   989→      status: 'success',
   990→   990→      output,
   991→   991→      durationMs: Date.now() - start,
   992→   992→    }
   993→   993→  } catch (err: unknown) {
   994→   994→    const msg = err instanceof Error ? err.message : 'Unknown step execution error'
   995→   995→    return {
   996→   996→      stepId: step.id,
   997→   997→      action: step.action,
   998→   998→      status: 'error',
   999→   999→      error: msg,
  1000→  1000→      durationMs: Date.now() - start,
  1001→  1001→    }
  1002→  1002→  }
  1003→  1003→}

  1007→  1007→async function executeStepWithTimeout(
  1008→  1008→  step: ChainStep,
  1009→  1009→  ctx: ExecutionContext,
  1010→  1010→): Promise<StepResult> {
  1011→  1011→  const timeoutMs = step.timeoutMs ?? 30000
  1012→  1012→
  1013→  1013→  return Promise.race([
  1014→  1014→    executeStep(step, ctx),
  1015→  1015→    new Promise<StepResult>((resolve) =>
  1016→  1016→      setTimeout(() => {
  1017→  1017→        resolve({
  1018→  1018→          stepId: step.id,
  1019→  1019→          action: step.action,
  1020→  1020→          status: 'error',
  1021→  1021→          error: `Step timed out after ${timeoutMs}ms`,
  1022→  1022→          durationMs: timeoutMs,
  1023→  1023→        })
  1024→  1024→      }, timeoutMs)
  1025→  1025→    ),
  1026→  1026→  ])
  1027→  1027→}

  1079→  1079→/* ════════════════════════════════════════════════════════════════════════
  1080→  1080→   POST: Create & optionally execute a chained automation workflow
  1081→  1081→   ════════════════════════════════════════════════════════════════════════ */
  1082→  1082→
  1083→  1083→export async function POST(request: NextRequest) {
  1084→  1084→  const overallStart = Date.now()
  1085→  1085→  const visitorId = request.headers.get('x-visitor-id') || null
  1086→  1086→  const ipAddress = request.headers.get('x-forwarded-for') || request.headers.get('x-real-ip') || null
  1087→  1087→  const userAgent = request.headers.get('user-agent') || null
  1088→  1088→
  1089→  1089→  try {
  1090→  1090→    // ── Parse body ───────────────────────────────────────────────────
  1091→  1091→    let body: Record<string, unknown>
  1092→  1092→    try {
  1093→  1093→      body = await request.json()
  1094→  1094→    } catch {
  1095→  1095→      return NextResponse.json(
  1096→  1096→        { error: 'Request body must be valid JSON' },
  1097→  1097→        { status: 400 },
  1098→  1098→      )
  1099→  1099→    }
  1100→  1100→
  1101→  1101→    // ── Validate request ─────────────────────────────────────────────
  1102→  1102→    const validation = validateChainRequest(body)
  1103→  1103→    if (!validation.valid) {
  1104→  1104→      return NextResponse.json(
  1105→  1105→        {
  1106→  1106→          error: 'Validation failed',
  1107→  1107→          details: validation.errors,
  1108→  1108→        },
  1109→  1109→        { status: 400 },
  1110→  1110→      )
  1111→  1111→    }
  1112→  1112→
  1113→  1113→    const {
  1114→  1114→      name,
  1115→  1115→      description = '',
  1116→  1116→      steps,
  1117→  1117→      trigger,
  1118→  1118→      triggerConfig = {},
  1119→  1119→      executeImmediately = false,
  1120→  1120→    } = body as unknown as ChainRequest
  1121→  1121→
  1122→  1122→    // ── Persist the automation rule ──────────────────────────────────
  1123→  1123→    const rule = await prisma.automationRule.create({
  1124→  1124→      data: {
  1125→  1125→        visitorId,
  1126→  1126→        name,
  1127→  1127→        description: description ?? '',
  1128→  1128→        trigger,
  1129→  1129→        triggerConfig: JSON.stringify(triggerConfig),
  1130→  1130→        actions: JSON.stringify(
  1131→  1131→          steps.map(s => ({ id: s.id, action: s.action, config: s.config, condition: s.condition ?? 'always', timeoutMs: s.timeoutMs ?? 30000 })),
  1132→  1132→        ),
  1133→  1133→        isActive: true,
  1134→  1134→        lastStatus: executeImmediately ? 'running' : 'pending',
  1135→  1135→      },

  1138→  1138→    // ── If not executing immediately, return the plan ────────────────

  1140→  1140→      return NextResponse.json(
  1141→  1141→        {
  1142→  1142→          ruleId: rule.id,
  1143→  1143→          name: rule.name,
  1144→  1144→          status: 'saved',
  1145→  1145→          steps: steps.map(s => ({
  1146→  1146→            id: s.id,
  1147→  1147→            action: s.action,
  1148→  1148→            condition: s.condition ?? 'always',
  1149→  1149→            timeoutMs: s.timeoutMs ?? 30000,
  1150→  1150→          })),
  1151→  1151→          trigger,
  1152→  1152→          triggerConfig,
  1153→  1153→          message: 'Chain saved. Set executeImmediately to true to run now.',
  1154→  1154→        },
  1155→  1155→        { status: 201 },
  1156→  1156→      )
  1157→  1157→    }
  1158→  1158→
  1159→  1159→    // ── Execute the chain ────────────────────────────────────────────
  1160→  1160→    const stepResults: StepResult[] = []
  1161→  1161→    const stepResultsMap = new Map<string, StepResult>()
  1162→  1162→    let previousStatus: StepStatus | null = null
  1163→  1163→    let chainError: string | null = null
  1164→  1164→
  1165→  1165→    // Create an initial AutomationLog entry
  1166→  1166→    const automationLog = await prisma.automationLog.create({
  1167→  1167→      data: {
  1168→  1168→        ruleId: rule.id,
  1169→  1169→        status: 'running',
  1170→  1170→        input: JSON.stringify({ trigger, triggerConfig, executedAt: new Date().toISOString() }),
  1171→  1171→        startedAt: new Date(),
  1172→  1172→      },
  1173→  1173→    })
  1174→  1174→
  1175→  1175→    const execCtx: ExecutionContext = {
  1176→  1176→      stepResults: stepResultsMap,
  1177→  1177→      visitorId,
  1178→  1178→      ipAddress,
  1179→  1179→      userAgent,
  1180→  1180→    }
  1181→  1181→
  1182→  1182→    for (let i = 0; i < steps.length; i++) {
  1183→  1183→      const step = steps[i]
  1184→  1184→
  1185→  1185→      // Evaluate step condition
  1186→  1186→      if (!shouldExecuteStep(step, previousStatus)) {
  1187→  1187→        const skipped: StepResult = {
  1188→  1188→          stepId: step.id,
  1189→  1189→          action: step.action,
  1190→  1190→          status: 'skipped',
  1191→  1191→          durationMs: 0,
  1192→  1192→        }
  1193→  1193→        stepResults.push(skipped)
  1194→  1194→        stepResultsMap.set(step.id, skipped)
  1195→  1195→        // Keep previousStatus unchanged for chained conditions
  1196→  1196→        continue
  1197→  1197→      }

  1199→  1199→      // Execute step
  1200→  1200→      const result = await executeStepWithTimeout(step, execCtx)
  1201→  1201→      stepResults.push(result)
  1202→  1202→      stepResultsMap.set(step.id, result)
  1203→  1203→
  1204→  1204→      // Update the running status of the previous step
  1205→  1205→      previousStatus = result.status
  1206→  1206→

  1208→  1208→      if (result.status === 'error') {
  1209→  1209→        chainError = result.error ?? 'Unknown step error'
  1210→  1210→      }
  1211→  1211→    }
  1212→  1212→
  1213→  1213→    // ── Determine overall chain status ───────────────────────────────
  1214→  1214→    const successCount = stepResults.filter(r => r.status === 'success').length
  1215→  1215→    const skippedCount = stepResults.filter(r => r.status === 'skipped').length
  1216→  1216→    const errorCount = stepResults.filter(r => r.status === 'error').length
  1217→  1217→    const overallStatus = errorCount > 0 ? 'error' : 'success'
  1218→  1218→    const totalDurationMs = Date.now() - overallStart
  1219→  1219→
  1220→  1220→    // ── Update the automation rule ───────────────────────────────────
  1221→  1221→    await prisma.automationRule.update({
  1222→  1222→      where: { id: rule.id },
  1223→  1223→      data: {
  1224→  1224→        lastRun: new Date(),
  1225→  1225→        lastStatus: overallStatus,
  1226→  1226→        lastError: chainError,
  1227→  1227→        runCount: { increment: 1 },
  1228→  1228→      },
  1229→  1229→    })
  1230→  1230→
  1231→  1231→    // ── Update the automation log ────────────────────────────────────
  1232→  1232→    await prisma.automationLog.update({
  1233→  1233→      where: { id: automationLog.id },
  1234→  1234→      data: {
  1235→  1235→        status: overallStatus,
  1236→  1236→        output: JSON.stringify({
  1237→  1237→          stepResults: stepResults.map(r => ({
  1238→  1238→            stepId: r.stepId,
  1239→  1239→            action: r.action,
  1240→  1240→            status: r.status,
  1241→  1241→            durationMs: r.durationMs,
  1242→  1242→            error: r.error,
  1243→  1243→          })),
  1244→  1244→          summary: { successCount, skippedCount, errorCount, totalDurationMs },
  1245→  1245→        }),
  1246→  1246→        error: chainError ?? undefined,
  1247→  1247→        completedAt: new Date(),
  1248→  1248→        durationMs: totalDurationMs,
  1249→  1249→      },
  1250→  1250→    })
  1251→  1251→
  1252→  1252→    // ── Audit log ────────────────────────────────────────────────────
  1253→  1253→    await auditLog({
  1254→  1254→      visitorId,
  1255→  1255→      action: 'automation_run',
  1256→  1256→      details: {
  1257→  1257→        context: 'automation_chain',
  1258→  1258→        ruleId: rule.id,
  1259→  1259→        ruleName: name,
  1260→  1260→        trigger,
  1261→  1261→        stepCount: steps.length,
  1262→  1262→        successCount,
  1263→  1263→        skippedCount,
  1264→  1264→        errorCount,
  1265→  1265→      },
  1266→  1266→      inputData: { name, trigger, stepCount: steps.length },
  1267→  1267→      outputData: { overallStatus, totalDurationMs },
  1268→  1268→      ipAddress,
  1269→  1269→      userAgent,
  1270→  1270→      durationMs: totalDurationMs,
  1271→  1271→      error: chainError ?? undefined,
  1272→  1272→    })
  1273→  1273→
  1274→  1274→    // ── Return results ───────────────────────────────────────────────
  1275→  1275→    return NextResponse.json({
  1276→  1276→      ruleId: rule.id,
  1277→  1277→      name,
  1278→  1278→      status: overallStatus,
  1279→  1279→      trigger,
  1280→  1280→      triggerConfig,
  1281→  1281→      steps: stepResults,
  1282→  1282→      summary: {
  1283→  1283→        totalSteps: steps.length,
  1284→  1284→        executed: successCount + errorCount,
  1285→  1285→        succeeded: successCount,
  1286→  1286→        skipped: skippedCount,
  1287→  1287→        failed: errorCount,
  1288→  1288→        totalDurationMs,
  1289→  1289→      },
  1290→  1290→      logId: automationLog.id,
  1291→  1291→    })
  1292→  1292→  } catch (error: unknown) {
  1293→  1293→    const durationMs = Date.now() - overallStart
  1294→  1294→    const errorMsg = error instanceof Error ? error.message : 'Unknown error'
  1295→  1295→
  1296→  1296→    await auditLog({
  1297→  1297→      visitorId,
  1298→  1298→      action: 'automation_run',
  1299→  1299→      details: { context: 'automation_chain', error: true },
  1300→  1300→      ipAddress,
  1301→  1301→      userAgent,
  1302→  1302→      durationMs,
  1303→  1303→      error: errorMsg,
  1304→  1304→    })
  1305→  1305→
  1306→  1306→    return NextResponse.json(
  1307→  1307→      { error: 'Failed to execute automation chain', details: errorMsg },
  1308→  1308→      { status: 500 },
  1309→  1309→    )
  1310→  1310→  }
  1311→  1311→}
