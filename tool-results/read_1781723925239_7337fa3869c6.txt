     9→const ELO_K = 32; // K-factor for volatility

    11→interface ModelEloRecord {
    12→  id: string;
    13→  name: string;
    14→  provider: string;
    15→  elo: number;
    16→  wins: number;
    17→  losses: number;
    18→  ties: number;

    23→function expectedScore(ratingA: number, ratingB: number): number {
    24→  return 1 / (1 + Math.pow(10, (ratingB - ratingA) / 400));
    25→}

    27→function computeLeaderboard(battles: Array<{
    28→  modelAId: string;
    29→  modelAName: string;
    30→  modelBId: string;
    31→  modelBName: string;
    32→  votesA: number;
    33→  votesB: number;
    34→  votesTie: number;
    35→}>): ModelEloRecord[] {
    36→  // Aggregate model records from all battles
    37→  const modelMap = new Map<string, {
    38→    id: string;
    39→    name: string;
    40→    provider: string;
    41→    elo: number;
    42→    wins: number;
    43→    losses: number;
    44→    ties: number;
    45→    battles: number;
    46→  }>();
    47→
    48→  const getOrCreate = (id: string, name: string) => {
    49→    if (!modelMap.has(id)) {
    50→      const provider = inferProvider(name);
    51→      modelMap.set(id, {
    52→        id,
    53→        name,
    54→        provider,
    55→        elo: ELO_DEFAULT,
    56→        wins: 0,
    57→        losses: 0,
    58→        ties: 0,
    59→        battles: 0,
    60→      });
    61→    }
    62→    return modelMap.get(id)!;
    63→  };
    64→
    65→  // Process each battle to accumulate W/L/T and compute ELO
    66→  for (const battle of battles) {
    67→    const modelA = getOrCreate(battle.modelAId, battle.modelAName);
    68→    const modelB = getOrCreate(battle.modelBId, battle.modelBName);
    69→
    70→    modelA.battles += 1;
    71→    modelB.battles += 1;
    72→
    73→    // Determine outcome from votes
    74→    const totalValid = battle.votesA + battle.votesB;
    75→    if (totalValid === 0) {
    76→      // No decisive votes — treat as tie
    77→      modelA.ties += 1;
    78→      modelB.ties += 1;
    79→      const eA = expectedScore(modelA.elo, modelB.elo);
    80→      const eB = expectedScore(modelB.elo, modelA.elo);
    81→      modelA.elo = Math.round(modelA.elo + ELO_K * (0.5 - eA));
    82→      modelB.elo = Math.round(modelB.elo + ELO_K * (0.5 - eB));
    83→    } else if (battle.votesA > battle.votesB) {
    84→      // Model A wins
    85→      modelA.wins += 1;
    86→      modelB.losses += 1;
    87→      const eA = expectedScore(modelA.elo, modelB.elo);
    88→      const eB = expectedScore(modelB.elo, modelA.elo);
    89→      modelA.elo = Math.round(modelA.elo + ELO_K * (1 - eA));
    90→      modelB.elo = Math.round(modelB.elo + ELO_K * (0 - eB));
    91→    } else if (battle.votesB > battle.votesA) {
    92→      // Model B wins
    93→      modelB.wins += 1;
    94→      modelA.losses += 1;
    95→      const eA = expectedScore(modelA.elo, modelB.elo);
    96→      const eB = expectedScore(modelB.elo, modelA.elo);
    97→      modelA.elo = Math.round(modelA.elo + ELO_K * (0 - eA));
    98→      modelB.elo = Math.round(modelB.elo + ELO_K * (1 - eB));
    99→    } else {
   100→      // Exact tie
   101→      modelA.ties += 1;
   102→      modelB.ties += 1;
   103→      const eA = expectedScore(modelA.elo, modelB.elo);
   104→      const eB = expectedScore(modelB.elo, modelA.elo);
   105→      modelA.elo = Math.round(modelA.elo + ELO_K * (0.5 - eA));
   106→      modelB.elo = Math.round(modelB.elo + ELO_K * (0.5 - eB));
   107→    }
   108→  }
   109→
   110→  // Convert to sorted array
   111→  const entries: ModelEloRecord[] = Array.from(modelMap.values()).map((m) => ({
   112→    ...m,
   113→    winRate: m.battles > 0 ? Math.round((m.wins / m.battles) * 100) : 0,
   114→  }));
   115→
   116→  return entries.sort((a, b) => b.elo - a.elo);
   117→}
