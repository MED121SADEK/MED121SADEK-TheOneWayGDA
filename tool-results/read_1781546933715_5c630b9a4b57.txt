   180→   180→function rowCount(data: Record<string, any[]>): number {
   181→   181→  const keys = Object.keys(data)
   182→   182→  if (keys.length === 0) return 0
   183→   183→  return Math.max(...keys.map(k => data[k]?.length ?? 0))
   184→   184→}

   186→   186→// ── localStorage size monitoring ──
   187→   187→const STORAGE_WARNING_THRESHOLD = 3 * 1024 * 1024 // 3MB — warn before 4MB hard limit
   188→   188→let _lastStorageWarning = 0
   189→   189→
   190→   190→export function getStorageUsage(): { usedBytes: number; usedMB: number; percentage: number } {
   191→   191→  if (typeof window === 'undefined') return { usedBytes: 0, usedMB: 0, percentage: 0 }
   192→   192→  let total = 0
   193→   193→  for (let i = 0; i < localStorage.length; i++) {
   194→   194→    const key = localStorage.key(i)
   195→   195→    if (key) total += (localStorage.getItem(key) || '').length
   196→   196→  }
   197→   197→  // Typical browser limit is 5-10MB; use 5MB as reference
   198→   198→  const limit = 5 * 1024 * 1024
   199→   199→  return { usedBytes: total, usedMB: Math.round(total / 1024 / 1024 * 100) / 100, percentage: Math.round(total / limit * 100) }
   200→   200→}
   201→   201→
   202→   202→function checkStorageSize() {
   203→   203→  if (typeof window === 'undefined') return
   204→   204→  const { usedBytes } = getStorageUsage()
   205→   205→  const now = Date.now()
   206→   206→  // Throttle warnings to once per 60 seconds
   207→   207→  if (usedBytes > STORAGE_WARNING_THRESHOLD && now - _lastStorageWarning > 60_000) {
   208→   208→    _lastStorageWarning = now
   209→   209→    const mb = (usedBytes / 1024 / 1024).toFixed(1)
   210→   210→    console.warn(`[STORE] localStorage usage at ${mb}MB — consider exporting old projects and clearing history`)
   211→   211→  }
   212→   212→}

   468→   468→        function parseCsvLine(line: string): string[] {
   469→   469→          const result: string[] = []
   470→   470→          let current = ''
   471→   471→          let inQuotes = false
   472→   472→          for (let i = 0; i < line.length; i++) {
   473→   473→            const ch = line[i]
   474→   474→            if (inQuotes) {
   475→   475→              if (ch === '"' && i + 1 < line.length && line[i + 1] === '"') {
   476→   476→                current += '"'; i++ // escaped quote
   477→   477→              } else if (ch === '"') {
   478→   478→                inQuotes = false
   479→   479→              } else {
   480→   480→                current += ch
   481→   481→              }
   482→   482→            } else {
   483→   483→              if (ch === '"') {
   484→   484→                inQuotes = true
   485→   485→              } else if (ch === ',') {
   486→   486→                result.push(current.trim())
   487→   487→                current = ''
   488→   488→              } else {
   489→   489→                current += ch
   490→   490→              }
   491→   491→            }
   492→   492→          }
   493→   493→          result.push(current.trim())
   494→   494→          return result
   495→   495→        }
