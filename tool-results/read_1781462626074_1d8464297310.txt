   161→   161→/* ─── Panel 2: Data Import ─── */
   162→   162→export function ImportPanel(h: HandlerHook) {
   163→   163→  const fileInputRef = useRef<HTMLInputElement>(null)
   164→   164→  const [dragOver, setDragOver] = useState(false)
   165→   165→  const [importError, setImportError] = useState('')
   166→   166→  const [importSuccess, setImportSuccess] = useState('')
   167→   167→
   168→   168→  const MAX_FILE_SIZE = 50 * 1024 * 1024 // 50MB
   169→   169→  const ACCEPTED_EXTENSIONS = ['.csv', '.xlsx', '.xls', '.json', '.tsv']
   170→   170→
   171→   171→  const validateAndProcessFile = (file: File) => {
   172→   172→    setImportError('')
   173→   173→    setImportSuccess('')
   174→   174→
   175→   175→    // Validate file size
   176→   176→    if (file.size > MAX_FILE_SIZE) {
   177→   177→      setImportError(`File too large (${(file.size / 1024 / 1024).toFixed(1)}MB). Maximum is 50MB.`)
   178→   178→      return
   179→   179→    }
   180→   180→    if (file.size === 0) {
   181→   181→      setImportError('File is empty.')
   182→   182→      return
   183→   183→    }
   184→   184→
   185→   185→    // Validate extension
   186→   186→    const ext = '.' + file.name.split('.').pop()?.toLowerCase()
   187→   187→    if (!ACCEPTED_EXTENSIONS.includes(ext)) {
   188→   188→      setImportError(`Unsupported file type "${ext}". Accepted: ${ACCEPTED_EXTENSIONS.join(', ')}`)
   189→   189→      return
   190→   190→    }
   191→   191→
   192→   192→    try {
   193→   193→      h.store.importFile(file)
   194→   194→      setImportSuccess(`Imported "${file.name}" (${(file.size / 1024).toFixed(1)}KB)`)
   195→   195→    } catch (err) {
   196→   196→      setImportError(`Import failed: ${err instanceof Error ? err.message : 'Unknown error'}`)
   197→   197→    }
   198→   198→  }
   199→   199→
   200→   200→  const handleDrop = (e: React.DragEvent) => {
   201→   201→    e.preventDefault()
   202→   202→    e.stopPropagation()
   203→   203→    setDragOver(false)
   204→   204→    const files = Array.from(e.dataTransfer.files)
   205→   205→    if (files.length === 0) return
   206→   206→    validateAndProcessFile(files[0])
   207→   207→  }
   208→   208→
   209→   209→  const handleDragOver = (e: React.DragEvent) => {
   210→   210→    e.preventDefault()
   211→   211→    e.stopPropagation()
   212→   212→    setDragOver(true)
   213→   213→  }
   214→   214→
   215→   215→  const handleDragLeave = (e: React.DragEvent) => {
   216→   216→    e.preventDefault()
   217→   217→    e.stopPropagation()
   218→   218→    setDragOver(false)
   219→   219→  }
   220→   220→
   221→   221→  return (
   222→   222→    <div className="p-4 space-y-4 overflow-y-auto h-full scrollbar-thin" style={{ contain: 'layout style paint' }}>
   223→   223→      <div
   224→   224→        className={cn(
   225→   225→          'border-2 border-dashed rounded-xl p-8 text-center cursor-pointer transition-all',
   226→   226→          dragOver ? 'border-teal-400 bg-teal-500/10 scale-[1.01]' : 'border-border hover:border-teal-400/50 hover:bg-teal-500/5'
   227→   227→        )}
   228→   228→        onClick={() => fileInputRef.current?.click()}
   229→   229→        onDrop={handleDrop}
   230→   230→        onDragOver={handleDragOver}
   231→   231→        onDragLeave={handleDragLeave}
   232→   232→      >
   233→   233→        <input
   234→   234→          ref={fileInputRef}
   235→   235→          type="file"
   236→   236→          accept=".csv,.xlsx,.xls,.json,.tsv"
   237→   237→          className="hidden"
   238→   238→          onChange={(e) => {
   239→   239→            const file = e.target.files?.[0]
   240→   240→            if (file) validateAndProcessFile(file)
   241→   241→            // Reset input so same file can be re-selected
   242→   242→            if (fileInputRef.current) fileInputRef.current.value = ''
   243→   243→          }}
   244→   244→        />
   245→   245→        {dragOver ? (
   246→   246→          <>
   247→   247→            <Download className="w-8 h-8 mx-auto mb-3 text-teal-400" />
   248→   248→            <p className="text-sm font-medium text-teal-400">Drop your file here</p>
   249→   249→          </>
   250→   250→        ) : (
   251→   251→          <>
   252→   252→            <Upload className="w-8 h-8 mx-auto mb-3 text-teal-400" />
   253→   253→            <p className="text-sm font-medium text-foreground">{h.t('import.dragDrop')}</p>
   254→   254→            <p className="text-xs text-muted-foreground mt-1">{h.t('import.supportedFormats')}</p>
   255→   255→            <p className="text-[10px] text-muted-foreground/60 mt-1">Max 50MB</p>
   256→   256→          </>
   257→   257→        )}
   258→   258→      </div>
   259→   259→
   260→   260→      {importError && (
   261→   261→        <div className="flex items-start gap-2 bg-red-500/10 border border-red-500/20 rounded-lg p-2.5 text-[10px] text-red-400">
   262→   262→          <X className="w-3 h-3 mt-0.5 flex-shrink-0" />
   263→   263→          <div>
   264→   264→            <p className="font-medium">Import Error</p>
   265→   265→            <p>{importError}</p>
   266→   266→          </div>
   267→   267→          <button onClick={() => setImportError('')} className="ml-auto flex-shrink-0 hover:text-red-300">
   268→   268→            <X className="w-3 h-3" />
   269→   269→          </button>
   270→   270→        </div>
   271→   271→      )}
   272→   272→      {importSuccess && (
   273→   273→        <div className="flex items-center gap-2 bg-emerald-500/10 border border-emerald-500/20 rounded-lg p-2.5 text-[10px] text-emerald-400">
   274→   274→          <Check className="w-3 h-3" />
   275→   275→          <span>{importSuccess}</span>
   276→   276→        </div>
   277→   277→      )}
   278→   278→
   279→   279→      <Separator />
   280→   280→
   281→   281→      <div className="space-y-2">
   282→   282→        <p className="text-xs font-semibold text-foreground">Paste CSV Data</p>
   283→   283→        <Textarea
   284→   284→          value={h.importText}
   285→   285→          onChange={(e) => h.setImportText(e.target.value)}
   286→   286→          placeholder="Name,Age,Score&#10;Alice,25,89&#10;Bob,30,92"
   287→   287→          className="text-xs min-h-24 font-mono"
   288→   288→        />
   289→   289→        <Button size="sm" className="w-full bg-teal-600 hover:bg-teal-700 text-white text-xs" onClick={h.handleImportCSV} disabled={!h.importText.trim()}>
   290→   290→          <Database className="w-3.5 h-3.5 mr-1.5" />
   291→   291→          Import CSV
   292→   292→        </Button>
   293→   293→      </div>
   294→   294→
   295→   295→      <Separator />
   296→   296→
   297→   297→      <div className="space-y-2">
   298→   298→        <p className="text-xs font-semibold text-foreground">Quick Templates</p>
   299→   299→        <div className="grid grid-cols-2 gap-2">
   300→   300→          {[
   301→   301→            { name: 'Sample Survey', data: 'Respondent,Age,Gender,Satisfaction,Income\n1,25,Male,4,45000\n2,30,Female,5,52000\n3,22,Male,3,38000\n4,35,Female,4,61000\n5,28,Male,5,48000\n6,40,Female,3,72000\n7,33,Male,4,55000\n8,27,Female,5,42000' },
   302→   302→            { name: 'Student Grades', data: 'Student,Math,Science,English,History\nAlice,85,92,88,76\nBob,78,85,90,82\nCarol,92,88,76,95\nDave,65,72,88,70\nEve,88,95,82,90' },
   303→   303→            { name: 'Sales Data', data: 'Month,Revenue,Profit,Customers\nJan,45000,12000,320\nFeb,52000,15000,380\nMar,48000,11000,350\nApr,61000,18000,420\nMay,55000,14000,390' },
   304→   304→          ].map((tpl) => (
   305→   305→            <Button

   319→   319→  )

   934→   934→/* ─── Panel 7: Scan & OCR ─── */
   935→   935→export function ScanPanel(h: HandlerHook) {
   936→   936→  const [dragOver, setDragOver] = useState(false)
   937→   937→  const [scanError, setScanError] = useState('')
   938→   938→
   939→   939→  const MAX_SCAN_SIZE = 20 * 1024 * 1024 // 20MB
   940→   940→  const SCAN_ACCEPTED = ['.pdf', '.png', '.jpg', '.jpeg', '.webp', '.bmp', '.tiff']
   941→   941→
   942→   942→  const validateScanFile = (file: File): string | null => {
   943→   943→    if (file.size > MAX_SCAN_SIZE) return `File too large (${(file.size / 1024 / 1024).toFixed(1)}MB). Max 20MB.`
   944→   944→    if (file.size === 0) return 'File is empty.'
   945→   945→    const ext = '.' + file.name.split('.').pop()?.toLowerCase()
   946→   946→    if (!SCAN_ACCEPTED.includes(ext)) return `Unsupported type. Accepted: ${SCAN_ACCEPTED.join(', ')}`
   947→   947→    return null
   948→   948→  }
   949→   949→
   950→   950→  const processScanFile = (file: File) => {
   951→   951→    setScanError('')
   952→   952→    const err = validateScanFile(file)
   953→   953→    if (err) { setScanError(err); return }
   954→   954→    h.setScanFile(file)
   955→   955→    const reader = new FileReader()
   956→   956→    reader.onload = (ev) => h.setScanPreview(ev.target?.result as string)
   957→   957→    reader.onerror = () => setScanError('Failed to read file.')
   958→   958→    reader.readAsDataURL(file)
   959→   959→  }
   960→   960→
   961→   961→  const handleScanDrop = (e: React.DragEvent) => {
   962→   962→    e.preventDefault(); e.stopPropagation(); setDragOver(false)
   963→   963→    const files = Array.from(e.dataTransfer.files)
   964→   964→    if (files.length > 0) processScanFile(files[0])
   965→   965→  }
   966→   966→
   967→   967→  const handleScanDragOver = (e: React.DragEvent) => { e.preventDefault(); e.stopPropagation(); setDragOver(true) }
   968→   968→  const handleScanDragLeave = (e: React.DragEvent) => { e.preventDefault(); e.stopPropagation(); setDragOver(false) }
   969→   969→
   970→   970→  const handleScanSubmit = async () => {
   971→   971→    if (!h.scanFile) return
   972→   972→    setScanError('')
   973→   973→    h.store.setScanState('processing')
   974→   974→
   975→   975→    const formData = new FormData()
   976→   976→    formData.append('file', h.scanFile)
   977→   977→
   978→   978→    if (h.scanFile.name.toLowerCase().endsWith('.pdf')) {
   979→   979→      formData.append('templateHint', 'PDF document')
   980→   980→    }
   981→   981→
   982→   982→    try {
   983→   983→      const res = await fetch('/api/scan', { method: 'POST', body: formData })
   984→   984→      const data = await res.json()
   985→   985→
   986→   986→      if (!res.ok) {
   987→   987→        setScanError(data.error || `Scan failed (HTTP ${res.status})`)
   988→   988→        h.store.setScanState('error')
   989→   989→        return
   990→   990→      }
   991→   991→
   992→   992→      if (data.fields && data.fields.length > 0) {
   993→   993→        h.store.setScanResults(data as any)
   994→   994→      } else if (data.rawText && data.rawText.length > 10) {
   995→   995→        h.store.setScanResults({
   996→   996→          ...data,
   997→   997→          fields: [{ label: 'Raw Text', value: data.rawText.slice(0, 500), confidence: 0.5, type: 'string' }],
   998→   998→          summary: data.summary || 'Document text extracted. Review the raw text below.',
   999→   999→        } as any)
  1000→  1000→      } else {
  1001→  1001→        setScanError('Could not extract structured data from this document. The content may be unclear, handwritten, or image-based. Try uploading a clearer scan or a different file format.')
  1002→  1002→        h.store.setScanState('error')
  1003→  1003→      }
  1004→  1004→    } catch (err) {
  1005→  1005→      setScanError(`Network error: ${err instanceof Error ? err.message : 'Connection failed'}`)
  1006→  1006→      h.store.setScanState('error')
  1007→  1007→    }
  1008→  1008→  }
  1009→  1009→
  1010→  1010→  return (
  1011→  1011→    <div className="p-4 space-y-4 overflow-y-auto h-full scrollbar-thin" style={{ contain: 'layout style paint' }}>
  1012→  1012→      <div
  1013→  1013→        className={cn(
  1014→  1014→          'border-2 border-dashed rounded-xl p-8 text-center cursor-pointer transition-all',
  1015→  1015→          h.scanFile ? 'border-cyan-400 bg-cyan-50/10' : dragOver ? 'border-cyan-400 bg-cyan-500/10 scale-[1.01]' : 'border-border hover:border-cyan-400/50 hover:bg-cyan-500/5'
  1016→  1016→        )}
  1017→  1017→        onClick={() => h.fileInputRef.current?.click()}
  1018→  1018→        onDrop={handleScanDrop}
  1019→  1019→        onDragOver={handleScanDragOver}
  1020→  1020→        onDragLeave={handleScanDragLeave}
  1021→  1021→      >
  1022→  1022→        <input
  1023→  1023→          ref={h.fileInputRef}
  1024→  1024→          type="file"
  1025→  1025→          accept=".pdf,.png,.jpg,.jpeg,.webp,.bmp,.tiff"
  1026→  1026→          className="hidden"
  1027→  1027→          onChange={(e) => {
  1028→  1028→            const f = e.target.files?.[0]
  1029→  1029→            if (f) processScanFile(f)
  1030→  1030→            if (h.fileInputRef.current) h.fileInputRef.current.value = ''
  1031→  1031→          }}
  1032→  1032→        />
  1033→  1033→        {h.store.scanState === 'processing' ? (
  1034→  1034→          <div className="flex flex-col items-center gap-2">
  1035→  1035→            <div className="w-8 h-8 border-2 border-cyan-500 border-t-transparent rounded-full animate-spin" />
  1036→  1036→            <p className="text-sm text-muted-foreground">Analyzing document with AI...</p>
  1037→  1037→            <p className="text-[10px] text-muted-foreground/60">This may take 10-30 seconds for complex documents</p>
  1038→  1038→          </div>
  1039→  1039→        ) : h.scanPreview ? (
  1040→  1040→          <div className="flex flex-col items-center gap-2">
  1041→  1041→            {/* eslint-disable-next-line @next/next/no-img-element */}
  1042→  1042→            <img src={h.scanPreview} alt="Preview" className="max-h-40 rounded-lg shadow-sm" />
  1043→  1043→            <p className="text-[10px] text-muted-foreground">{h.scanFile?.name} ({((h.scanFile?.size || 0) / 1024).toFixed(1)}KB)</p>
  1044→  1044→          </div>
  1045→  1045→        ) : (
  1046→  1046→          <>
  1047→  1047→            <ScanLine className="w-8 h-8 mx-auto mb-2 text-cyan-400" />
  1048→  1048→            <p className="text-sm font-medium">{h.t('scan.dragDrop')}</p>
  1049→  1049→            <p className="text-[10px] text-muted-foreground mt-1">{h.t('scan.supportedFormats')}</p>
  1050→  1050→            <p className="text-[10px] text-muted-foreground/60 mt-1">Max 20MB - Drag and drop or click</p>
  1051→  1051→          </>
  1052→  1052→        )}
  1053→  1053→      </div>
  1054→  1054→
  1055→  1055→      {scanError && (
  1056→  1056→        <div className="flex items-start gap-2 bg-red-500/10 border border-red-500/20 rounded-lg p-2.5 text-[10px] text-red-400">
