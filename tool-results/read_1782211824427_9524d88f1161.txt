   183→function uid(): string {
   184→  return Date.now().toString(36) + Math.random().toString(36).slice(2)

   187→function rowCount(data: Record<string, any[]>): number {
   188→  const keys = Object.keys(data)
   189→  if (keys.length === 0) return 0
   190→  return Math.max(...keys.map(k => data[k]?.length ?? 0))
   191→}

   193→// ── localStorage size monitoring ──
   194→const STORAGE_WARNING_THRESHOLD = 3 * 1024 * 1024 // 3MB — warn before 4MB hard limit
   195→let _lastStorageWarning = 0
   196→
   197→export function getStorageUsage(): { usedBytes: number; usedMB: number; percentage: number } {
   198→  if (typeof window === 'undefined') return { usedBytes: 0, usedMB: 0, percentage: 0 }
   199→  let total = 0
   200→  for (let i = 0; i < localStorage.length; i++) {
   201→    const key = localStorage.key(i)
   202→    if (key) total += (localStorage.getItem(key) || '').length
   203→  }
   204→  // Typical browser limit is 5-10MB; use 5MB as reference
   205→  const limit = 5 * 1024 * 1024
   206→  return { usedBytes: total, usedMB: Math.round(total / 1024 / 1024 * 100) / 100, percentage: Math.round(total / limit * 100) }
   207→}
   208→
   209→function checkStorageSize() {
   210→  if (typeof window === 'undefined') return
   211→  const { usedBytes } = getStorageUsage()
   212→  const now = Date.now()
   213→  // Throttle warnings to once per 60 seconds
   214→  if (usedBytes > STORAGE_WARNING_THRESHOLD && now - _lastStorageWarning > 60_000) {
   215→    _lastStorageWarning = now
   216→    const mb = (usedBytes / 1024 / 1024).toFixed(1)
   217→    console.warn(`[STORE] localStorage usage at ${mb}MB — consider exporting old projects and clearing history`)
   218→  }
   219→}

   566→
   567→        function parseCsvLine(line: string): string[] {
   568→          const result: string[] = []
   569→          let current = ''
   570→          let inQuotes = false
   571→          for (let i = 0; i < line.length; i++) {
   572→            const ch = line[i]
   573→            if (inQuotes) {
   574→              if (ch === '"' && i + 1 < line.length && line[i + 1] === '"') {
   575→                current += '"'; i++ // escaped quote
   576→              } else if (ch === '"') {
   577→                inQuotes = false
   578→              } else {
   579→                current += ch
   580→              }
   581→            } else {
   582→              if (ch === '"') {
   583→                inQuotes = true
   584→              } else if (ch === ',') {
   585→                result.push(current.trim())
   586→                current = ''
   587→              } else {
   588→                current += ch
   589→              }
   590→            }
   591→          }
   592→          result.push(current.trim())
   593→          return result
   594→        }
   595→
   596→        const headers = parseCsvLine(lines[0])
   597→        const data: Record<string, any[]> = {}
   598→        const variables: Variable[] = []
   599→
   600→        headers.forEach(name => {
   601→          data[name] = []
   602→          variables.push({
   603→            id: uid(), name, type: 'numeric', label: name, width: 8, decimals: 2, missing: '', values: {},
   604→          })
   605→        })
   606→
   607→        for (let i = 1; i < lines.length; i++) {
   608→          const line = lines[i].trim()
   609→          if (!line) continue
   610→          const vals = parseCsvLine(line)
   611→          headers.forEach((h, j) => {
   612→            const raw = vals[j] || ''
   613→            const num = parseFloat(raw)
   614→            data[h].push(isNaN(num) || raw === '' ? raw : num)
   615→          })
   616→        }
   617→
   618→        // Ensure all columns have same length
   619→        const maxLen = Math.max(...Object.values(data).map(a => a.length), 0)
   620→        for (const key of Object.keys(data)) {
   621→          while (data[key].length < maxLen) data[key].push('')
   622→        }
   623→
   624→        variables.forEach(v => {
   625→          const col = data[v.name]

   740→              scheduleAutoSave()
   741→            } catch (err) {
   742→              console.error('JSON import error:', err)
   743→            }
   744→          }
   745→          reader.readAsText(file)

   762→        // Proper CSV escaping: wrap fields containing commas, quotes, or newlines
   763→        const esc = (val: unknown): string => {
   764→          const s = val === null || val === undefined ? '' : String(val)
   765→          if (s.includes(',') || s.includes('"') || s.includes('\n') || s.includes('\r')) {
   766→            return '"' + s.replace(/"/g, '""') + '"'
   767→          }
   768→          return s
   769→        }
   770→        let csv = names.map(esc).join(',') + '\n'
   771→        for (let i = 0; i < n; i++) {
   772→          csv += names.map(name => esc(data[name]?.[i])).join(',') + '\n'
   773→        }

   809→        const newVariables: Variable[] = []
   810→        const newData: Record<string, any[]> = {}
   811→
   812→        // Create variables and data from extracted fields
   813→        for (const field of results.fields) {
   814→          const varName = field.label
   815→            .toLowerCase()
   816→            .replace(/[^a-z0-9]+/g, '_')

   824→            field.type === 'numeric' || field.type === 'currency' || field.type === 'percentage'
