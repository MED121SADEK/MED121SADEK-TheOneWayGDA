    12→// ─── Live Data: Fetch real benchmark data from the database ───
    13→async function fetchLiveBenchmarkData(): Promise<string> {
    14→  try {
    15→    const models = await prisma.aiModel.findMany({
    16→      where: { isActive: true },
    17→      include: {
    18→        benchmarkScores: true,
    19→        pricing: { where: { isActive: true }, take: 1 },
    20→      },
    21→      orderBy: { name: 'asc' },
    22→    })
    23→
    24→    if (models.length === 0) return ''
    25→
    26→    const benchmarkNames = [...new Set(models.flatMap(m => m.benchmarkScores.map(b => b.benchmark)))]
    27→    const pricingMap: Record<string, { input: number; output: number }> = {}
    28→
    29→    // Build benchmark comparison table
    30→    const header = '| Model | ' + benchmarkNames.map(b => b).join(' | ') + ' | Price (in/out per 1M) |'
    31→    const separator = '|---|' + benchmarkNames.map(() => '---:').join('|') + '|---:|'
    32→
    33→    const rows = models.map(model => {
    34→      const scores = benchmarkNames.map(bench => {
    35→        const score = model.benchmarkScores.find(s => s.benchmark === bench)
    36→        return score ? `${score.score}/${score.maxScore}` : '-'
    37→      })
    38→      const price = model.pricing[0]
    39→      const priceStr = price ? `$${price.inputPrice}/$${price.outputPrice}` : '-'
    40→      pricingMap[model.name] = price ? { input: price.inputPrice, output: price.outputPrice } : { input: 0, output: 0 }
    41→      return `| ${model.name} (${model.provider}) | ${scores.join(' | ')} | ${priceStr} |`
    42→    }).join('\n')
    43→
    44→    // Find top model per benchmark
    45→    const topPerBenchmark = benchmarkNames.map(bench => {
    46→      const best = models
    47→        .map(m => ({ name: m.name, score: m.benchmarkScores.find(s => s.benchmark === bench)?.score || 0 }))
    48→        .sort((a, b) => b.score - a.score)[0]
    49→      return `  - ${bench}: ${best.name} (${best.score})`
    50→    }).join('\n')
    51→
    52→    return `
    53→## LIVE LEADERBOARD DATA (from database, not hallucinated)
    54→${header}
    55→${separator}
    56→${rows}
    57→
    58→### Top Models per Benchmark
    59→${topPerBenchmark}
    60→
    61→### Pricing Reference
    62→${Object.entries(pricingMap).map(([name, p]) => `  - ${name}: $${p.input}/$${p.output} per 1M tokens`).join('\n')}
    63→
    64→> CRITICAL: Always use these REAL numbers when comparing models. Never fabricate benchmark scores. If you don't have data for a specific model, say so explicitly.`
    65→  } catch {
    66→    return ''
    67→  }
    68→}

    70→// ─── Conversation History: Fetch recent conversation for continuity ───
    71→async function fetchConversationHistory(visitorId: string | null | undefined, currentSessionId?: string): Promise<string> {
    72→  if (!visitorId) return ''
    73→  try {
    74→    const recentConversations = await prisma.aiConversation.findMany({
    75→      where: { visitorId },
    76→      orderBy: { updatedAt: 'desc' },
    77→      take: 3,
    78→      select: { id: true, sessionId: true, context: true, messages: true, updatedAt: true },
    79→    })
    80→
    81→    if (recentConversations.length === 0) return ''
    82→
    83→    const parts: string[] = ['### Recent Conversations (for continuity)']
    84→    for (const conv of recentConversations) {
    85→      if (conv.id === currentSessionId) continue // Skip current session
    86→      try {
    87→        const msgs = JSON.parse(conv.messages) as Array<{ role: string; content: string }>
    88→        const lastExchange = msgs.slice(-4) // Last 2 exchanges
    89→        if (lastExchange.length > 0) {
    90→          const summary = lastExchange.map(m => `${m.role}: ${m.content.slice(0, 150)}...`).join('\n  ')
    91→          parts.push(`- [${conv.context}, ${conv.updatedAt.toISOString().slice(0, 10)}]:\n  ${summary}`)
    92→        }
    93→      } catch {
    94→        // Skip malformed conversations
    95→      }
    96→    }
    97→
    98→    return parts.length > 1 ? '\n\n[CONVERSATION HISTORY]\n' + parts.join('\n') + '\nUse this to maintain conversation continuity — reference past topics naturally when relevant.' : ''
    99→  } catch {
   100→    return ''
   101→  }
   102→}

   489→// Benchmark data changes only when models/scores update. Cache for 15 minutes.
   490→const _benchmarkCache = { data: '', fetchedAt: 0, TTL: 15 * 60 * 1000 }
   491→async function getCachedBenchmarkData(): Promise<string> {
   492→  const now = Date.now()
   493→  if (_benchmarkCache.data && now - _benchmarkCache.fetchedAt < _benchmarkCache.TTL) {
   494→    return _benchmarkCache.data
   495→  }
   496→  _benchmarkCache.data = await fetchLiveBenchmarkData()
   497→  _benchmarkCache.fetchedAt = now
   498→  return _benchmarkCache.data
   499→}

   501→// Memory context: cache per visitor for 5 minutes (user profile rarely changes)
   502→const _memoryCache = new Map<string, { data: string; fetchedAt: number }>()
   503→const MEMORY_TTL = 5 * 60 * 1000
   504→async function getCachedMemoryContext(visitorId: string | null | undefined): Promise<string> {
   505→  if (!visitorId) return ''
   506→  const cached = _memoryCache.get(visitorId)
   507→  if (cached && Date.now() - cached.fetchedAt < MEMORY_TTL) return cached.data
   508→  const data = await fetchMemoryContext(visitorId)
   509→  _memoryCache.set(visitorId, { data, fetchedAt: Date.now() })
   510→  // Cleanup old entries every 50 calls
   511→  if (_memoryCache.size > 500) {
   512→    const now = Date.now()
   513→    for (const [k, v] of _memoryCache) {
   514→      if (now - v.fetchedAt > MEMORY_TTL) _memoryCache.delete(k)
   515→    }
   516→  }
   517→  return data
   518→}

   520→export async function POST(request: NextRequest) {
   521→  const startTime = Date.now()
   522→  const user = await requireAuth(request)
   523→  const visitorId = user?.id || request.headers.get('x-visitor-id') || null
   524→  const ipAddress = request.headers.get('x-forwarded-for') || request.headers.get('x-real-ip') || null
   525→  const userAgent = request.headers.get('user-agent') || null
   526→
   527→  try {
   528→    const body = await request.json()
   529→    const { messages, context = 'general', pageData, stream = false } = body
   530→
   531→    if (!messages || !Array.isArray(messages) || messages.length === 0) {
   532→      return NextResponse.json({ error: 'Messages array is required' }, { status: 400 })
   533→    }
   534→
   535→    const zai = await ZAI.create()
   536→    const sessionId = body.sessionId || `session_${Date.now()}`
   537→
   538→    // Smart context injection: only include expensive data when relevant
   539→    const isFirstMessage = messages.length <= 1
   540→    const shouldIncludeLiveData = context === 'leaderboard' || context === 'general'
   541→    const shouldIncludeMemory = isFirstMessage
   542→    const shouldIncludeHistory = isFirstMessage && messages.length === 1 // skip for single-shot
   543→
   544→    // ─── Parallel data fetch (cached internally) ───
   545→    const [baseSystemPrompt, liveData, memoryContext, conversationHistory] = await Promise.all([
   546→      Promise.resolve(getBaseSystemPrompt(context, pageData)),
   547→      shouldIncludeLiveData ? getCachedBenchmarkData() : Promise.resolve(''),
   548→      shouldIncludeMemory ? getCachedMemoryContext(visitorId) : Promise.resolve(''),
   549→      shouldIncludeHistory ? fetchConversationHistory(visitorId, sessionId) : Promise.resolve(''),
   550→    ])
   551→
   552→    // Assemble the full system prompt with all context layers
   553→    const systemPrompt = baseSystemPrompt
   554→      + (liveData ? '\n\n' + liveData : '')
   555→      + memoryContext
   556→      + conversationHistory
   557→
   558→    // Build user messages preserving conversation flow
   559→    const userMessages = messages.map((m: { role: string; content: string }) => ({
   560→      role: m.role as 'user' | 'assistant',
   561→      content: m.content,
   562→    }))
   563→
   564→    // ─── Streaming mode ───
   565→    if (stream) {
   566→      const encoder = new TextEncoder()
   567→      const streamResponse = new ReadableStream({
   568→        async start(controller) {
   569→          try {
   570→            const completion = await zai.chat.completions.create({
   571→              messages: [
   572→                { role: 'system', content: systemPrompt },
   573→                ...userMessages,
   574→              ],
   575→              max_tokens: 8192,
   576→              temperature: 0.7,
   577→              stream: true as unknown as boolean,
   578→            } as any)
   579→
   580→            if (completion && typeof completion === 'object') {
   581→              const streamable = completion as any
   582→              if (streamable.body && typeof streamable.body.getReader === 'function') {
   583→                const reader = streamable.body.getReader()
   584→                while (true) {
   585→                  const { done, value } = await reader.read()
   586→                  if (done) break
   587→                  controller.enqueue(value)
   588→                }
   589→              } else if (Array.isArray(streamable.choices)) {
   590→                const text = streamable.choices[0]?.message?.content || ''
   591→                controller.enqueue(encoder.encode(`data: ${JSON.stringify({ choices: [{ delta: { content: text } }] })}\n\n`))
   592→              }
   593→            }
   594→
   595→            controller.enqueue(encoder.encode('data: [DONE]\n\n'))
   596→            controller.close()
   597→          } catch (error) {
   598→            const errorMsg = error instanceof Error ? error.message : 'Stream error'
   599→            controller.enqueue(encoder.encode(`data: ${JSON.stringify({ error: errorMsg })}\n\n`))
   600→            controller.close()
   601→          }
   602→        },
   603→      })
   604→
   605→      // Fire-and-forget: audit log + conversation save
   606→      const lastUserMsg = messages[messages.length - 1]?.content || ''
   607→      ;(async () => {
   608→        try {
   609→          await prisma.aiAuditLog.create({
   610→            data: {
   611→              visitorId: visitorId || null,
   612→              action: 'ai_query',
   613→              details: JSON.stringify({ context, messageCount: messages.length, mode: 'stream', version: 'v4', withLiveData: !!liveData, withHistory: !!conversationHistory }),
   614→              inputData: JSON.stringify({ lastUserMessage: lastUserMsg.slice(0, 1000) }),
   615→              durationMs: Date.now() - startTime,
   616→              ipAddress,
   617→              userAgent,
   618→            },
   619→          })
   620→        } catch { /* silent */ }
   621→      })()
   622→
   623→      return new Response(streamResponse, {
   624→        headers: {
   625→          'Content-Type': 'text/event-stream',
   626→          'Cache-Control': 'no-cache',
   627→          'Connection': 'keep-alive',
   628→        },
   629→      })
   630→    }
   631→
   632→    // ─── Standard mode (deep response) ───
   633→    const completion = await zai.chat.completions.create({
   634→      messages: [
   635→        { role: 'system', content: systemPrompt },
   636→        ...userMessages,
   637→      ],
   638→      max_tokens: 8192,
   639→      temperature: 0.7,
   640→    })
   641→
   642→    const aiMessage = completion.choices?.[0]?.message?.content || 'I apologize, I could not generate a response.'
   643→    const tokensUsed = completion.usage?.total_tokens || 0
   644→    const durationMs = Date.now() - startTime
   645→
   646→    // Audit log
   647→    try {
   648→      await prisma.aiAuditLog.create({
   649→        data: {
   650→          visitorId: visitorId || null,
   651→          action: 'ai_query',
   652→          details: JSON.stringify({
   653→            context, messageCount: messages.length, mode: 'standard', version: 'v4',
   654→            withLiveData: !!liveData, withMemory: !!memoryContext, withHistory: !!conversationHistory,
   655→          }),
   656→          inputData: JSON.stringify({ lastUserMessage: messages[messages.length - 1]?.content?.slice(0, 1000) }),
   657→          outputData: JSON.stringify({ responseLength: aiMessage.length, tokensUsed }),
   658→          tokensUsed,
   659→          durationMs,
   660→          ipAddress,
   661→          userAgent,
   662→        },
   663→      })
   664→    } catch {
   665→      // Audit log failure should not block the response
   666→    }
   667→
   668→    // Save conversation with proper history
   669→    try {
   670→      await prisma.aiConversation.upsert({
   671→        where: { id: sessionId },
   672→        create: {
   673→          id: sessionId,
   674→          sessionId,
   675→          visitorId: visitorId || null,
   676→          context,
   677→          messages: JSON.stringify([...messages, { role: 'assistant', content: aiMessage, timestamp: new Date().toISOString() }]),
   678→        },
   679→        update: {
   680→          messages: JSON.stringify([...messages, { role: 'assistant', content: aiMessage, timestamp: new Date().toISOString() }]),
   681→          updatedAt: new Date(),
   682→        },
   683→      })
   684→    } catch {
   685→      // Conversation save failure should not block the response
   686→    }
   687→
   688→    return NextResponse.json({
   689→      message: aiMessage,
   690→      meta: {
   691→        tokensUsed,
   692→        durationMs,
   693→        context,
   694→        version: 'v4',
   695→        contextLayers: {
   696→          liveData: !!liveData,
   697→          memory: !!memoryContext,
   698→          conversationHistory: !!conversationHistory,
   699→        },
   700→      },
   701→    })
   702→  } catch (error: unknown) {
   703→    const durationMs = Date.now() - startTime
   704→    const errorMsg = error instanceof Error ? error.message : 'Unknown error'
   705→
   706→    try {
   707→      await prisma.aiAuditLog.create({
   708→        data: {
   709→          visitorId: visitorId || null,
   710→          action: 'ai_query',
   711→          details: JSON.stringify({ context: 'general', error: true, version: 'v4' }),
   712→          error: errorMsg,
   713→          durationMs,
   714→          ipAddress,
   715→          userAgent,
   716→        },
   717→      })
   718→    } catch { /* silent */ }
   719→
   720→    return NextResponse.json({ error: 'Failed to get AI response' }, { status: 500 })
   721→  }
   722→}

   724→export async function GET(request: NextRequest) {
   725→  try {
   726→    const { searchParams } = new URL(request.url)
   727→    const context = searchParams.get('context') || 'general'
   728→    const visitorId = searchParams.get('visitorId')
   729→
   730→    const suggestions = await prisma.aiSuggestion.findMany({
   731→      where: {
   732→        context,
   733→        ...(visitorId ? { visitorId } : {}),
   734→        isDismissed: false,
   735→        createdAt: { gte: new Date(Date.now() - 24 * 60 * 60 * 1000) },
   736→      },
   737→      orderBy: { createdAt: 'desc' },
   738→      take: 5,
   739→    })
   740→
   741→    return NextResponse.json({ suggestions })
   742→  } catch (error: unknown) {
   743→    return NextResponse.json({ error: 'Internal server error' }, { status: 500 })
   744→  }
   745→}
