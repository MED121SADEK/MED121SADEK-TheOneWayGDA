   446→// ─── Validation ─────────────────────────────────────────────────────────
   447→
   448→interface ValidationError {
   449→  stepId?: string
   450→  field: string
   451→  message: string
   452→}
   453→
   454→function validateActionType(action: string): action is ActionType {
   455→  return (ALL_ACTION_TYPES as readonly string[]).includes(action)
   456→}
   457→
   458→function validateTriggerType(trigger: string): trigger is TriggerType {
   459→  return (ALL_TRIGGER_TYPES as readonly string[]).includes(trigger)
   460→}
   461→
   462→function validateStepConfig(step: ChainStep, index: number): ValidationError[] {
   463→  const errors: ValidationError[] = []
   464→  const prefix = step.id || `steps[${index}]`
   465→  const cfg = step.config as unknown as Record<string, unknown>
   466→
   467→  switch (step.action) {
   468→    case 'clean_data': {
   469→      const strat = cfg.strategy as string | undefined
   470→      if (!strat || !['auto', 'standard', 'aggressive'].includes(strat)) {
   471→        errors.push({ stepId: prefix, field: 'config.strategy', message: `Invalid strategy "${strat}". Expected: auto | standard | aggressive` })
   472→      }
   473→      if (cfg.columns !== undefined && !Array.isArray(cfg.columns)) {
   474→        errors.push({ stepId: prefix, field: 'config.columns', message: 'columns must be a string array' })
   475→      }
   476→      break
   477→    }
   478→
   479→    case 'run_model': {
   480→      const model = cfg.model as string | undefined
   481→      const validModels = ['descriptive-stats', 'regression', 'random-forest', 'anomaly-detection', 'classification', 'clustering', 'time-series', 'neural-net']
   482→      if (!model || !validModels.includes(model)) {
   483→        errors.push({ stepId: prefix, field: 'config.model', message: `Invalid model "${model}". Expected one of: ${validModels.join(', ')}` })
   484→      }
   485→      break
   486→    }
   487→
   488→    case 'generate_report': {
   489→      const fmt = cfg.format as string | undefined
   490→      if (!fmt || !['pdf', 'html', 'csv', 'json'].includes(fmt)) {
   491→        errors.push({ stepId: prefix, field: 'config.format', message: `Invalid format "${fmt}". Expected: pdf | html | csv | json` })
   492→      }
   493→      break
   494→    }
   495→
   496→    case 'send_notification': {
   497→      const ch = cfg.channel as string | undefined
   498→      if (!ch || !['email', 'push', 'webhook', 'slack', 'teams'].includes(ch)) {
   499→        errors.push({ stepId: prefix, field: 'config.channel', message: `Invalid channel "${ch}". Expected: email | push | webhook | slack | teams` })
   500→      }
   501→      const rec = cfg.recipients as string | undefined
   502→      if (!rec || typeof rec !== 'string' || rec.trim().length === 0) {
   503→        errors.push({ stepId: prefix, field: 'config.recipients', message: 'recipients is required (team, admin, self, or an email)' })
   504→      }
   505→      break
   506→    }
   507→
   508→    case 'feature_engineering': {
   509→      const method = cfg.method as string | undefined
   510→      if (!method || !['auto', 'pca', 'polynomial', 'interaction', 'encoding'].includes(method)) {
   511→        errors.push({ stepId: prefix, field: 'config.method', message: `Invalid method "${method}". Expected: auto | pca | polynomial | interaction | encoding` })
   512→      }
   513→      break
   514→    }
   515→
   516→    case 'data_validation': {
   517→      if (!Array.isArray(cfg.rules) || cfg.rules.length === 0) {
   518→        errors.push({ stepId: prefix, field: 'config.rules', message: 'rules must be a non-empty array' })
   519→      } else {
   520→        const validRuleTypes = ['not_null', 'range', 'regex', 'unique']
   521→        for (let r = 0; r < cfg.rules.length; r++) {
   522→          const rule = cfg.rules[r] as Record<string, unknown>
   523→          if (!rule.column || typeof rule.column !== 'string') {
   524→            errors.push({ stepId: prefix, field: `config.rules[${r}].column`, message: 'Each rule must have a string column' })
   525→          }
   526→          if (!rule.type || !validRuleTypes.includes(rule.type as string)) {
   527→            errors.push({ stepId: prefix, field: `config.rules[${r}].type`, message: `Invalid rule type. Expected: ${validRuleTypes.join(' | ')}` })
   528→          }
   529→        }
   530→      }
   531→      break
   532→    }
   533→
   534→    case 'data_transform': {
   535→      if (!Array.isArray(cfg.operations) || cfg.operations.length === 0) {
   536→        errors.push({ stepId: prefix, field: 'config.operations', message: 'operations must be a non-empty array' })
   537→      } else {
   538→        const validOpTypes = ['normalize', 'standardize', 'log', 'bin', 'encode']
   539→        for (let o = 0; o < cfg.operations.length; o++) {
   540→          const op = cfg.operations[o] as Record<string, unknown>
   541→          if (!op.column || typeof op.column !== 'string') {
   542→            errors.push({ stepId: prefix, field: `config.operations[${o}].column`, message: 'Each operation must have a string column' })
   543→          }
   544→          if (!op.type || !validOpTypes.includes(op.type as string)) {
   545→            errors.push({ stepId: prefix, field: `config.operations[${o}].type`, message: `Invalid operation type. Expected: ${validOpTypes.join(' | ')}` })
   546→          }
   547→        }
   548→      }
   549→      break
   550→    }
   551→
   552→    case 'model_evaluation': {
   553→      if (!Array.isArray(cfg.metrics) || cfg.metrics.length === 0) {
   554→        errors.push({ stepId: prefix, field: 'config.metrics', message: 'metrics must be a non-empty array' })
   555→      } else {
   556→        const validMetrics = ['accuracy', 'precision', 'recall', 'f1', 'auc', 'rmse', 'mae']
   557→        for (const m of cfg.metrics) {
   558→          if (!validMetrics.includes(m as string)) {
   559→            errors.push({ stepId: prefix, field: 'config.metrics', message: `Invalid metric "${m}". Expected: ${validMetrics.join(', ')}` })
   560→          }
   561→        }
   562→      }
   563→      if (cfg.threshold !== undefined && (typeof cfg.threshold !== 'number' || cfg.threshold < 0 || cfg.threshold > 1)) {
   564→        errors.push({ stepId: prefix, field: 'config.threshold', message: 'threshold must be a number between 0 and 1' })
   565→      }
   566→      break
   567→    }
   568→
   569→    case 'export_data': {
   570→      const fmt = cfg.format as string | undefined
   571→      if (!fmt || !['csv', 'json', 'parquet', 'excel'].includes(fmt)) {
   572→        errors.push({ stepId: prefix, field: 'config.format', message: `Invalid format "${fmt}". Expected: csv | json | parquet | excel` })
   573→      }
   574→      const dest = cfg.destination as string | undefined
   575→      if (!dest || !['download', 's3', 'database', 'api'].includes(dest)) {
   576→        errors.push({ stepId: prefix, field: 'config.destination', message: `Invalid destination "${dest}". Expected: download | s3 | database | api` })
   577→      }
   578→      break
   579→    }
   580→
   581→    case 'webhook_call': {
   582→      const url = cfg.url as string | undefined
   583→      if (!url || typeof url !== 'string' || url.trim().length === 0) {
   584→        errors.push({ stepId: prefix, field: 'config.url', message: 'url is required and must be a valid URL string' })
   585→      } else {
   586→        try {
   587→          new URL(url)
   588→        } catch {
   589→          errors.push({ stepId: prefix, field: 'config.url', message: `Invalid URL: "${url}"` })
   590→        }
   591→      }
   592→      const method = cfg.method as string | undefined
   593→      if (!method || !['GET', 'POST', 'PUT'].includes(method)) {
   594→        errors.push({ stepId: prefix, field: 'config.method', message: `Invalid method "${method}". Expected: GET | POST | PUT` })
   595→      }
   596→      break
   597→    }
   598→
   599→    case 'conditional_logic': {
   600→      const cond = cfg.condition as string | undefined
   601→      if (!cond || typeof cond !== 'string' || cond.trim().length === 0) {
   602→        errors.push({ stepId: prefix, field: 'config.condition', message: 'condition expression is required' })
   603→      }
   604→      if (!Array.isArray(cfg.onTrue)) {
   605→        errors.push({ stepId: prefix, field: 'config.onTrue', message: 'onTrue must be an array of step IDs' })
   606→      }
   607→      if (!Array.isArray(cfg.onFalse)) {
   608→        errors.push({ stepId: prefix, field: 'config.onFalse', message: 'onFalse must be an array of step IDs' })
   609→      }
   610→      break
   611→    }
   612→
   613→    case 'ai_analysis': {
   614→      const prompt = cfg.prompt as string | undefined
   615→      if (!prompt || typeof prompt !== 'string' || prompt.trim().length === 0) {
   616→        errors.push({ stepId: prefix, field: 'config.prompt', message: 'prompt is required' })
   617→      }
   618→      if (cfg.maxTokens !== undefined && (typeof cfg.maxTokens !== 'number' || cfg.maxTokens < 1 || cfg.maxTokens > 16384)) {
   619→        errors.push({ stepId: prefix, field: 'config.maxTokens', message: 'maxTokens must be a number between 1 and 16384' })
   620→      }
   621→      break
   622→    }
   623→  }
   624→
   625→  return errors
   626→}
   627→
   628→function validateChainRequest(body: Record<string, unknown>): { valid: boolean; errors: ValidationError[] } {
   629→  const errors: ValidationError[] = []
   630→
   631→  // name
   632→  const name = body.name as string | undefined
   633→  if (!name || typeof name !== 'string' || name.trim().length === 0) {
   634→    errors.push({ field: 'name', message: 'name is required' })
   635→  } else if (name.length > 200) {
   636→    errors.push({ field: 'name', message: 'name must be 200 characters or fewer' })
   637→  }
   638→
   639→  // steps
   640→  const steps = body.steps as ChainStep[] | undefined
   641→  if (!Array.isArray(steps) || steps.length === 0) {
   642→    errors.push({ field: 'steps', message: 'steps must be a non-empty array' })
   643→  } else if (steps.length > 100) {
   644→    errors.push({ field: 'steps', message: 'A chain may contain at most 100 steps' })
   645→  } else {
   646→    // Check for duplicate step IDs
   647→    const seenIds = new Set<string>()
   648→    for (let i = 0; i < steps.length; i++) {
   649→      const step = steps[i]
   650→      if (!step.id || typeof step.id !== 'string' || step.id.trim().length === 0) {
   651→        errors.push({ stepId: step.id || undefined, field: `steps[${i}].id`, message: 'Each step must have a non-empty id' })
   652→      } else if (seenIds.has(step.id)) {
   653→        errors.push({ stepId: step.id, field: `steps[${i}].id`, message: `Duplicate step id: "${step.id}"` })
   654→      } else {
   655→        seenIds.add(step.id)
   656→      }
   657→
   658→      // Validate action type
   659→      if (!step.action || !validateActionType(step.action)) {
   660→        errors.push({ stepId: step.id, field: `steps[${i}].action`, message: `Invalid action type "${step.action}". Valid types: ${ALL_ACTION_TYPES.join(', ')}` })
   661→      } else {
   662→        // Validate step config
   663→        errors.push(...validateStepConfig(step, i))
   664→      }
   665→
   666→      // Validate condition if provided
   667→      if (step.condition !== undefined) {
   668→        const cond = step.condition as string
   669→        const validBaseConditions = ['always', 'on_success', 'on_failure']
   670→        if (!validBaseConditions.includes(cond) && !cond.startsWith('expression:')) {
   671→          errors.push({ stepId: step.id, field: `steps[${i}].condition`, message: `Invalid condition "${cond}". Use: always | on_success | on_failure | expression:<expr>` })
   672→        }
   673→      }
   674→    }
   675→  }
   676→
   677→  // trigger
   678→  const trigger = body.trigger as string | undefined
   679→  if (!trigger || !validateTriggerType(trigger)) {
   680→    errors.push({ field: 'trigger', message: `Invalid trigger type "${trigger}". Expected: ${ALL_TRIGGER_TYPES.join(', ')}` })
   681→  }
   682→
   683→  // executeImmediately
   684→  if (body.executeImmediately !== undefined && typeof body.executeImmediately !== 'boolean') {
   685→    errors.push({ field: 'executeImmediately', message: 'executeImmediately must be a boolean' })
   686→  }
   687→
   688→  return { valid: errors.length === 0, errors }
   689→}

   691→// ─── Condition evaluation ───────────────────────────────────────────────
   692→
   693→function shouldExecuteStep(
   694→  step: ChainStep,
   695→  previousStatus: StepStatus | null,
   696→): boolean {
   697→  const condition: string = step.condition ?? 'always'
   698→
   699→  // First step always runs (unless condition explicitly says otherwise)
   700→  if (previousStatus === null) {
   701→    return condition === 'always' || condition === 'on_success'
   702→  }
   703→
   704→  switch (condition) {
   705→    case 'always':
   706→      return true
   707→    case 'on_success':
   708→      return previousStatus === 'success'
   709→    case 'on_failure':
   710→      return previousStatus === 'error'
   711→    default:
   712→      // Expression-based condition — evaluate against step results context
   713→      // Format: "expression:<javascript-like condition>"
   714→      if (condition.startsWith('expression:')) {
   715→        // In a production system, this would use a safe expression evaluator.
   716→        // For now, expression conditions always evaluate to true so the step
   717→        // runs and the output can be inspected.  The expression is recorded
   718→        // for observability.
   719→        return true
   720→      }
   721→      return true
   722→  }
   723→}

   965→            status: result.status,
   966→            output: result.output,
   967→          }

   970→          condition: clCfg.condition,

   975→        }
   976→        break
   977→      }
   978→
   979→      default: {
   980→        // Exhaustiveness check — should never reach here due to validation
   981→        const _exhaustive: never = step.action
   982→        output = { error: `Unknown action: ${_exhaustive}` }
   983→      }
   984→    }
   985→
   986→    return {
   987→      stepId: step.id,
   988→      action: step.action,
   989→      status: 'success',
   990→      output,
   991→      durationMs: Date.now() - start,
   992→    }
   993→  } catch (err: unknown) {
   994→    const msg = err instanceof Error ? err.message : 'Unknown step execution error'
   995→    return {
   996→      stepId: step.id,
   997→      action: step.action,
   998→      status: 'error',
   999→      error: msg,
  1000→      durationMs: Date.now() - start,
  1001→    }
  1002→  }
  1003→}

  1080→   POST: Create & optionally execute a chained automation workflow
  1081→   ════════════════════════════════════════════════════════════════════════ */
  1082→
  1083→export async function POST(request: NextRequest) {
  1084→  const overallStart = Date.now()
  1085→  const visitorId = request.headers.get('x-visitor-id') || null
  1086→  const ipAddress = request.headers.get('x-forwarded-for') || request.headers.get('x-real-ip') || null
  1087→  const userAgent = request.headers.get('user-agent') || null
  1088→
  1089→  try {
  1090→    // ── Parse body ───────────────────────────────────────────────────
  1091→    let body: Record<string, unknown>
  1092→    try {
  1093→      body = await request.json()
  1094→    } catch {
  1095→      return NextResponse.json(
  1096→        { error: 'Request body must be valid JSON' },
  1097→        { status: 400 },
  1098→      )
  1099→    }
  1100→
  1101→    // ── Validate request ─────────────────────────────────────────────
  1102→    const validation = validateChainRequest(body)
  1103→    if (!validation.valid) {
  1104→      return NextResponse.json(
  1105→        {
  1106→          error: 'Validation failed',
  1107→          details: validation.errors,
  1108→        },
  1109→        { status: 400 },
  1110→      )
  1111→    }
  1112→
  1113→    const {
  1114→      name,
  1115→      description = '',
  1116→      steps,
  1117→      trigger,
  1118→      triggerConfig = {},
  1119→      executeImmediately = false,
  1120→    } = body as unknown as ChainRequest
  1121→
  1122→    // ── Persist the automation rule ──────────────────────────────────
  1123→    const rule = await prisma.automationRule.create({
  1124→      data: {
  1125→        visitorId,
  1126→        name,
  1127→        description: description ?? '',
  1128→        trigger,
  1129→        triggerConfig: JSON.stringify(triggerConfig),
  1130→        actions: JSON.stringify(
  1131→          steps.map(s => ({ id: s.id, action: s.action, config: s.config, condition: s.condition ?? 'always', timeoutMs: s.timeoutMs ?? 30000 })),
  1132→        ),
  1133→        isActive: true,
  1134→        lastStatus: executeImmediately ? 'running' : 'pending',
  1135→      },
  1136→    })
  1137→
  1138→    // ── If not executing immediately, return the plan ────────────────
  1139→    if (!executeImmediately) {
  1140→      return NextResponse.json(
  1141→        {
  1142→          ruleId: rule.id,
  1143→          name: rule.name,
  1144→          status: 'saved',
  1145→          steps: steps.map(s => ({
  1146→            id: s.id,
  1147→            action: s.action,
  1148→            condition: s.condition ?? 'always',
  1149→            timeoutMs: s.timeoutMs ?? 30000,
  1150→          })),
  1151→          trigger,
  1152→          triggerConfig,
  1153→          message: 'Chain saved. Set executeImmediately to true to run now.',
  1154→        },
  1155→        { status: 201 },
  1156→      )
  1157→    }
  1158→
  1159→    // ── Execute the chain ────────────────────────────────────────────
  1160→    const stepResults: StepResult[] = []
  1161→    const stepResultsMap = new Map<string, StepResult>()
  1162→    let previousStatus: StepStatus | null = null
  1163→    let chainError: string | null = null
  1164→
  1165→    // Create an initial AutomationLog entry
  1166→    const automationLog = await prisma.automationLog.create({
  1167→      data: {
  1168→        ruleId: rule.id,
  1169→        status: 'running',
  1170→        input: JSON.stringify({ trigger, triggerConfig, executedAt: new Date().toISOString() }),
  1171→        startedAt: new Date(),
  1172→      },
  1173→    })
  1174→
  1175→    const execCtx: ExecutionContext = {
  1176→      stepResults: stepResultsMap,
  1177→      visitorId,
  1178→      ipAddress,
  1179→      userAgent,
  1180→    }
  1181→
  1182→    for (let i = 0; i < steps.length; i++) {
  1183→      const step = steps[i]
  1184→
  1185→      // Evaluate step condition
  1186→      if (!shouldExecuteStep(step, previousStatus)) {
  1187→        const skipped: StepResult = {
  1188→          stepId: step.id,
  1189→          action: step.action,
  1190→          status: 'skipped',
  1191→          durationMs: 0,
  1192→        }
  1193→        stepResults.push(skipped)
  1194→        stepResultsMap.set(step.id, skipped)
  1195→        // Keep previousStatus unchanged for chained conditions
  1196→        continue
  1197→      }
  1198→
  1199→      // Execute step
  1200→      const result = await executeStepWithTimeout(step, execCtx)
  1201→      stepResults.push(result)
  1202→      stepResultsMap.set(step.id, result)
  1203→
  1204→      // Update the running status of the previous step
  1205→      previousStatus = result.status
  1206→
  1207→      // If a step fails, we continue — later steps decide via condition
  1208→      if (result.status === 'error') {
  1209→        chainError = result.error ?? 'Unknown step error'
  1210→      }
  1211→    }
  1212→
  1213→    // ── Determine overall chain status ───────────────────────────────
  1214→    const successCount = stepResults.filter(r => r.status === 'success').length
  1215→    const skippedCount = stepResults.filter(r => r.status === 'skipped').length
  1216→    const errorCount = stepResults.filter(r => r.status === 'error').length
  1217→    const overallStatus = errorCount > 0 ? 'error' : 'success'
  1218→    const totalDurationMs = Date.now() - overallStart
  1219→
  1220→    // ── Update the automation rule ───────────────────────────────────
  1221→    await prisma.automationRule.update({
  1222→      where: { id: rule.id },
  1223→      data: {
  1224→        lastRun: new Date(),
  1225→        lastStatus: overallStatus,
  1226→        lastError: chainError,
  1227→        runCount: { increment: 1 },
  1228→      },
  1229→    })
  1230→
  1231→    // ── Update the automation log ────────────────────────────────────
  1232→    await prisma.automationLog.update({
  1233→      where: { id: automationLog.id },
  1234→      data: {
  1235→        status: overallStatus,
  1236→        output: JSON.stringify({
  1237→          stepResults: stepResults.map(r => ({
  1238→            stepId: r.stepId,
  1239→            action: r.action,
  1240→            status: r.status,
  1241→            durationMs: r.durationMs,
  1242→            error: r.error,
  1243→          })),
  1244→          summary: { successCount, skippedCount, errorCount, totalDurationMs },
  1245→        }),
  1246→        error: chainError ?? undefined,
  1247→        completedAt: new Date(),
  1248→        durationMs: totalDurationMs,
  1249→      },
  1250→    })
  1251→
  1252→    // ── Audit log ────────────────────────────────────────────────────
  1253→    await auditLog({
  1254→      visitorId,
  1255→      action: 'automation_run',
  1256→      details: {
  1257→        context: 'automation_chain',
  1258→        ruleId: rule.id,
  1259→        ruleName: name,
  1260→        trigger,
  1261→        stepCount: steps.length,
  1262→        successCount,
  1263→        skippedCount,
  1264→        errorCount,
  1265→      },
  1266→      inputData: { name, trigger, stepCount: steps.length },
  1267→      outputData: { overallStatus, totalDurationMs },
  1268→      ipAddress,
  1269→      userAgent,
  1270→      durationMs: totalDurationMs,
  1271→      error: chainError ?? undefined,
  1272→    })
  1273→
  1274→    // ── Return results ───────────────────────────────────────────────
  1275→    return NextResponse.json({
  1276→      ruleId: rule.id,
  1277→      name,
  1278→      status: overallStatus,
  1279→      trigger,
  1280→      triggerConfig,
  1281→      steps: stepResults,
  1282→      summary: {
  1283→        totalSteps: steps.length,
  1284→        executed: successCount + errorCount,
  1285→        succeeded: successCount,
  1286→        skipped: skippedCount,
  1287→        failed: errorCount,
  1288→        totalDurationMs,
  1289→      },
  1290→      logId: automationLog.id,
  1291→    })
  1292→  } catch (error: unknown) {
  1293→    const durationMs = Date.now() - overallStart
  1294→    const errorMsg = error instanceof Error ? error.message : 'Unknown error'
  1295→
  1296→    await auditLog({
  1297→      visitorId,
  1298→      action: 'automation_run',
  1299→      details: { context: 'automation_chain', error: true },
  1300→      ipAddress,
  1301→      userAgent,
  1302→      durationMs,
  1303→      error: errorMsg,
  1304→    })
  1305→
  1306→    return NextResponse.json(
  1307→      { error: 'Failed to execute automation chain', details: errorMsg },
  1308→      { status: 500 },
  1309→    )
  1310→  }
  1311→}
