    82→    82→function timeAgo(date: string): string {
    83→    83→  const now = Date.now()
    84→    84→  const d = new Date(date).getTime()
    85→    85→  const diff = now - d
    86→    86→  const mins = Math.floor(diff / 60000)
    87→    87→  if (mins < 1) return 'just now'
    88→    88→  if (mins < 60) return `${mins}m ago`
    89→    89→  const hrs = Math.floor(mins / 60)
    90→    90→  if (hrs < 24) return `${hrs}h ago`
    91→    91→  const days = Math.floor(hrs / 24)
    92→    92→  if (days < 7) return `${days}d ago`
    93→    93→  return new Date(date).toLocaleDateString()
    94→    94→}

   117→   117→    default:
