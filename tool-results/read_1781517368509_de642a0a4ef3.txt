    11→// ─── Live Data: Fetch real benchmark data from the database ───
    12→async function fetchLiveBenchmarkData(): Promise<string> {
    13→  try {
    14→    const models = await prisma.aiModel.findMany({
    15→      where: { isActive: true },
    16→      include: {
    17→        benchmarkScores: true,
    18→        pricing: { where: { isActive: true }, take: 1 },
    19→      },
    20→      orderBy: { name: 'asc' },
    21→    })
    22→
    23→    if (models.length === 0) return ''
    24→
    25→    const benchmarkNames = [...new Set(models.flatMap(m => m.benchmarkScores.map(b => b.benchmark)))]
    26→    const pricingMap: Record<string, { input: number; output: number }> = {}
    27→
    28→    // Build benchmark comparison table
    29→    const header = '| Model | ' + benchmarkNames.map(b => b).join(' | ') + ' | Price (in/out per 1M) |'
    30→    const separator = '|---|' + benchmarkNames.map(() => '---:').join('|') + '|---:|'
    31→
    32→    const rows = models.map(model => {
    33→      const scores = benchmarkNames.map(bench => {
    34→        const score = model.benchmarkScores.find(s => s.benchmark === bench)
    35→        return score ? `${score.score}/${score.maxScore}` : '-'
    36→      })
    37→      const price = model.pricing[0]
    38→      const priceStr = price ? `$${price.inputPrice}/$${price.outputPrice}` : '-'
    39→      pricingMap[model.name] = price ? { input: price.inputPrice, output: price.outputPrice } : { input: 0, output: 0 }
    40→      return `| ${model.name} (${model.provider}) | ${scores.join(' | ')} | ${priceStr} |`
    41→    }).join('\n')
    42→
    43→    // Find top model per benchmark
    44→    const topPerBenchmark = benchmarkNames.map(bench => {
    45→      const best = models
    46→        .map(m => ({ name: m.name, score: m.benchmarkScores.find(s => s.benchmark === bench)?.score || 0 }))
    47→        .sort((a, b) => b.score - a.score)[0]
    48→      return `  - ${bench}: ${best.name} (${best.score})`
    49→    }).join('\n')
    50→
    51→    return `
    52→## LIVE LEADERBOARD DATA (from database, not hallucinated)
    53→${header}
    54→${separator}
    55→${rows}
    56→
    57→### Top Models per Benchmark
    58→${topPerBenchmark}
    59→
    60→### Pricing Reference
    61→${Object.entries(pricingMap).map(([name, p]) => `  - ${name}: $${p.input}/$${p.output} per 1M tokens`).join('\n')}
    62→
    63→> CRITICAL: Always use these REAL numbers when comparing models. Never fabricate benchmark scores. If you don't have data for a specific model, say so explicitly.`
    64→  } catch {
    65→    return ''
    66→  }
    67→}

    69→// ─── Conversation History: Fetch recent conversation for continuity ───
    70→async function fetchConversationHistory(visitorId: string | null | undefined, currentSessionId?: string): Promise<string> {
    71→  if (!visitorId) return ''
    72→  try {
    73→    const recentConversations = await prisma.aiConversation.findMany({
    74→      where: { visitorId },
    75→      orderBy: { updatedAt: 'desc' },
    76→      take: 3,
    77→      select: { id: true, sessionId: true, context: true, messages: true, updatedAt: true },
    78→    })
    79→
    80→    if (recentConversations.length === 0) return ''
    81→
    82→    const parts: string[] = ['### Recent Conversations (for continuity)']
    83→    for (const conv of recentConversations) {
    84→      if (conv.id === currentSessionId) continue // Skip current session
    85→      try {
    86→        const msgs = JSON.parse(conv.messages) as Array<{ role: string; content: string }>
    87→        const lastExchange = msgs.slice(-4) // Last 2 exchanges
    88→        if (lastExchange.length > 0) {
    89→          const summary = lastExchange.map(m => `${m.role}: ${m.content.slice(0, 150)}...`).join('\n  ')
    90→          parts.push(`- [${conv.context}, ${conv.updatedAt.toISOString().slice(0, 10)}]:\n  ${summary}`)
    91→        }
    92→      } catch {
    93→        // Skip malformed conversations
    94→      }
    95→    }
    96→
    97→    return parts.length > 1 ? '\n\n[CONVERSATION HISTORY]\n' + parts.join('\n') + '\nUse this to maintain conversation continuity — reference past topics naturally when relevant.' : ''
    98→  } catch {
    99→    return ''
   100→  }
   101→}

   480→export async function POST(request: NextRequest) {
   481→  const startTime = Date.now()
   482→  let visitorId = request.headers.get('x-visitor-id')
   483→  const ipAddress = request.headers.get('x-forwarded-for') || request.headers.get('x-real-ip') || null
   484→  const userAgent = request.headers.get('user-agent') || null
   485→
   486→  try {
   487→    const body = await request.json()
   488→    const { messages, context = 'general', pageData, stream = false } = body
   489→
   490→    if (!messages || !Array.isArray(messages) || messages.length === 0) {
   491→      return NextResponse.json({ error: 'Messages array is required' }, { status: 400 })
   492→    }
   493→
   494→    const zai = await ZAI.create()
   495→    const sessionId = body.sessionId || `session_${Date.now()}`
   496→
   497→    // ─── Parallel data fetch for maximum context ───
   498→    const [baseSystemPrompt, liveData, memoryContext, conversationHistory] = await Promise.all([
   499→      Promise.resolve(getSystemPrompt(context, pageData)),
   500→      fetchLiveBenchmarkData(),
   501→      fetchMemoryContext(visitorId),
   502→      fetchConversationHistory(visitorId, sessionId),
   503→    ])
   504→
   505→    // Assemble the full system prompt with all context layers
   506→    const systemPrompt = baseSystemPrompt
   507→      + (liveData ? '\n\n' + liveData : '')
   508→      + memoryContext
   509→      + conversationHistory
   510→
   511→    // Build user messages preserving conversation flow
   512→    const userMessages = messages.map((m: { role: string; content: string }) => ({
   513→      role: m.role as 'user' | 'assistant',
   514→      content: m.content,
   515→    }))
   516→
   517→    // ─── Streaming mode ───
   518→    if (stream) {
   519→      const encoder = new TextEncoder()
   520→      const streamResponse = new ReadableStream({
   521→        async start(controller) {
   522→          try {
   523→            const completion = await zai.chat.completions.create({
   524→              messages: [
   525→                { role: 'system', content: systemPrompt },
   526→                ...userMessages,
   527→              ],
   528→              max_tokens: 8192,
   529→              temperature: 0.7,
   530→              stream: true as unknown as boolean,
   531→            } as any)
   532→
   533→            if (completion && typeof completion === 'object') {
   534→              const streamable = completion as any
   535→              if (streamable.body && typeof streamable.body.getReader === 'function') {
   536→                const reader = streamable.body.getReader()
   537→                while (true) {
   538→                  const { done, value } = await reader.read()
   539→                  if (done) break
   540→                  controller.enqueue(value)
   541→                }
   542→              } else if (Array.isArray(streamable.choices)) {
   543→                const text = streamable.choices[0]?.message?.content || ''
   544→                controller.enqueue(encoder.encode(`data: ${JSON.stringify({ choices: [{ delta: { content: text } }] })}\n\n`))
   545→              }
   546→            }
   547→
   548→            controller.enqueue(encoder.encode('data: [DONE]\n\n'))
   549→            controller.close()
   550→          } catch (error) {
   551→            const errorMsg = error instanceof Error ? error.message : 'Stream error'
   552→            controller.enqueue(encoder.encode(`data: ${JSON.stringify({ error: errorMsg })}\n\n`))
   553→            controller.close()
   554→          }
   555→        },
   556→      })
   557→
   558→      // Fire-and-forget: audit log + conversation save
   559→      const lastUserMsg = messages[messages.length - 1]?.content || ''
   560→      ;(async () => {
   561→        try {
   562→          await prisma.aiAuditLog.create({
   563→            data: {
   564→              visitorId: visitorId || null,
   565→              action: 'ai_query',
   566→              details: JSON.stringify({ context, messageCount: messages.length, mode: 'stream', version: 'v4', withLiveData: !!liveData, withHistory: !!conversationHistory }),
   567→              inputData: JSON.stringify({ lastUserMessage: lastUserMsg.slice(0, 1000) }),
   568→              durationMs: Date.now() - startTime,
   569→              ipAddress,
   570→              userAgent,
   571→            },
   572→          })
   573→        } catch { /* silent */ }
   574→      })()
   575→
   576→      return new Response(streamResponse, {
   577→        headers: {
   578→          'Content-Type': 'text/event-stream',
   579→          'Cache-Control': 'no-cache',
   580→          'Connection': 'keep-alive',
   581→        },
   582→      })
   583→    }
   584→
   585→    // ─── Standard mode (deep response) ───
   586→    const completion = await zai.chat.completions.create({
   587→      messages: [
   588→        { role: 'system', content: systemPrompt },
   589→        ...userMessages,
   590→      ],
   591→      max_tokens: 8192,
   592→      temperature: 0.7,
   593→    })
   594→
   595→    const aiMessage = completion.choices?.[0]?.message?.content || 'I apologize, I could not generate a response.'
   596→    const tokensUsed = completion.usage?.total_tokens || 0
   597→    const durationMs = Date.now() - startTime
   598→
   599→    // Audit log
   600→    try {
   601→      await prisma.aiAuditLog.create({
   602→        data: {
   603→          visitorId: visitorId || null,
   604→          action: 'ai_query',
   605→          details: JSON.stringify({
   606→            context, messageCount: messages.length, mode: 'standard', version: 'v4',
   607→            withLiveData: !!liveData, withMemory: !!memoryContext, withHistory: !!conversationHistory,
   608→          }),
   609→          inputData: JSON.stringify({ lastUserMessage: messages[messages.length - 1]?.content?.slice(0, 1000) }),
   610→          outputData: JSON.stringify({ responseLength: aiMessage.length, tokensUsed }),
   611→          tokensUsed,
   612→          durationMs,
   613→          ipAddress,
   614→          userAgent,
   615→        },
   616→      })
   617→    } catch {
   618→      // Audit log failure should not block the response
   619→    }
   620→
   621→    // Save conversation with proper history
   622→    try {
   623→      await prisma.aiConversation.upsert({
   624→        where: { id: sessionId },
   625→        create: {
   626→          id: sessionId,
   627→          sessionId,
   628→          visitorId: visitorId || null,
   629→          context,
   630→          messages: JSON.stringify([...messages, { role: 'assistant', content: aiMessage, timestamp: new Date().toISOString() }]),
   631→        },
   632→        update: {
   633→          messages: JSON.stringify([...messages, { role: 'assistant', content: aiMessage, timestamp: new Date().toISOString() }]),
   634→          updatedAt: new Date(),
   635→        },
   636→      })
   637→    } catch {
   638→      // Conversation save failure should not block the response
   639→    }
   640→
   641→    return NextResponse.json({
   642→      message: aiMessage,
   643→      meta: {
   644→        tokensUsed,
   645→        durationMs,
   646→        context,
   647→        version: 'v4',
   648→        contextLayers: {
   649→          liveData: !!liveData,
   650→          memory: !!memoryContext,
   651→          conversationHistory: !!conversationHistory,
   652→        },
   653→      },
   654→    })
   655→  } catch (error: unknown) {
   656→    const durationMs = Date.now() - startTime
   657→    const errorMsg = error instanceof Error ? error.message : 'Unknown error'
   658→
   659→    try {
   660→      await prisma.aiAuditLog.create({
   661→        data: {
   662→          visitorId: visitorId || null,
   663→          action: 'ai_query',
   664→          details: JSON.stringify({ context: 'general', error: true, version: 'v4' }),
   665→          error: errorMsg,
   666→          durationMs,
   667→          ipAddress,
   668→          userAgent,
   669→        },
   670→      })
   671→    } catch { /* silent */ }
   672→
   673→    return NextResponse.json({ error: 'Failed to get AI response' }, { status: 500 })
   674→  }
   675→}

   677→export async function GET(request: NextRequest) {
   678→  try {
   679→    const { searchParams } = new URL(request.url)
   680→    const context = searchParams.get('context') || 'general'
   681→    const visitorId = searchParams.get('visitorId')
   682→
   683→    const suggestions = await prisma.aiSuggestion.findMany({
   684→      where: {
   685→        context,
   686→        ...(visitorId ? { visitorId } : {}),
   687→        isDismissed: false,
   688→        createdAt: { gte: new Date(Date.now() - 24 * 60 * 60 * 1000) },
   689→      },
   690→      orderBy: { createdAt: 'desc' },
   691→      take: 5,
   692→    })
   693→
   694→    return NextResponse.json({ suggestions })
   695→  } catch (error: unknown) {
   696→    return NextResponse.json({ error: 'Internal server error' }, { status: 500 })
   697→  }
   698→}
