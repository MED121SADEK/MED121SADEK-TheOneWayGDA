   151→/* ─── Panel 2: Data Import ─── */
   152→export function ImportPanel(h: HandlerHook) {
   153→  const fileInputRef = useRef<HTMLInputElement>(null)
   154→  const [dragOver, setDragOver] = useState(false)
   155→  const [importError, setImportError] = useState('')
   156→  const [importSuccess, setImportSuccess] = useState('')
   157→
   158→  const MAX_FILE_SIZE = 50 * 1024 * 1024 // 50MB
   159→  const ACCEPTED_EXTENSIONS = ['.csv', '.xlsx', '.xls', '.json', '.tsv']
   160→
   161→  const validateAndProcessFile = (file: File) => {
   162→    setImportError('')
   163→    setImportSuccess('')
   164→
   165→    // Validate file size
   166→    if (file.size > MAX_FILE_SIZE) {
   167→      setImportError(`File too large (${(file.size / 1024 / 1024).toFixed(1)}MB). Maximum is 50MB.`)
   168→      return
   169→    }
   170→    if (file.size === 0) {
   171→      setImportError('File is empty.')
   172→      return
   173→    }
   174→
   175→    // Validate extension
   176→    const ext = '.' + file.name.split('.').pop()?.toLowerCase()
   177→    if (!ACCEPTED_EXTENSIONS.includes(ext)) {
   178→      setImportError(`Unsupported file type "${ext}". Accepted: ${ACCEPTED_EXTENSIONS.join(', ')}`)
   179→      return
   180→    }
   181→
   182→    try {
   183→      h.store.importFile(file)
   184→      setImportSuccess(`Imported "${file.name}" (${(file.size / 1024).toFixed(1)}KB)`)
   185→    } catch (err) {
   186→      setImportError(`Import failed: ${err instanceof Error ? err.message : 'Unknown error'}`)
   187→    }
   188→  }
   189→
   190→  const handleDrop = (e: React.DragEvent) => {
   191→    e.preventDefault()
   192→    e.stopPropagation()
   193→    setDragOver(false)
   194→    const files = Array.from(e.dataTransfer.files)
   195→    if (files.length === 0) return
   196→    validateAndProcessFile(files[0])
   197→  }
   198→
   199→  const handleDragOver = (e: React.DragEvent) => {
   200→    e.preventDefault()
   201→    e.stopPropagation()
   202→    setDragOver(true)
   203→  }
   204→
   205→  const handleDragLeave = (e: React.DragEvent) => {
   206→    e.preventDefault()
   207→    e.stopPropagation()
   208→    setDragOver(false)
   209→  }
   210→
   211→  return (
   212→    <div className="p-4 space-y-4 overflow-y-auto h-full scrollbar-thin" style={{ contain: 'layout style paint' }}>
   213→      <div
   214→        className={cn(
   215→          'border-2 border-dashed rounded-xl p-8 text-center cursor-pointer transition-all',
   216→          dragOver ? 'border-teal-400 bg-teal-500/10 scale-[1.01]' : 'border-border hover:border-teal-400/50 hover:bg-teal-500/5'
   217→        )}
   218→        onClick={() => fileInputRef.current?.click()}
   219→        onDrop={handleDrop}
   220→        onDragOver={handleDragOver}
   221→        onDragLeave={handleDragLeave}
   222→      >
   223→        <input
   224→          ref={fileInputRef}
   225→          type="file"
   226→          accept=".csv,.xlsx,.xls,.json,.tsv"
   227→          className="hidden"
   228→          onChange={(e) => {
   229→            const file = e.target.files?.[0]
   230→            if (file) validateAndProcessFile(file)
   231→            // Reset input so same file can be re-selected
   232→            if (fileInputRef.current) fileInputRef.current.value = ''
   233→          }}
   234→        />
   235→        {dragOver ? (
   236→          <>
   237→            <Download className="w-8 h-8 mx-auto mb-3 text-teal-400" />
   238→            <p className="text-sm font-medium text-teal-400">Drop your file here</p>
   239→          </>
   240→        ) : (
   241→          <>
   242→            <Upload className="w-8 h-8 mx-auto mb-3 text-teal-400" />
   243→            <p className="text-sm font-medium text-foreground">{h.t('import.dragDrop')}</p>
   244→            <p className="text-xs text-muted-foreground mt-1">{h.t('import.supportedFormats')}</p>
   245→            <p className="text-[10px] text-muted-foreground/60 mt-1">Max 50MB</p>
   246→          </>
   247→        )}
   248→      </div>
   249→
   250→      {importError && (
   251→        <div className="flex items-start gap-2 bg-red-500/10 border border-red-500/20 rounded-lg p-2.5 text-[10px] text-red-400">
   252→          <X className="w-3 h-3 mt-0.5 flex-shrink-0" />
   253→          <div>
   254→            <p className="font-medium">Import Error</p>
   255→            <p>{importError}</p>
   256→          </div>
   257→          <button onClick={() => setImportError('')} className="ml-auto flex-shrink-0 hover:text-red-300">
   258→            <X className="w-3 h-3" />
   259→          </button>
   260→        </div>
   261→      )}
   262→      {importSuccess && (
   263→        <div className="flex items-center gap-2 bg-emerald-500/10 border border-emerald-500/20 rounded-lg p-2.5 text-[10px] text-emerald-400">
   264→          <Check className="w-3 h-3" />
   265→          <span>{importSuccess}</span>
   266→        </div>
   267→      )}
   268→
   269→      <Separator />
   270→
   271→      <div className="space-y-2">
   272→        <p className="text-xs font-semibold text-foreground">Paste CSV Data</p>
   273→        <Textarea
   274→          value={h.importText}
   275→          onChange={(e) => h.setImportText(e.target.value)}
   276→          placeholder="Name,Age,Score&#10;Alice,25,89&#10;Bob,30,92"
   277→          className="text-xs min-h-24 font-mono"
   278→        />
   279→        <Button size="sm" className="w-full bg-teal-600 hover:bg-teal-700 text-white text-xs" onClick={h.handleImportCSV} disabled={!h.importText.trim()}>
   280→          <Database className="w-3.5 h-3.5 mr-1.5" />
   281→          Import CSV
   282→        </Button>
   283→      </div>
   284→
   285→      <Separator />
   286→
   287→      <div className="space-y-2">
   288→        <p className="text-xs font-semibold text-foreground">Quick Templates</p>
   289→        <div className="grid grid-cols-2 gap-2">
   290→          {[
   291→            { name: 'Sample Survey', data: 'Respondent,Age,Gender,Satisfaction,Income\n1,25,Male,4,45000\n2,30,Female,5,52000\n3,22,Male,3,38000\n4,35,Female,4,61000\n5,28,Male,5,48000\n6,40,Female,3,72000\n7,33,Male,4,55000\n8,27,Female,5,42000' },
   292→            { name: 'Student Grades', data: 'Student,Math,Science,English,History\nAlice,85,92,88,76\nBob,78,85,90,82\nCarol,92,88,76,95\nDave,65,72,88,70\nEve,88,95,82,90' },
   293→            { name: 'Sales Data', data: 'Month,Revenue,Profit,Customers\nJan,45000,12000,320\nFeb,52000,15000,380\nMar,48000,11000,350\nApr,61000,18000,420\nMay,55000,14000,390' },
   294→          ].map((tpl) => (

   399→/* ─── Panel 4: Analysis Center (Self-Contained) ─── */
   400→export function AnalysisPanel(h: HandlerHook) {
   401→  const [lastRun, setLastRun] = useState<string | null>(null)
   402→  const [lastRunTime, setLastRunTime] = useState<number>(0)
   403→  const [inlineMode, setInlineMode] = useState<string | null>(null) // for dialog-based tools
   404→  const [dialogVars, setDialogVars] = useState<{ var1: string; var2: string }>({ var1: '', var2: '' })
   405→
   406→  const hasData = h.store.variables.length > 0
   407→  const hasSelection = h.store.selectedVariables.length > 0
   408→  const selCount = h.store.selectedVariables.length
   409→
   410→  const runTool = useCallback((name: string, action: () => void) => {
   411→    setLastRun(name)
   412→    setLastRunTime(Date.now())
   413→    action()
   414→  }, [])
   415→
   416→  const selectAllNumeric = () => {
   417→    const numeric = h.store.variables.filter(v => v.type === 'numeric').map(v => v.name)
   418→    h.store.setSelectedVariables(numeric)
   419→  }
   420→
   421→  const selectAll = () => {
   422→    h.store.setSelectedVariables(h.store.variables.map(v => v.name))
   423→  }
   424→
   425→  const runInlineAnalysis = useCallback((type: string) => {
   426→    if (!dialogVars.var1 || !dialogVars.var2) return
   427→    if (type === 'crosstabs') {
   428→      h.setCrosstabRowVar(dialogVars.var1)
   429→      h.setCrosstabColVar(dialogVars.var2)
   430→      h.handleRunCrosstabs()
   431→    } else if (type === 'ttest') {
   432→      h.setTtestGroupVar(dialogVars.var1)
   433→      h.setTtestValueVar(dialogVars.var2)
   434→      h.handleRunTTest()
   435→    } else if (type === 'anova') {
   436→      h.setAnovaGroupVar(dialogVars.var1)
   437→      h.setAnovaValueVar(dialogVars.var2)
   438→      h.handleRunANOVA()
   439→    } else if (type === 'mann-whitney') {
   440→      h.setNonparamType('mann-whitney')
   441→      h.setNonparamVar1(dialogVars.var1)
   442→      h.setNonparamVar2(dialogVars.var2)
   443→      h.handleRunNonparametric()
   444→    } else if (type === 'wilcoxon') {
   445→      h.setNonparamType('wilcoxon')

   743→function OutputTable({ item }: { item: OutputItem }) {
   744→  if (item.type !== 'table' || !item.content) return null
   745→  const { headers = [], rows = [] } = item.content as { headers?: string[]; rows?: string[][] }
   746→  return (
   747→    <div className="overflow-x-auto border rounded-lg">
   748→      <table className="w-full text-[10px] border-collapse">
   749→        <thead className="sticky top-0">
   750→          <tr className="bg-muted/60">
   751→            {headers.map((h, i) => (
   752→              <th key={i} className="border px-2 py-1.5 text-left font-semibold text-foreground whitespace-nowrap">{h}</th>
   753→            ))}
   754→          </tr>
   755→        </thead>
   756→        <tbody>
   757→          {rows.map((row, ri) => (
   758→            <tr key={ri} className="hover:bg-muted/20">
   759→              {row.map((cell, ci) => (
   760→                <td key={ci} className="border px-2 py-1 text-muted-foreground whitespace-nowrap">{cell}</td>
   761→              ))}
   762→            </tr>
   763→          ))}
   764→        </tbody>
   765→      </table>
   766→    </div>
   767→  )

   770→function OutputChart({ item }: { item: OutputItem }) {
   771→  if (item.type !== 'chart' || !item.content) return null
   772→  const c = item.content as any
   773→  const chartId = `chart-${item.id}`
   774→
   775→  const renderChart = () => {
   776→    switch (c.chartType) {
   777→      case 'scatter':
   778→        return <ChartScatter data={c.data} title={c.title} />
   779→      case 'histogram':
   780→        return <ChartHistogram values={c.values} title={c.title} />
   781→      case 'pie':
   782→        return <ChartPie data={c.data} title={c.title} />
   783→      case 'bar':
   784→        return <ChartBar data={c.data} title={c.title} bars={[]} />
   785→      case 'boxplot':
   786→        return <ChartBoxPlot groups={c.groups} title={c.title} />
   787→      case 'line':
   788→        return <ChartLine data={c.data || []} lines={c.lines || []} title={c.title} />
   789→      default:
   790→        return <p className="text-xs text-muted-foreground">Unknown chart type</p>
   791→    }
   792→  }
   793→
   794→  return (
   795→    <Card className="border-border/50">
   796→      <CardContent className="p-3">
   797→        <div className="flex items-center justify-between mb-2">
   798→          <span className="text-[10px] font-semibold truncate">{item.title}</span>
   799→          <Button
   800→            variant="ghost"
   801→            size="sm"
   802→            className="h-5 w-5 p-0 text-muted-foreground hover:text-foreground"
   803→            onClick={() => exportChartAsImage(chartId as unknown as React.RefObject<HTMLDivElement>)}
   804→          >
   805→            <Download className="w-3 h-3" />
   806→          </Button>
   807→        </div>
   808→        <div id={chartId} className="w-full">
   809→          {renderChart()}
   810→        </div>
   811→      </CardContent>
   812→    </Card>
   813→  )
   814→}

   924→/* ─── Panel 7: Scan & OCR ─── */
   925→export function ScanPanel(h: HandlerHook) {
   926→  const [dragOver, setDragOver] = useState(false)
   927→  const [scanError, setScanError] = useState('')
   928→
   929→  const MAX_SCAN_SIZE = 20 * 1024 * 1024 // 20MB
   930→  const SCAN_ACCEPTED = ['.pdf', '.png', '.jpg', '.jpeg', '.webp', '.bmp', '.tiff']
   931→
   932→  const validateScanFile = (file: File): string | null => {
   933→    if (file.size > MAX_SCAN_SIZE) return `File too large (${(file.size / 1024 / 1024).toFixed(1)}MB). Max 20MB.`
   934→    if (file.size === 0) return 'File is empty.'
   935→    const ext = '.' + file.name.split('.').pop()?.toLowerCase()
   936→    if (!SCAN_ACCEPTED.includes(ext)) return `Unsupported type. Accepted: ${SCAN_ACCEPTED.join(', ')}`
   937→    return null
   938→  }
   939→
   940→  const processScanFile = (file: File) => {
   941→    setScanError('')
   942→    const err = validateScanFile(file)
   943→    if (err) { setScanError(err); return }
   944→    h.setScanFile(file)
   945→    const reader = new FileReader()
   946→    reader.onload = (ev) => h.setScanPreview(ev.target?.result as string)
   947→    reader.onerror = () => setScanError('Failed to read file.')
   948→    reader.readAsDataURL(file)
   949→  }
   950→
   951→  const handleScanDrop = (e: React.DragEvent) => {
   952→    e.preventDefault(); e.stopPropagation(); setDragOver(false)
   953→    const files = Array.from(e.dataTransfer.files)
   954→    if (files.length > 0) processScanFile(files[0])
   955→  }
   956→
   957→  const handleScanDragOver = (e: React.DragEvent) => { e.preventDefault(); e.stopPropagation(); setDragOver(true) }
   958→  const handleScanDragLeave = (e: React.DragEvent) => { e.preventDefault(); e.stopPropagation(); setDragOver(false) }
   959→
   960→  const handleScanSubmit = async () => {
   961→    if (!h.scanFile) return
   962→    setScanError('')
   963→    h.store.setScanState('processing')
   964→
   965→    const formData = new FormData()
   966→    formData.append('file', h.scanFile)
   967→
   968→    if (h.scanFile.name.toLowerCase().endsWith('.pdf')) {
   969→      formData.append('templateHint', 'PDF document')
   970→    }
   971→
   972→    try {
   973→      const res = await fetch('/api/scan', { method: 'POST', body: formData })
   974→      const data = await res.json()
   975→
   976→      if (!res.ok) {
   977→        setScanError(data.error || `Scan failed (HTTP ${res.status})`)
   978→        h.store.setScanState('error')
   979→        return
   980→      }
   981→
   982→      if (data.fields && data.fields.length > 0) {
   983→        h.store.setScanResults(data as any)
   984→      } else if (data.rawText && data.rawText.length > 10) {
   985→        h.store.setScanResults({
   986→          ...data,
   987→          fields: [{ label: 'Raw Text', value: data.rawText.slice(0, 500), confidence: 0.5, type: 'string' }],
   988→          summary: data.summary || 'Document text extracted. Review the raw text below.',
   989→        } as any)
   990→      } else {
   991→        setScanError('Could not extract structured data from this document. The content may be unclear, handwritten, or image-based. Try uploading a clearer scan or a different file format.')
   992→        h.store.setScanState('error')
   993→      }
   994→    } catch (err) {
   995→      setScanError(`Network error: ${err instanceof Error ? err.message : 'Connection failed'}`)
   996→      h.store.setScanState('error')
   997→    }
   998→  }
   999→
  1000→  return (
  1001→    <div className="p-4 space-y-4 overflow-y-auto h-full scrollbar-thin" style={{ contain: 'layout style paint' }}>
  1002→      <div
  1003→        className={cn(
  1004→          'border-2 border-dashed rounded-xl p-8 text-center cursor-pointer transition-all',
  1005→          h.scanFile ? 'border-cyan-400 bg-cyan-50/10' : dragOver ? 'border-cyan-400 bg-cyan-500/10 scale-[1.01]' : 'border-border hover:border-cyan-400/50 hover:bg-cyan-500/5'
  1006→        )}
  1007→        onClick={() => h.fileInputRef.current?.click()}
  1008→        onDrop={handleScanDrop}
  1009→        onDragOver={handleScanDragOver}
  1010→        onDragLeave={handleScanDragLeave}
  1011→      >
  1012→        <input
  1013→          ref={h.fileInputRef}
  1014→          type="file"
  1015→          accept=".pdf,.png,.jpg,.jpeg,.webp,.bmp,.tiff"
  1016→          className="hidden"
  1017→          onChange={(e) => {
  1018→            const f = e.target.files?.[0]
  1019→            if (f) processScanFile(f)
  1020→            if (h.fileInputRef.current) h.fileInputRef.current.value = ''
  1021→          }}
  1022→        />
  1023→        {h.store.scanState === 'processing' ? (
  1024→          <div className="flex flex-col items-center gap-2">
  1025→            <div className="w-8 h-8 border-2 border-cyan-500 border-t-transparent rounded-full animate-spin" />
  1026→            <p className="text-sm text-muted-foreground">Analyzing document with AI...</p>
  1027→            <p className="text-[10px] text-muted-foreground/60">This may take 10-30 seconds for complex documents</p>
  1028→          </div>
  1029→        ) : h.scanPreview ? (
  1030→          <div className="flex flex-col items-center gap-2">
  1031→            {/* eslint-disable-next-line @next/next/no-img-element */}
  1032→            <img src={h.scanPreview} alt="Preview" className="max-h-40 rounded-lg shadow-sm" />
  1033→            <p className="text-[10px] text-muted-foreground">{h.scanFile?.name} ({((h.scanFile?.size || 0) / 1024).toFixed(1)}KB)</p>
  1034→          </div>
  1035→        ) : (
  1036→          <>
  1037→            <ScanLine className="w-8 h-8 mx-auto mb-2 text-cyan-400" />
  1038→            <p className="text-sm font-medium">{h.t('scan.dragDrop')}</p>
  1039→            <p className="text-[10px] text-muted-foreground mt-1">{h.t('scan.supportedFormats')}</p>
  1040→            <p className="text-[10px] text-muted-foreground/60 mt-1">Max 20MB - Drag and drop or click</p>
  1041→          </>
  1042→        )}
  1043→      </div>
  1044→
  1045→      {scanError && (
  1046→        <div className="flex items-start gap-2 bg-red-500/10 border border-red-500/20 rounded-lg p-2.5 text-[10px] text-red-400">
  1047→          <X className="w-3 h-3 mt-0.5 flex-shrink-0" />
  1048→          <div className="flex-1">
  1049→            <p className="font-medium">Scan Error</p>
  1050→            <p>{scanError}</p>
  1051→          </div>
  1052→          <button onClick={() => setScanError('')} className="ml-auto flex-shrink-0 hover:text-red-300">
  1053→            <X className="w-3 h-3" />
  1054→          </button>
  1055→        </div>
  1056→      )}
  1057→
  1058→      <div className="flex gap-2">
  1059→        <Button size="sm" className="flex-1 bg-cyan-600 hover:bg-cyan-700 text-white text-xs" disabled={!h.scanFile || h.store.scanState === 'processing'} onClick={handleScanSubmit}>
  1060→          {h.store.scanState === 'processing' ? 'Analyzing...' : h.t('scan.title')}
  1061→        </Button>
  1062→        {h.store.scanState === 'done' && h.store.scanResults && (
  1063→          <Button size="sm" variant="outline" className="flex-1 text-xs" onClick={() => { h.store.importScanResults(h.store.scanResults!); h.setScanDialogOpen(false) }}>
  1064→            <Check className="w-3.5 h-3.5 mr-1" />{h.t('scan.approve')}
  1065→          </Button>
  1066→        )}
  1067→      </div>
  1068→
  1069→      {/* Raw text fallback when fields are limited */}
  1070→      {h.store.scanState === 'done' && h.store.scanResults && h.store.scanResults.rawText && h.store.scanResults.fields.length <= 2 && (
  1071→        <div className="space-y-1.5">
  1072→          <p className="text-[10px] font-semibold text-amber-400">Extracted Text (raw)</p>
  1073→          <div className="max-h-32 overflow-y-auto bg-muted/20 rounded-lg p-2 text-[10px] text-muted-foreground whitespace-pre-wrap">
  1074→            {h.store.scanResults.rawText.slice(0, 2000)}
  1075→          </div>
  1076→          <p className="text-[9px] text-muted-foreground/60">AI had difficulty parsing structured fields from this document. You can copy the raw text or re-upload a clearer version.</p>
  1077→        </div>
  1078→      )}
  1079→
  1080→      {/* Batch Scan */}
  1081→      <div className="flex items-center gap-2">
  1082→        <Button size="sm" variant="outline" className="text-[10px] h-7" onClick={() => h.batchInputRef.current?.click()}>
  1083→          <FolderOpen className="w-3 h-3 mr-1" />{h.t('scan.batch')}
  1084→        </Button>
  1085→        <input ref={h.batchInputRef} type="file" accept=".pdf,.png,.jpg,.jpeg,.webp" multiple className="hidden" onChange={async (e) => {
  1086→          const files = Array.from(e.target.files || [])
  1087→          for (const f of files) {
  1088→            const err = validateScanFile(f)
  1089→            if (err) continue
  1090→            h.store.addToBatchQueue({ id: Date.now().toString(36) + Math.random().toString(36).slice(2), name: f.name, status: 'pending' })
  1091→          }
  1092→          if (h.batchInputRef.current) h.batchInputRef.current.value = ''
  1093→        }} />
  1094→        {h.store.batchQueue.length > 0 && <span className="text-[10px] text-muted-foreground">{h.store.batchQueue.length} files queued</span>}
  1095→      </div>
  1096→
  1097→      {/* Results */}
  1098→      {h.store.scanState === 'done' && h.store.scanResults && h.store.scanResults.fields.length > 0 && (
  1099→        <div className="space-y-2">
  1100→          <p className="text-[10px] font-semibold">{h.t('scan.extracted')} ({h.store.scanResults.fields.length})</p>
  1101→          <div className="space-y-1.5 max-h-48 overflow-y-auto">
  1102→            {h.store.scanResults.fields.map((f: any, i: number) => (
  1103→              <div key={i} className="flex items-center gap-2 bg-muted/20 rounded-lg p-2 text-[10px]">
  1104→                <Badge variant="outline" className="text-[8px] w-12 justify-center shrink-0">{f.type || 'text'}</Badge>
  1105→                <span className="font-medium w-20 truncate">{f.label}</span>
  1106→                <input className="flex-1 text-[10px] bg-background border rounded px-1.5 py-0.5 min-w-0" value={h.editedFields[f.label] ?? f.value} onChange={e => h.setEditedFields(p => ({ ...p, [f.label]: e.target.value }))} />
  1107→                <Badge variant={f.confidence > 0.8 ? 'default' : 'secondary'} className={cn('text-[8px]', f.confidence > 0.8 ? 'bg-emerald-500 text-white' : 'bg-amber-500 text-white')}>
  1108→                  {Math.round((f.confidence || 0.5) * 100)}%
  1109→                </Badge>
  1110→              </div>
  1111→            ))}
  1112→          </div>
  1113→        </div>
  1114→      )}
  1115→
  1116→      {h.store.scanHistory.length > 0 && (
  1117→        <div>
  1118→          <p className="text-[10px] font-semibold mb-1.5">{h.t('scan.history')}</p>
  1119→          <div className="space-y-1 max-h-24 overflow-y-auto">
  1120→            {h.store.scanHistory.slice(-5).reverse().map((s: any, i: number) => (
  1121→              <div key={i} className="text-[10px] text-muted-foreground bg-muted/20 rounded p-1.5">{s.summary || `${s.fields.length} fields, ${s.tables.length} tables`}</div>
  1122→            ))}
  1123→          </div>
  1124→        </div>
  1125→      )}
  1126→    </div>
  1127→  )
