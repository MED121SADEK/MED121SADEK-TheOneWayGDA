   176→function uid(): string {
   177→  return Date.now().toString(36) + Math.random().toString(36).slice(2)
   178→}

   180→function rowCount(data: Record<string, any[]>): number {
   181→  const keys = Object.keys(data)
   182→  if (keys.length === 0) return 0
   183→  return Math.max(...keys.map(k => data[k]?.length ?? 0))
   184→}

   186→// ── localStorage size monitoring ──
   187→const STORAGE_WARNING_THRESHOLD = 3 * 1024 * 1024 // 3MB — warn before 4MB hard limit
   188→let _lastStorageWarning = 0
   189→
   190→export function getStorageUsage(): { usedBytes: number; usedMB: number; percentage: number } {
   191→  if (typeof window === 'undefined') return { usedBytes: 0, usedMB: 0, percentage: 0 }
   192→  let total = 0
   193→  for (let i = 0; i < localStorage.length; i++) {
   194→    const key = localStorage.key(i)
   195→    if (key) total += (localStorage.getItem(key) || '').length
   196→  }
   197→  // Typical browser limit is 5-10MB; use 5MB as reference
   198→  const limit = 5 * 1024 * 1024
   199→  return { usedBytes: total, usedMB: Math.round(total / 1024 / 1024 * 100) / 100, percentage: Math.round(total / limit * 100) }
   200→}
   201→
   202→function checkStorageSize() {
   203→  if (typeof window === 'undefined') return
   204→  const { usedBytes } = getStorageUsage()
   205→  const now = Date.now()
   206→  // Throttle warnings to once per 60 seconds
   207→  if (usedBytes > STORAGE_WARNING_THRESHOLD && now - _lastStorageWarning > 60_000) {
   208→    _lastStorageWarning = now
   209→    const mb = (usedBytes / 1024 / 1024).toFixed(1)
   210→    console.warn(`[STORE] localStorage usage at ${mb}MB — consider exporting old projects and clearing history`)
   211→  }
   212→}

   468→        function parseCsvLine(line: string): string[] {

   475→              if (ch === '"' && i + 1 < line.length && line[i + 1] === '"') {
   476→                current += '"'; i++ // escaped quote
   477→              } else if (ch === '"') {
   478→                inQuotes = false
   479→              } else {
   480→                current += ch
   481→              }
   482→            } else {
   483→              if (ch === '"') {
   484→                inQuotes = true
   485→              } else if (ch === ',') {
   486→                result.push(current.trim())
   487→                current = ''
   488→              } else {
   489→                current += ch
   490→              }
   491→            }
   492→          }
   493→          result.push(current.trim())
   494→          return result

   627→                  const num = parseFloat(raw)
   628→                  data[h].push(isNaN(num) || raw === '' ? raw : num)
   629→                })
   630→              }
   631→              variables.forEach(v => {
   632→                const col = data[v.name]
   633→                const nonEmpty = col.filter(x => x !== '' && x !== null && x !== undefined)
   634→                if (nonEmpty.length === 0) return
   635→                v.type = nonEmpty.every(x => typeof x === 'number') ? 'numeric' : 'string'
   636→                if (v.type === 'string') v.decimals = 0
   637→              })
   638→              set({ data, variables, selectedVariables: [], outputs: [], view: 'workspace', workspaceTab: 'data' })
   639→            } catch (err) {
   640→              console.error('JSON import error:', err)
   641→            }
   642→          }
   643→          reader.readAsText(file)
   644→        } else {
