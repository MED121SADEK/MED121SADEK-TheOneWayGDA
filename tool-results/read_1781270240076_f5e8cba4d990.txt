    16→/* ─── Inline Statistical Engine ─── */
    17→function calcStats(values: number[]) {
    18→  const nums = values.filter(v => typeof v === 'number' && !isNaN(v))
    19→  if (nums.length === 0) return null
    20→  const n = nums.length
    21→  const s = nums.reduce((a, b) => a + b, 0)
    22→  const mean = s / n
    23→  const sorted = [...nums].sort((a, b) => a - b)
    24→  const median = n % 2 === 0 ? (sorted[n / 2 - 1] + sorted[n / 2]) / 2 : sorted[Math.floor(n / 2)]
    25→  const variance = nums.reduce((a, b) => a + (b - mean) ** 2, 0) / (n > 1 ? n - 1 : 1)
    26→  const stddev = Math.sqrt(variance)
    27→  const min = sorted[0]
    28→  const max = sorted[n - 1]
    29→  const freq: Record<number, number> = {}
    30→  nums.forEach(v => { freq[v] = (freq[v] || 0) + 1 })
    31→  const maxFreq = Math.max(...Object.values(freq))
    32→  const mode = nums.find(v => freq[v] === maxFreq)
    33→  const skewness = n > 2 ? (nums.reduce((a, b) => a + ((b - mean) / stddev) ** 3, 0) * n) / ((n - 1) * (n - 2)) : 0
    34→  const kurtosis = n > 3 ? (nums.reduce((a, b) => a + ((b - mean) / stddev) ** 4, 0) * n * (n + 1)) / ((n - 1) * (n - 2) * (n - 3)) - 3 * (n - 1) ** 2 / ((n - 2) * (n - 3)) : 0
    35→  const p25 = sorted[Math.floor(n * 0.25)]
    36→  const p50 = sorted[Math.floor(n * 0.5)]
    37→  const p75 = sorted[Math.floor(n * 0.75)]
    38→  return { n, sum: s, mean, median, mode, variance, stddev, min, max, range: max - min, skewness, kurtosis, p25, p50, p75 }
    39→}

    41→function calcCorrelation(x: number[], y: number[]): { r: number; n: number } | null {
    42→  const pairs: [number, number][] = []
    43→  for (let i = 0; i < Math.min(x.length, y.length); i++) {
    44→    if (typeof x[i] === 'number' && typeof y[i] === 'number' && !isNaN(x[i]) && !isNaN(y[i])) {
    45→      pairs.push([x[i], y[i]])
    46→    }
    47→  }
    48→  if (pairs.length < 3) return null
    49→  const n = pairs.length
    50→  const mx = pairs.reduce((a, p) => a + p[0], 0) / n
    51→  const my = pairs.reduce((a, p) => a + p[1], 0) / n
    52→  let num = 0, dx = 0, dy = 0
    53→  for (const [px, py] of pairs) {
    54→    num += (px - mx) * (py - my)
    55→    dx += (px - mx) ** 2
    56→    dy += (py - my) ** 2
    57→  }
    58→  const r = dx === 0 || dy === 0 ? 0 : num / Math.sqrt(dx * dy)
    59→  return { r, n }
    60→}

    62→function calcRegression(x: number[], y: number[]): { slope: number; intercept: number; r2: number; n: number } | null {
    63→  const pairs: [number, number][] = []
    64→  for (let i = 0; i < Math.min(x.length, y.length); i++) {
    65→    if (typeof x[i] === 'number' && typeof y[i] === 'number' && !isNaN(x[i]) && !isNaN(y[i])) {
    66→      pairs.push([x[i], y[i]])
    67→    }
    68→  }
    69→  if (pairs.length < 3) return null
    70→  const n = pairs.length
    71→  const mx = pairs.reduce((a, p) => a + p[0], 0) / n
    72→  const my = pairs.reduce((a, p) => a + p[1], 0) / n
    73→  let num = 0, den = 0
    74→  for (const [px, py] of pairs) {
    75→    num += (px - mx) * (py - my)
    76→    den += (px - mx) ** 2
    77→  }
    78→  if (den === 0) return null
    79→  const slope = num / den
    80→  const intercept = my - slope * mx
    81→  const ssRes = pairs.reduce((a, [px, py]) => a + (py - (slope * px + intercept)) ** 2, 0)
    82→  const ssTot = pairs.reduce((a, [, py]) => a + (py - my) ** 2, 0)
    83→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot
    84→  return { slope, intercept, r2, n }
    85→}

    87→function sum(arr: number[]): number {
    88→  return arr.reduce((a, b) => a + b, 0)
    89→}

   683→  const handleValidate = useCallback(() => {

   692→    const issues: { row?: number; column: string; severity: 'error' | 'warning' | 'info'; message: string }[] = []
   693→    let missingCells = 0
   694→    let outlierCount = 0
   695→    let typeErrors = 0
   696→    const emptyCols: string[] = []
   697→
   698→    for (const v of variables) {
   699→      const col = data[v.name] || []
   700→      let colMissing = 0
   701→      const values: number[] = []
   702→
   703→      for (let i = 0; i < col.length; i++) {
   704→        const val = col[i]
   705→        if (val === '' || val === null || val === undefined) {
   706→          colMissing++
   707→          missingCells++
   708→          issues.push({ row: i + 1, column: v.name, severity: 'warning', message: 'Missing value' })
   709→        } else if (v.type === 'numeric' && typeof val === 'string' && isNaN(parseFloat(val))) {
   710→          typeErrors++
   711→          issues.push({ row: i + 1, column: v.name, severity: 'error', message: `Non-numeric value "${val}" in numeric column` })
   712→        } else if (v.type === 'numeric') {
   713→          const num = typeof val === 'string' ? parseFloat(val) : val
   714→          if (typeof num === 'number' && !isNaN(num)) values.push(num)
   715→        }
   716→      }
   717→
   718→      if (colMissing === col.length && col.length > 0) {
   719→        emptyCols.push(v.name)
   720→        issues.push({ column: v.name, severity: 'error', message: 'Entire column is empty' })
   721→      }
   722→
   723→      // Check outliers using IQR method
   724→      if (values.length >= 4) {
   725→        const sorted = [...values].sort((a, b) => a - b)
   726→        const q1 = sorted[Math.floor(sorted.length * 0.25)]
   727→        const q3 = sorted[Math.floor(sorted.length * 0.75)]
   728→        const iqr = q3 - q1
   729→        const lower = q1 - 1.5 * iqr
   730→        const upper = q3 + 1.5 * iqr
   731→        for (let i = 0; i < col.length; i++) {
   732→          const val = col[i]
   733→          const num = typeof val === 'string' ? parseFloat(val) : val
   734→          if (typeof num === 'number' && !isNaN(num) && (num < lower || num > upper)) {
   735→            outlierCount++
   736→            if (issues.length < 50) {
   737→              issues.push({ row: i + 1, column: v.name, severity: 'info', message: `Potential outlier: ${num} (range: ${lower.toFixed(1)}\u2013${upper.toFixed(1)})` })
   738→            }
   739→          }
   740→        }
   741→      }
   742→    }
   743→
   744→    // Check duplicate rows
   745→    const rowCount = Math.max(0, ...Object.values(data).map(a => a.length))
   746→    let duplicates = 0
   747→    const seen = new Set<string>()

   749→      const key = variables.map(v => String(data[v.name]?.[i] ?? '')).join('|')

   755→      }

   757→    }
   758→

   765→
   766→    // ── ALSO produce OutputPanel results so the user sees something ──
   767→    // 1) Summary table

   770→    const infos = issues.filter(i => i.severity === 'info')

   772→    const overallStatus = healthScore >= 80 ? 'Good' : healthScore >= 50 ? 'Needs Attention' : 'Poor'
   773→
   774→    store.addOutput({
   775→      id: Date.now().toString(36),
   776→      title: 'Data Validation Report',
   777→      type: 'table',
   778→      content: {
   779→        headers: ['Metric', 'Value'],
   780→        rows: [
   781→          ['Overall Health', `${healthScore}/100 (${overallStatus})`],

   783→          ['Variables', String(variables.length)],
   784→          ['Total Issues', String(totalIssues)],
   785→          ['Errors', String(errors.length)],
   786→          ['Warnings', String(warnings.length)],
   787→          ['Info (Outliers)', String(infos.length)],
   788→          ['Missing Cells', String(missingCells)],
   789→          ['Type Mismatches', String(typeErrors)],
   790→          ['Duplicate Rows', String(duplicates)],

   801→        title: `Validation Issues (${Math.min(30, issues.length)} of ${totalIssues})`,
   802→        type: 'table',
   803→        content: {
   804→          headers: ['Severity', 'Row', 'Column', 'Message'],

   808→            i.column,
   809→            i.message,
   810→          ]),
   811→        },
   812→        timestamp: new Date().toISOString(),
   813→      })
   814→    } else {
   815→      store.addOutput({
   816→        id: (Date.now() + 1).toString(36),
   817→        title: 'Validation Result',
   818→        type: 'text',
   819→        content: 'All checks passed. Your data looks clean with no issues detected.',
   820→        timestamp: new Date().toISOString(),
   821→      })
   822→    }
   823→
   824→    store.addSyntax(`DATA VALIDATION /VARIABLES=ALL`)
   825→      } finally {
   826→        setIsValidating(false)
   827→      }
   828→    }, 0)

   831→  /* ─── Data Cleaning (non-blocking) ─── */
   832→  const handleClean = useCallback(() => {
   833→    if (isCleaning) return
   834→    setIsCleaning(true)

   837→    setTimeout(() => {
   838→      try {

   841→    const variables = [...store.variables]
   842→    let trimmedCount = 0
   843→    let filledCount = 0

   846→      const col = [...(data[v.name] || [])]
   847→

   850→          const trimmed = col[i].trim()
   851→          if (trimmed !== col[i]) {
   852→            col[i] = trimmed
   853→            trimmedCount++
   854→          }
   855→        }
   856→      }
   857→
   858→      // Fill missing values: mean for numeric, mode for string
   859→      if (v.type === 'numeric') {
   860→        const nums = col.filter((x): x is number => typeof x === 'number' && !isNaN(x))
   861→        const mean = nums.length > 0 ? nums.reduce((a, b) => a + b, 0) / nums.length : 0
   862→        for (let i = 0; i < col.length; i++) {
   863→          if (col[i] === '' || col[i] === null || col[i] === undefined) {

   865→            filledCount++
   866→          }
   867→        }
   868→      } else {
   869→        const nonEmpty = col.filter(x => x !== '' && x !== null && x !== undefined)
   870→        const freq = new Map<string, number>()
   871→        nonEmpty.forEach(x => freq.set(String(x), (freq.get(String(x)) || 0) + 1))
   872→        let mode = ''
   873→        let maxFreq = 0
   874→        freq.forEach((count, val) => { if (count > maxFreq) { maxFreq = count; mode = val } })
   875→        for (let i = 0; i < col.length; i++) {
   876→          if (col[i] === '' || col[i] === null || col[i] === undefined) {
   877→            col[i] = mode
   878→            filledCount++
   879→          }
   880→        }
   881→      }
   882→
   883→      data[v.name] = col
   884→    }
   885→
   886→    // Remove duplicate rows
   887→    const rowCount = Math.max(0, ...Object.values(data).map(a => a.length))
   888→    const keepIndices: number[] = []
   889→    const seen = new Set<string>()
   890→    for (let i = 0; i < rowCount; i++) {
   891→      const key = variables.map(v => String(data[v.name]?.[i] ?? '')).join('|')
   892→      if (!seen.has(key)) {
   893→        seen.add(key)
   894→        keepIndices.push(i)
   895→      }
   896→    }
   897→
   898→    const removedRows = rowCount - keepIndices.length
   899→    if (keepIndices.length < rowCount) {
   900→      for (const v of variables) {
   901→        data[v.name] = keepIndices.map(i => data[v.name][i])
   902→      }
   903→    }
   904→
   905→    store.setData(data)
   906→
   907→    // Detailed cleaning report
   908→    const newRowCount = Math.max(0, ...Object.values(data).map(a => a.length))
   909→    store.addOutput({
   910→      id: Date.now().toString(36),
   911→      title: 'Data Cleaning Report',
   912→      type: 'table',
   913→      content: {
   914→        headers: ['Metric', 'Before', 'After', 'Change'],
   915→        rows: [
   916→          ['Total Rows', String(rowCount), String(newRowCount), removedRows > 0 ? `${removedRows} removed` : 'No change'],
   917→          ['Whitespace Trimmed', '-', String(trimmedCount), trimmedCount > 0 ? `${trimmedCount} cells` : 'None needed'],
   918→          ['Missing Values Filled', '-', String(filledCount), filledCount > 0 ? `${filledCount} cells` : 'None found'],

   956→  const handleTransformZScore = useCallback(() => {
   957→    const data: Record<string, any[]> = {}
   958→    for (const k of Object.keys(store.data)) data[k] = [...store.data[k]]
   959→    const createdVars: string[] = []
   960→
   961→    for (const varName of store.selectedVariables) {
   962→      const vals = getNumericVals(varName)
   963→      if (vals.length < 2) continue
   964→      const m = vals.reduce((a, b) => a + b, 0) / vals.length
   965→      const s = Math.sqrt(vals.reduce((a, b) => a + (b - m) ** 2, 0) / (vals.length - 1))
   966→      if (s === 0) continue
   967→      const col = store.data[varName] || []
   968→      const newName = `z_${varName}`
   969→      data[newName] = col.map(v => {
   970→        const n = typeof v === 'string' ? parseFloat(v) : v
   971→        return typeof n === 'number' && !isNaN(n) ? parseFloat(((n - m) / s).toFixed(4)) : ''
   972→      })
   973→      store.addVariable({
   974→        id: Date.now().toString(36) + varName,
   975→        name: newName,
   976→        type: 'numeric',
   977→        label: `Z-score of ${varName}`,
   978→        width: 10, decimals: 4, missing: '', values: {},
   979→      })
   980→      createdVars.push(varName)
   981→    }
   982→
   983→    store.setData(data)
   984→    store.addOutput({
   985→      id: Date.now().toString(36),
   986→      title: 'Z-Score Transformation',
   987→      type: 'table',
   988→      content: {
   989→        headers: ['Original', 'Transformed', 'Formula'],
   990→        rows: createdVars.map(v => [v, `z_${v}`, '(x − mean) / SD']),
   991→      },
   992→      timestamp: new Date().toISOString(),
   993→    })
   994→    store.addSyntax(`COMPUTE z_*= (var - MEAN(var)) / SD(var)`)

  1034→      timestamp: new Date().toISOString(),
