    90→    90→function getSession(): { token: string; user: UserData } | null {
    91→    91→  if (typeof window === 'undefined') return null
    92→    92→  try {
    93→    93→    const token = localStorage.getItem('oneway-auth-token')
    94→    94→    const userStr = localStorage.getItem('oneway-user')
    95→    95→    if (!token || !userStr) return null
    96→    96→    return { token, user: JSON.parse(userStr) }
    97→    97→  } catch {
    98→    98→    return null
    99→    99→  }
   100→   100→}

   102→   102→function timeAgo(date: string): string {
   103→   103→  const diff = Date.now() - new Date(date).getTime()
   104→   104→  const mins = Math.floor(diff / 60000)
   105→   105→  if (mins < 1) return 'Just now'
   106→   106→  if (mins < 60) return `${mins}m ago`
   107→   107→  const hrs = Math.floor(mins / 60)
   108→   108→  if (hrs < 24) return `${hrs}h ago`
   109→   109→  const days = Math.floor(hrs / 24)
   110→   110→  return `${days}d ago`
   111→   111→}
