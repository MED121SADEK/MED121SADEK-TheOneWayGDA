    17→    17→function calcStats(values: number[]) {
    18→    18→  const nums = values.filter(v => typeof v === 'number' && !isNaN(v))
    19→    19→  if (nums.length === 0) return null
    20→    20→  const n = nums.length
    21→    21→  const s = nums.reduce((a, b) => a + b, 0)
    22→    22→  const mean = s / n
    23→    23→  const sorted = [...nums].sort((a, b) => a - b)
    24→    24→  const median = n % 2 === 0 ? (sorted[n / 2 - 1] + sorted[n / 2]) / 2 : sorted[Math.floor(n / 2)]
    25→    25→  const variance = nums.reduce((a, b) => a + (b - mean) ** 2, 0) / (n > 1 ? n - 1 : 1)
    26→    26→  const stddev = Math.sqrt(variance)
    27→    27→  const min = sorted[0]
    28→    28→  const max = sorted[n - 1]
    29→    29→  const freq: Record<number, number> = {}
    30→    30→  nums.forEach(v => { freq[v] = (freq[v] || 0) + 1 })
    31→    31→  const maxFreq = Math.max(...Object.values(freq))
    32→    32→  const mode = nums.find(v => freq[v] === maxFreq)
    33→    33→  const skewness = n > 2 ? (nums.reduce((a, b) => a + ((b - mean) / stddev) ** 3, 0) * n) / ((n - 1) * (n - 2)) : 0
    34→    34→  const kurtosis = n > 3 ? (nums.reduce((a, b) => a + ((b - mean) / stddev) ** 4, 0) * n * (n + 1)) / ((n - 1) * (n - 2) * (n - 3)) - 3 * (n - 1) ** 2 / ((n - 2) * (n - 3)) : 0
    35→    35→  const p25 = sorted[Math.floor(n * 0.25)]
    36→    36→  const p50 = sorted[Math.floor(n * 0.5)]
    37→    37→  const p75 = sorted[Math.floor(n * 0.75)]
    38→    38→  return { n, sum: s, mean, median, mode, variance, stddev, min, max, range: max - min, skewness, kurtosis, p25, p50, p75 }
    39→    39→}

    41→    41→function calcCorrelation(x: number[], y: number[]): { r: number; n: number } | null {
    42→    42→  const pairs: [number, number][] = []
    43→    43→  for (let i = 0; i < Math.min(x.length, y.length); i++) {
    44→    44→    if (typeof x[i] === 'number' && typeof y[i] === 'number' && !isNaN(x[i]) && !isNaN(y[i])) {
    45→    45→      pairs.push([x[i], y[i]])
    46→    46→    }
    47→    47→  }
    48→    48→  if (pairs.length < 3) return null
    49→    49→  const n = pairs.length
    50→    50→  const mx = pairs.reduce((a, p) => a + p[0], 0) / n
    51→    51→  const my = pairs.reduce((a, p) => a + p[1], 0) / n
    52→    52→  let num = 0, dx = 0, dy = 0
    53→    53→  for (const [px, py] of pairs) {
    54→    54→    num += (px - mx) * (py - my)
    55→    55→    dx += (px - mx) ** 2
    56→    56→    dy += (py - my) ** 2
    57→    57→  }
    58→    58→  const r = dx === 0 || dy === 0 ? 0 : num / Math.sqrt(dx * dy)
    59→    59→  return { r, n }
    60→    60→}

    62→    62→function calcRegression(x: number[], y: number[]): { slope: number; intercept: number; r2: number; n: number } | null {
    63→    63→  const pairs: [number, number][] = []
    64→    64→  for (let i = 0; i < Math.min(x.length, y.length); i++) {
    65→    65→    if (typeof x[i] === 'number' && typeof y[i] === 'number' && !isNaN(x[i]) && !isNaN(y[i])) {
    66→    66→      pairs.push([x[i], y[i]])
    67→    67→    }
    68→    68→  }
    69→    69→  if (pairs.length < 3) return null
    70→    70→  const n = pairs.length
    71→    71→  const mx = pairs.reduce((a, p) => a + p[0], 0) / n
    72→    72→  const my = pairs.reduce((a, p) => a + p[1], 0) / n
    73→    73→  let num = 0, den = 0
    74→    74→  for (const [px, py] of pairs) {
    75→    75→    num += (px - mx) * (py - my)
    76→    76→    den += (px - mx) ** 2
    77→    77→  }
    78→    78→  if (den === 0) return null
    79→    79→  const slope = num / den
    80→    80→  const intercept = my - slope * mx
    81→    81→  const ssRes = pairs.reduce((a, [px, py]) => a + (py - (slope * px + intercept)) ** 2, 0)
    82→    82→  const ssTot = pairs.reduce((a, [, py]) => a + (py - my) ** 2, 0)
    83→    83→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot
    84→    84→  return { slope, intercept, r2, n }
    85→    85→}

    87→    87→function sum(arr: number[]): number {
    88→    88→  return arr.reduce((a, b) => a + b, 0)
    89→    89→}

   691→   691→        const data = store.data
   692→   692→        const variables = store.variables
   693→   693→    const issues: { row?: number; column: string; severity: 'error' | 'warning' | 'info'; message: string }[] = []
   694→   694→    let missingCells = 0
   695→   695→    let outlierCount = 0
   696→   696→    let typeErrors = 0
   697→   697→    const emptyCols: string[] = []
   698→   698→
   699→   699→    for (const v of variables) {
   700→   700→      const col = data[v.name] || []
   701→   701→      let colMissing = 0
   702→   702→      const values: number[] = []
   703→   703→
   704→   704→      for (let i = 0; i < col.length; i++) {
   705→   705→        const val = col[i]
   706→   706→        if (val === '' || val === null || val === undefined) {
   707→   707→          colMissing++
   708→   708→          missingCells++
   709→   709→          issues.push({ row: i + 1, column: v.name, severity: 'warning', message: 'Missing value' })
   710→   710→        } else if (v.type === 'numeric' && typeof val === 'string' && isNaN(parseFloat(val))) {
   711→   711→          typeErrors++
   712→   712→          issues.push({ row: i + 1, column: v.name, severity: 'error', message: `Non-numeric value "${val}" in numeric column` })
   713→   713→        } else if (v.type === 'numeric') {
   714→   714→          const num = typeof val === 'string' ? parseFloat(val) : val
   715→   715→          if (typeof num === 'number' && !isNaN(num)) values.push(num)
   716→   716→        }
   717→   717→      }
   718→   718→
   719→   719→      if (colMissing === col.length && col.length > 0) {
   720→   720→        emptyCols.push(v.name)
   721→   721→        issues.push({ column: v.name, severity: 'error', message: 'Entire column is empty' })
   722→   722→      }
   723→   723→
   724→   724→      // Check outliers using IQR method
   725→   725→      if (values.length >= 4) {
   726→   726→        const sorted = [...values].sort((a, b) => a - b)
   727→   727→        const q1 = sorted[Math.floor(sorted.length * 0.25)]
   728→   728→        const q3 = sorted[Math.floor(sorted.length * 0.75)]
   729→   729→        const iqr = q3 - q1
   730→   730→        const lower = q1 - 1.5 * iqr
   731→   731→        const upper = q3 + 1.5 * iqr
   732→   732→        for (let i = 0; i < col.length; i++) {
   733→   733→          const val = col[i]
   734→   734→          const num = typeof val === 'string' ? parseFloat(val) : val
   735→   735→          if (typeof num === 'number' && !isNaN(num) && (num < lower || num > upper)) {
   736→   736→            outlierCount++
   737→   737→            if (issues.length < 50) {
   738→   738→              issues.push({ row: i + 1, column: v.name, severity: 'info', message: `Potential outlier: ${num} (range: ${lower.toFixed(1)}\u2013${upper.toFixed(1)})` })
   739→   739→            }
   740→   740→          }
   741→   741→        }
   742→   742→      }
   743→   743→    }
   744→   744→
   745→   745→    // Check duplicate rows
   746→   746→    const rowCount = Math.max(0, ...Object.values(data).map(a => a.length))
   747→   747→    let duplicates = 0
   748→   748→    const seen = new Set<string>()
   749→   749→    for (let i = 0; i < rowCount; i++) {
   750→   750→      const key = variables.map(v => String(data[v.name]?.[i] ?? '')).join('|')

   752→   752→        duplicates++
   753→   753→        if (issues.length < 50) {
   754→   754→          issues.push({ row: i + 1, column: '*', severity: 'warning', message: 'Duplicate row' })
   755→   755→        }
   756→   756→      }
   757→   757→      seen.add(key)
   758→   758→    }
   759→   759→
   760→   760→    const totalIssues = issues.length
   761→   761→    const result = {
   762→   762→      issues: issues.slice(0, 50),
   763→   763→      stats: { totalRows: rowCount, missingCells, outliers: outlierCount, duplicates, emptyCols },
   764→   764→    }
