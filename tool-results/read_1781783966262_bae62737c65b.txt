   110→function formatDate(dateStr: string | null): string {
   111→  if (!dateStr) return '—'
   112→  const d = new Date(dateStr)
   113→  return d.toLocaleDateString('en-US', {
   114→    year: 'numeric',
   115→    month: 'short',
   116→    day: 'numeric',
   117→    hour: '2-digit',
   118→    minute: '2-digit',
   119→  })
   120→}

   122→function exportCSV(visitors: Visitor[], filename = 'visitors-export.csv') {
   123→  const headers = ['Name', 'Email', 'Type', 'Status', 'Country', 'Language', 'IP Address', 'Notes', 'Created At', 'Last Seen']
   124→  const rows = visitors.map(v => [
   125→    v.name || '',
   126→    v.email,
   127→    v.visitorType,
   128→    v.status,
   129→    v.country || '',
   130→    v.language || '',
   131→    v.ipAddress || '',
   132→    v.notes || '',
   133→    v.createdAt,
   134→    v.lastSeen || '',
   135→  ])
   136→  const csv = [headers, ...rows].map(row =>
   137→    row.map(cell => `"${String(cell).replace(/"/g, '""')}"`).join(',')
   138→  ).join('\n')
   139→  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' })
   140→  const url = URL.createObjectURL(blob)
   141→  const a = document.createElement('a')
   142→  a.href = url
   143→  a.download = filename
   144→  a.click()
   145→  URL.revokeObjectURL(url)
   146→}

  1030→                          >
  1031→                            <Trash2 className="h-4 w-4" />
  1032→                          </Button>
  1033→                        </div>
  1034→                      </div>
  1035→
  1036→                      {/* Expanded Details */}
  1037→                      <AnimatePresence>
  1038→                        {expandedId === visitor.id && (
  1039→                          <motion.div
  1040→                            initial={{ height: 0, opacity: 0 }}
  1041→                            animate={{ height: 'auto', opacity: 1 }}

  1045→                          >
  1046→                            <div className="px-4 pb-4 pt-0">

  1074→                                  icon={<Globe className="h-3.5 w-3.5" />}

  1076→                                  value={visitor.path || '—'}
  1077→                                />
  1078→                                <DetailItem
  1079→                                  icon={<Mail className="h-3.5 w-3.5" />}
  1080→                                  label="Email"
  1081→                                  value={visitor.email}
  1082→                                  full={true}
  1083→                                />
  1084→                                {visitor.userAgent && (
  1085→                                  <DetailItem
  1086→                                    icon={<Users className="h-3.5 w-3.5" />}
  1087→                                    label="User Agent"
  1088→                                    value={visitor.userAgent}
  1089→                                    full={true}
  1090→                                  />
  1091→                                )}
  1092→                                {visitor.notes && (
  1093→                                  <DetailItem
  1094→                                    icon={<StickyNote className="h-3.5 w-3.5" />}
  1095→                                    label="Notes"
  1096→                                    value={visitor.notes}
  1097→                                    full={true}
  1098→                                  />
  1099→                                )}
  1100→                              </div>
  1101→                            </div>
  1102→                          </motion.div>
  1103→                        )}
  1104→                      </AnimatePresence>
  1105→                    </Card>
  1106→                  </motion.div>
  1107→                ))}
  1108→              </AnimatePresence>
  1109→
  1110→              {/* ── Pagination ──────────────────────────────────────────────── */}
  1111→              {pagination.totalPages > 1 && (
  1112→                <div className="flex items-center justify-between pt-4">
  1113→                  <p className="text-xs text-muted-foreground">
  1114→                    Showing {((pagination.page - 1) * pagination.limit) + 1}–{Math.min(pagination.page * pagination.limit, pagination.total)} of {pagination.total}
  1115→                  </p>
  1116→                  <div className="flex items-center gap-1">
  1117→                    <Button
  1118→                      variant="outline"
  1119→                      size="sm"

  1145→                        >
  1146→                          {pageNum}
  1147→                        </Button>
  1148→                      )
  1149→                    })}
  1150→                    {pagination.totalPages > 5 && pagination.page < pagination.totalPages - 2 && (

  1152→                    )}
  1153→                    <Button
  1154→                      variant="outline"
  1155→                      size="sm"
  1156→                      className="h-8 text-xs bg-muted/30 border-border/50"
  1157→                      disabled={pagination.page >= pagination.totalPages}
  1158→                      onClick={() => setPagination(p => ({ ...p, page: p.page + 1 }))}
  1159→                    >
  1160→                      Next
  1161→                    </Button>
  1162→                  </div>
  1163→                </div>
  1164→              )}
  1165→            </motion.div>
  1166→          )}
  1167→        </motion.div>
  1168→      </main>
  1169→
  1170→      {/* ── Note Modal ─────────────────────────────────────────────────── */}
  1171→      <Dialog
  1172→        open={noteModal.open}
  1173→        onOpenChange={(open) => {
  1174→          if (!open) {
  1175→            setNoteModal({ open: false, visitor: null })
  1176→            setNoteText('')
  1177→          }
  1178→        }}
  1179→      >
  1180→        <DialogContent className="sm:max-w-md bg-card border-border/20">
  1181→          <DialogHeader>
  1182→            <DialogTitle className="flex items-center gap-2">
  1183→              <StickyNote className="h-5 w-5 text-primary" />
  1184→              Add Note
  1185→            </DialogTitle>
  1186→            <DialogDescription>
  1187→              {noteModal.visitor && (
  1188→                <>
  1189→                  For <span className="font-medium text-foreground">{noteModal.visitor.name || noteModal.visitor.email}</span>
  1190→                </>
  1191→              )}
  1192→            </DialogDescription>
  1193→          </DialogHeader>
  1194→          <Textarea
  1195→            value={noteText}
  1196→            onChange={e => setNoteText(e.target.value)}
  1197→            placeholder="Write a note about this visitor..."
  1198→            className="min-h-[120px] bg-muted/40 border-border/50 text-sm"
  1199→            autoFocus
  1200→          />
  1201→          <DialogFooter className="gap-2">
  1202→            <Button
  1203→              variant="outline"
  1204→              size="sm"
  1205→              className="bg-muted/40 border-border/50"
  1206→              onClick={() => {
  1207→                setNoteModal({ open: false, visitor: null })
  1208→                setNoteText('')
  1209→              }}
  1210→            >
  1211→              Cancel
  1212→            </Button>
  1213→            <Button
  1214→              size="sm"
  1215→              className="bg-primary hover:bg-primary/90 text-primary-foreground"
  1216→              onClick={handleSaveNote}
  1217→              disabled={noteSaving}
  1218→            >
  1219→              {noteSaving ? (
  1220→                <Loader2 className="h-4 w-4 animate-spin mr-1.5" />
  1221→              ) : (
  1222→                <Check className="h-4 w-4 mr-1.5" />
  1223→              )}
  1224→              Save Note
  1225→            </Button>
  1226→          </DialogFooter>
  1227→        </DialogContent>
  1228→      </Dialog>
  1229→    </div>
  1230→  )
  1231→}
