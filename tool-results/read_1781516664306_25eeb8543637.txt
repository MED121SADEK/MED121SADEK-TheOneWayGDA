    59→function getSession() {
    60→  if (typeof window === 'undefined') return null
    61→  try {
    62→    const token = localStorage.getItem('oneway-auth-token')
    63→    const userStr = localStorage.getItem('oneway-user')
    64→    if (!token || !userStr) return null
    65→    return { token, user: JSON.parse(userStr) as { id: string; email: string; name: string | null; role: string } }
    66→  } catch { return null }
    67→}

    69→function timeAgo(date: string): string {
    70→  const diff = Date.now() - new Date(date).getTime()
    71→  const mins = Math.floor(diff / 60000)
    72→  if (mins < 1) return 'Just now'
    73→  if (mins < 60) return `${mins}m ago`
    74→  const hrs = Math.floor(mins / 60)
    75→  if (hrs < 24) return `${hrs}h ago`
    76→  return `${Math.floor(hrs / 24)}d ago`
    77→}

    79→function getInitials(name: string | null, email: string): string {
    80→  if (name) return name.split(' ').map(n => n.charAt(0).toUpperCase()).join('').slice(0, 2)
    81→  return email.charAt(0).toUpperCase()
    82→}

    85→  'bg-blue-500', 'bg-purple-500', 'bg-pink-500', 'bg-red-500',

    89→function getAvatarColor(str: string): string {
    90→  let hash = 0
    91→  for (let i = 0; i < str.length; i++) hash = str.charCodeAt(i) + ((hash << 5) - hash)
    92→  return AVATAR_COLORS[Math.abs(hash) % AVATAR_COLORS.length]
    93→}
