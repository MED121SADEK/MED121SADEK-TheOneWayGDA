    89→/* ─── Helpers ─── */
    90→function getSession(): { token: string; user: UserData } | null {
    91→  if (typeof window === 'undefined') return null
    92→  try {
    93→    const token = localStorage.getItem('oneway-auth-token')
    94→    const userStr = localStorage.getItem('oneway-user')
    95→    if (!token || !userStr) return null
    96→    return { token, user: JSON.parse(userStr) }
    97→  } catch {
    98→    return null
    99→  }
   100→}
   101→
   102→function timeAgo(date: string): string {
   103→  const diff = Date.now() - new Date(date).getTime()
   104→  const mins = Math.floor(diff / 60000)
   105→  if (mins < 1) return 'Just now'
   106→  if (mins < 60) return `${mins}m ago`
   107→  const hrs = Math.floor(mins / 60)
   108→  if (hrs < 24) return `${hrs}h ago`
   109→  const days = Math.floor(hrs / 24)
   110→  return `${days}d ago`
   111→}

   116→    case 'project_created': return { icon: FolderOpen, color: 'text-blue-400 bg-blue-500/10', label: 'Project Created' }
   117→    case 'analysis_run': return { icon: BarChart3, color: 'text-purple-400 bg-purple-500/10', label: 'Analysis Run' }
   118→    case 'report_generated': return { icon: FileText, color: 'text-sky-400 bg-sky-500/10', label: 'Report Generated' }
   119→    case 'workflow_executed': return { icon: Workflow, color: 'text-orange-400 bg-orange-500/10', label: 'Workflow Executed' }

   122→    default: return { icon: Activity, color: 'text-muted-foreground bg-muted/30', label: type }
