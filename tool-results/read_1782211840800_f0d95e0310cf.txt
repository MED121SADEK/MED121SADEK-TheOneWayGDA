   309→// ═══════════════════════════════════════════════════════════════
   310→// POST: Chat with a specialist assistant
   311→// ═══════════════════════════════════════════════════════════════
   312→export async function POST(request: NextRequest) {
   313→  const user = await requireAuth(request)
   314→  if (!user) return NextResponse.json({ error: 'Unauthorized. Please sign in.' }, { status: 401 })
   315→
   316→  const startTime = Date.now()
   317→  const visitorId = user.userId
   318→  const ipAddress = request.headers.get('x-forwarded-for') || request.headers.get('x-real-ip') || null
   319→  const userAgent = request.headers.get('user-agent') || null
   320→
   321→  try {
   322→    const body = await request.json()
   323→    const {
   324→      specialist: specialistId,
   325→      messages,
   326→      context,
   327→      stream = false,
   328→    } = body as {
   329→      specialist: string
   330→      messages: Array<{ role: string; content: string }>
   331→      context?: string
   332→      stream?: boolean
   333→    }
   334→
   335→    if (!specialistId || typeof specialistId !== 'string') {
   336→      return NextResponse.json({ error: 'Specialist ID is required' }, { status: 400 })
   337→    }
   338→
   339→    if (!messages || !Array.isArray(messages) || messages.length === 0) {
   340→      return NextResponse.json({ error: 'Messages array is required' }, { status: 400 })
   341→    }
   342→
   343→    // Find specialist or default to data_analyst
   344→    const specialist = SPECIALISTS[specialistId] || SPECIALISTS.data_analyst
   345→
   346→    // Fetch memory for personalization
   347→    let memoryContext = ''
   348→    if (visitorId) {
   349→      try {
   350→        const [pastPipelines, preferences] = await Promise.all([
   351→          prisma.workflowPipeline.findMany({ where: { visitorId }, orderBy: { createdAt: 'desc' }, take: 3, select: { name: true, intent: true } }),
   352→          prisma.userPreference.findUnique({ where: { visitorId } }),
   353→        ])
   354→        const parts: string[] = []
   355→        if (pastPipelines.length > 0) parts.push(`User's recent analyses: ${pastPipelines.map(p => `"${p.name}"`).join(', ')}`)
   356→        if (preferences) parts.push(`Skill level: ${preferences.skillLevel}`)
   357→        if (parts.length > 0) memoryContext = `\n\n[USER CONTEXT]\n${parts.join('\n')}`
   358→      } catch { /* non-blocking */ }
   359→    }
   360→
   361→    // Build system prompt with context injection
   362→    const systemPrompt = `${specialist.systemPrompt}\n\nYou are operating within The One-Way platform context.${context ? `\n\nCurrent context: ${context}` : ''}${memoryContext}\n\nIMPORTANT: Provide COMPREHENSIVE, DETAILED responses. Never cut short or say "in summary" prematurely. Every answer should be thorough and actionable.`
   363→
   364→    const zai = await ZAI.create()
   365→
   366→    // ─── Caching strategy ───
   367→    // Cache AI responses keyed by the last user message + specialist context.
   368→    // Deduplicates identical questions within a 5-minute window to avoid
   369→    // redundant LLM API calls for rapid retries.
   370→    const cacheContext = `assistant:${specialist.id}${context ? `:${context}` : ''}`
   371→    const cacheKey = getAnalysisCacheKey(messages, cacheContext)
   372→
   373→    // ─── Streaming mode ───
   374→    if (stream) {
   375→      // Check cache first — return cached response as a stream if available
   376→      const cachedStreamText = getCachedResponse(cacheKey)
   377→      if (cachedStreamText) {
   378→        const encoder = new TextEncoder()
   379→        // Async audit log for cache hit
   380→        ;(async () => {
   381→          try {
   382→            await prisma.aiAuditLog.create({
   383→              data: {
   384→                visitorId: user.userId,
   385→                action: 'ai_query',
   386→                details: JSON.stringify({ action: 'assistant_stream', specialist: specialist.id, mode: 'stream', cacheHit: true, userId: user.userId }),
   387→                inputData: JSON.stringify({ lastMessage: messages[messages.length - 1]?.content?.slice(0, 1000) }),
   388→                durationMs: Date.now() - startTime,
   389→                ipAddress,
   390→                userAgent,
   391→              },
   392→            })
   393→          } catch { /* silent */ }
   394→        })()
   395→
   396→        const cachedStream = new ReadableStream({
   397→          start(controller) {
   398→            controller.enqueue(encoder.encode(`data: ${JSON.stringify({ choices: [{ delta: { content: cachedStreamText } }] })}\n\n`))
   399→            controller.enqueue(encoder.encode('data: [DONE]\n\n'))
   400→            controller.close()
   401→          },
   402→        })
   403→
   404→        return new Response(cachedStream, {
   405→          headers: { 'Content-Type': 'text/event-stream', 'Cache-Control': 'no-cache', 'Connection': 'keep-alive' },
   406→        })
   407→      }
   408→
   409→      const encoder = new TextEncoder()
   410→      const streamResponse = new ReadableStream({
   411→        async start(controller) {
   412→          // Accumulate raw chunks to cache the assembled text after stream completes
   413→          const rawChunks: Uint8Array[] = []
   414→          try {
   415→            const completion = await zai.chat.completions.create({
   416→              messages: [
   417→                { role: 'system', content: systemPrompt },
   418→                ...messages.map((m: { role: string; content: string }) => ({
   419→                  role: m.role as 'user' | 'assistant',
   420→                  content: m.content,
   421→                })),
   422→              ],
   423→              max_tokens: 4096,
   424→              temperature: 0.7,
   425→              stream: true as unknown as boolean,
   426→            } as any)
   427→
   428→            if (completion && typeof completion === 'object') {
   429→              const streamable = completion as any
   430→              if (streamable.body && typeof streamable.body.getReader === 'function') {
   431→                const reader = streamable.body.getReader()
   432→                while (true) {
   433→                  const { done, value } = await reader.read()
   434→                  if (done) break
   435→                  rawChunks.push(value)
   436→                  controller.enqueue(value)
   437→                }
   438→              } else if (Array.isArray(streamable.choices)) {
   439→                const text = streamable.choices[0]?.message?.content || ''
   440→                const sseData = encoder.encode(`data: ${JSON.stringify({ choices: [{ delta: { content: text } }] })}\n\n`)
   441→                rawChunks.push(sseData)
   442→                controller.enqueue(sseData)
   443→              }
   444→            }
   445→            controller.enqueue(encoder.encode('data: [DONE]\n\n'))
   446→            controller.close()
   447→
   448→            // Cache the fully assembled response text for future deduplication
   449→            try {
   450→              const combined = new TextDecoder().decode(
   451→                rawChunks.reduce((acc, chunk) => {
   452→                  const merged = new Uint8Array(acc.length + chunk.length)
   453→                  merged.set(acc)
   454→                  merged.set(chunk, acc.length)
   455→                  return merged
   456→                }, new Uint8Array(0))
   457→              )
   458→              let fullText = ''
   459→              for (const line of combined.split('\n')) {
   460→                if (line.startsWith('data: ') && line !== 'data: [DONE]') {
   461→                  try {
   462→                    const json = JSON.parse(line.slice(6))
   463→                    fullText += json.choices?.[0]?.delta?.content || json.choices?.[0]?.message?.content || ''
   464→                  } catch { /* skip */ }
   465→                }
   466→              }
   467→              if (fullText) setCachedResponse(cacheKey, fullText)
   468→            } catch { /* caching failure should not affect the response */ }
   469→          } catch (error) {
   470→            const errorMsg = error instanceof Error ? error.message : 'Stream error'
   471→            controller.enqueue(encoder.encode(`data: ${JSON.stringify({ error: errorMsg })}\n\n`))
   472→            controller.close()
   473→          }
   474→        },
   475→      })
   476→
   477→      // Async audit log
   478→      ;(async () => {
   479→        try {
   480→          await prisma.aiAuditLog.create({
   481→            data: {
   482→              visitorId: user.userId,
   483→              action: 'ai_query',
   484→              details: JSON.stringify({ action: 'assistant_stream', specialist: specialist.id, mode: 'stream', userId: user.userId }),
   485→              inputData: JSON.stringify({ lastMessage: messages[messages.length - 1]?.content?.slice(0, 1000) }),
   486→              durationMs: Date.now() - startTime,
   487→              ipAddress,
   488→              userAgent,
   489→            },
   490→          })
   491→        } catch { /* silent */ }
   492→      })()
   493→
   494→      return new Response(streamResponse, {
   495→        headers: { 'Content-Type': 'text/event-stream', 'Cache-Control': 'no-cache', 'Connection': 'keep-alive' },
   496→      })
   497→    }
   498→
   499→    // ─── Standard mode ───
   500→    // Check cache first — return immediately if we have a cached response
   501→    const cachedStandardText = getCachedResponse(cacheKey)
   502→    if (cachedStandardText) {
   503→      const durationMs = Date.now() - startTime
   504→      // Audit log for cache hit
   505→      try {
   506→        await prisma.aiAuditLog.create({
   507→          data: {
   508→            visitorId: user.userId,
   509→            action: 'ai_query',
   510→            details: JSON.stringify({ action: 'assistant_response', specialist: specialist.id, specialistName: specialist.name, messageCount: messages.length, cacheHit: true, userId: user.userId }),
   511→            inputData: JSON.stringify({ lastMessage: messages[messages.length - 1]?.content?.slice(0, 1000) }),
   512→            outputData: JSON.stringify({ responseLength: cachedStandardText.length, tokensUsed: 0 }),
   513→            tokensUsed: 0,
   514→            durationMs,
   515→            ipAddress,
   516→            userAgent,
   517→          },
   518→        })
   519→      } catch { /* silent */ }
   520→
   521→      return NextResponse.json({
   522→        message: cachedStandardText,
   523→        specialist: {
   524→          id: specialist.id,
   525→          name: specialist.name,
   526→          title: specialist.title,
   527→          icon: specialist.icon,
   528→          color: specialist.color,
   529→        },
   530→        meta: { tokensUsed: 0, durationMs, withMemory: !!memoryContext, cacheHit: true },
   531→      })
   532→    }
   533→
   534→    const completion = await zai.chat.completions.create({
   535→      messages: [
   536→        { role: 'system', content: systemPrompt },
   537→        ...messages.map((m: { role: string; content: string }) => ({
   538→          role: m.role as 'user' | 'assistant',
   539→          content: m.content,
   540→        })),
   541→      ],
   542→      max_tokens: 4096,
   543→      temperature: 0.7,
   544→    })
   545→
   546→    const aiMessage = completion.choices?.[0]?.message?.content || 'I apologize, I could not generate a response.'
   547→
   548→    // Cache the successful response for future deduplication
   549→    if (aiMessage && !aiMessage.startsWith('I apologize')) {
   550→      setCachedResponse(cacheKey, aiMessage)
   551→    }
   552→    const tokensUsed = completion.usage?.total_tokens || 0
   553→    const durationMs = Date.now() - startTime
   554→
   555→    // Audit log
   556→    try {
   557→      await prisma.aiAuditLog.create({
   558→        data: {
   559→          visitorId: user.userId,
   560→          action: 'ai_query',
   561→          details: JSON.stringify({ action: 'assistant_response', specialist: specialist.id, specialistName: specialist.name, messageCount: messages.length, userId: user.userId }),
   562→          inputData: JSON.stringify({ lastMessage: messages[messages.length - 1]?.content?.slice(0, 1000) }),
   563→          outputData: JSON.stringify({ responseLength: aiMessage.length, tokensUsed }),
   564→          tokensUsed,
   565→          durationMs,
   566→          ipAddress,
   567→          userAgent,
   568→        },
   569→      })
   570→    } catch { /* silent */ }
   571→
   572→    return NextResponse.json({
   573→      message: aiMessage,
   574→      specialist: {
   575→        id: specialist.id,
   576→        name: specialist.name,
   577→        title: specialist.title,
   578→        icon: specialist.icon,
   579→        color: specialist.color,
   580→      },
   581→      meta: { tokensUsed, durationMs, withMemory: !!memoryContext },
   582→    })
   583→  } catch (error: unknown) {
   584→    const durationMs = Date.now() - startTime
   585→    const errorMsg = error instanceof Error ? error.message : 'Unknown error'
   586→
   587→    return NextResponse.json({ error: `Assistant error: ${errorMsg}` }, { status: 500 })
   588→  }
   589→}
