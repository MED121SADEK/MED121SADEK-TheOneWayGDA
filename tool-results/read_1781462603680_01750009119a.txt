    23→export function generateActionToken(userId: string, action: 'approve' | 'reject' | 'reset' | 'sub_approve' | 'sub_reject'): string {
    24→  const payload = `${userId}:${action}:${Date.now()}`
    25→  const signature = createHmac('sha256', EMAIL_SECRET).update(payload).digest('hex')
    26→  return Buffer.from(`${payload}:${signature}`).toString('base64url')
    27→}

    29→export function verifyActionToken(token: string): { userId: string; action: 'approve' | 'reject' | 'reset'; valid: boolean } {
    30→  try {
    31→    const decoded = Buffer.from(token, 'base64url').toString('utf-8')
    32→    const [payload, signature] = decoded.split(':')
    33→
    34→    // Reconstruct and verify signature
    35→    const parts = payload.split(':')
    36→    if (parts.length < 3) return { userId: '', action: 'approve', valid: false }
    37→
    38→    const userId = parts[0]
    39→    const action = parts[1] as 'approve' | 'reject' | 'reset'
    40→    const timestamp = parseInt(parts[2])
    41→
    42→    // Check token expiry (24 hours)
    43→    if (Date.now() - timestamp > 24 * 60 * 60 * 1000) {
    44→      return { userId, action, valid: false }
    45→    }
    46→
    47→    // Verify HMAC signature
    48→    const expectedPayload = `${userId}:${action}:${timestamp}`
    49→    const expectedSig = createHmac('sha256', EMAIL_SECRET).update(expectedPayload).digest('hex')
    50→
    51→    // Find the signature part (everything after the last : of payload)
    52→    const sigIndex = decoded.lastIndexOf(':') + 1
    53→    const actualSig = decoded.slice(sigIndex)
    54→
    55→    if (actualSig === expectedSig) {
    56→      return { userId, action, valid: true }
    57→    }
    58→    return { userId, action, valid: false }
    59→  } catch {
    60→    return { userId: '', action: 'approve', valid: false }
    61→  }
    62→}
