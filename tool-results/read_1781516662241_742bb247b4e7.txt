    41→/* ─── Helpers ─── */
    42→function getSession() {
    43→  if (typeof window === 'undefined') return null
    44→  try {
    45→    const token = localStorage.getItem('oneway-auth-token')
    46→    const userStr = localStorage.getItem('oneway-user')
    47→    if (!token || !userStr) return null
    48→    return { token, user: JSON.parse(userStr) as { id: string; email: string; name: string | null; role: string } }
    49→  } catch { return null }
    50→}
    51→
    52→function timeAgo(date: string): string {
    53→  const diff = Date.now() - new Date(date).getTime()
    54→  const mins = Math.floor(diff / 60000)
    55→  if (mins < 1) return 'Just now'
    56→  if (mins < 60) return `${mins}m ago`
    57→  const hrs = Math.floor(mins / 60)
    58→  if (hrs < 24) return `${hrs}h ago`
    59→  return `${Math.floor(hrs / 24)}d ago`
    60→}
    61→
    62→function getInitials(name: string | null, email: string): string {
    63→  if (name) return name.split(' ').map(n => n.charAt(0).toUpperCase()).join('').slice(0, 2)
    64→  return email.charAt(0).toUpperCase()
    65→}
