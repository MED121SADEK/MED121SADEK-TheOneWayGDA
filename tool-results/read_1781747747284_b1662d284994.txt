     1→import { NextResponse } from 'next/server'
     2→import { db as prisma } from '@/lib/db'
     3→import { getPolicies } from '@/app/api/ai/policies/route'
     4→
     5→/* ══════════════════════════════════════════════════════
     6→   Types
     7→   ══════════════════════════════════════════════════════ */
     8→
     9→interface ComplianceCategory {
    10→  id: string
    11→  name: string
    12→  score: number
    13→  maxScore: number
    14→  percentage: number
    15→  status: 'compliant' | 'partial' | 'non_compliant' | 'unknown'
    16→  checks: ComplianceCheck[]
    17→}
    18→
    19→interface ComplianceCheck {
    20→  id: string
    21→  name: string
    22→  status: 'pass' | 'fail' | 'warning' | 'unknown'
    23→  score: number
    24→  maxScore: number
    25→  details: string
    26→}
    27→
    28→interface ComplianceRecommendation {
    29→  id: string
    30→  priority: 'high' | 'medium' | 'low'
    31→  category: string
    32→  title: string
    33→  description: string
    34→  effort: 'quick' | 'moderate' | 'significant'
    35→}
    36→
    37→/* ══════════════════════════════════════════════════════
    38→   GET /api/ai/governance/compliance
    39→   Returns compliance status across all governance areas
    40→   ══════════════════════════════════════════════════════ */
    41→
    42→export async function GET() {
    43→  try {
    44→    const categories: ComplianceCategory[] = []
    45→    const recommendations: ComplianceRecommendation[] = []
    46→
    47→    // ─── 1. Data Retention Compliance ───────────────────
    48→    const retentionChecks: ComplianceCheck[] = []
    49→    let retentionScore = 0
    50→    let retentionMax = 0
    51→
    52→    // Check if retention policy exists
    53→    const policies = getPolicies()
    54→    const compliancePolicy = policies.find(
    55→      (p) => p.category === 'compliance' && p.isActive
    56→    )
    57→    const retentionRule = compliancePolicy?.rules.find(
    58→      (r) => r.key === 'data_retention_days'
    59→    )
    60→    const retentionDays =
    61→      typeof retentionRule?.value === 'number' ? retentionRule.value : 90
    62→
    63→    retentionMax += 25
    64→    if (retentionRule) {
    65→      retentionScore += 25
    66→      retentionChecks.push({
    67→        id: 'retention-policy-defined',
    68→        name: 'Data retention policy defined',
    69→        status: 'pass' as const,
    70→        score: 25,
    71→        maxScore: 25,
    72→        details: `Retention period set to ${retentionDays} days`,
    73→      })
    74→    } else {
    75→      retentionChecks.push({
    76→        id: 'retention-policy-defined',
    77→        name: 'Data retention policy defined',
    78→        status: 'warning' as const,
    79→        score: 0,
    80→        maxScore: 25,
    81→        details: 'No explicit data retention policy found in compliance rules',
    82→      })
    83→      recommendations.push({
    84→        id: 'rec-retention-policy',
    85→        priority: 'high' as const,
    86→        category: 'data_retention',
    87→        title: 'Define Data Retention Policy',
    88→        description: `Add a data_retention_days rule to your compliance policy. Recommended: 90 days.`,
    89→        effort: 'quick' as const,
    90→      })
    91→    }
    92→
    93→    // Check if old audit logs are being cleaned up
    94→    const cutoffDate = new Date(
    95→      Date.now() - retentionDays * 24 * 60 * 60 * 1000
    96→    )
    97→    const expiredLogsCount = await prisma.aiAuditLog.count({
    98→      where: { createdAt: { lt: cutoffDate } },
    99→    })
   100→    const totalLogsCount = await prisma.aiAuditLog.count()
   101→
   102→    retentionMax += 25
   103→    if (totalLogsCount === 0 || expiredLogsCount === 0) {
   104→      retentionScore += 25
   105→      retentionChecks.push({
   106→        id: 'retention-enforcement',
   107→        name: 'Expired data cleanup',
   108→        status: totalLogsCount === 0 ? ('unknown' as const) : ('pass' as const),
   109→        score: 25,
   110→        maxScore: 25,
   111→        details:
   112→          totalLogsCount === 0
   113→            ? 'No audit logs to evaluate retention cleanup'
   114→            : `No expired audit logs found within ${retentionDays}-day retention window`,
   115→      })
   116→    } else {
   117→      retentionScore += 10
   118→      retentionChecks.push({
   119→        id: 'retention-enforcement',
   120→        name: 'Expired data cleanup',
   121→        status: 'warning' as const,
   122→        score: 10,
   123→        maxScore: 25,
   124→        details: `${expiredLogsCount} audit logs exceed the ${retentionDays}-day retention period`,
   125→      })
   126→      recommendations.push({
   127→        id: 'rec-retention-cleanup',
   128→        priority: 'medium' as const,
   129→        category: 'data_retention',
   130→        title: 'Implement Automated Data Cleanup',
   131→        description: `${expiredLogsCount} records exceed retention period. Implement a scheduled job to purge expired audit logs.`,
   132→        effort: 'moderate' as const,
   133→      })
   134→    }
   135→
   136→    // Check right to deletion support
   137→    retentionMax += 25
   138→    const deletionRule = compliancePolicy?.rules.find(
   139→      (r) => r.key === 'right_to_deletion'
   140→    )
   141→    if (deletionRule && deletionRule.value === true) {
   142→      retentionScore += 25
   143→      retentionChecks.push({
   144→        id: 'right-to-deletion',
   145→        name: 'Right to deletion supported',
   146→        status: 'pass' as const,
   147→        score: 25,
   148→        maxScore: 25,
   149→        details: 'Policy supports user right to deletion of AI-processed data',
   150→      })
   151→    } else {
   152→      retentionChecks.push({
   153→        id: 'right-to-deletion',
   154→        name: 'Right to deletion supported',
   155→        status: 'fail' as const,
   156→        score: 0,
   157→        maxScore: 25,
   158→        details: 'Right to deletion policy rule not enabled',
   159→      })
   160→      recommendations.push({
   161→        id: 'rec-right-to-deletion',
   162→        priority: 'high' as const,
   163→        category: 'data_retention',
   164→        title: 'Enable Right to Deletion',
   165→        description: 'Enable the right_to_deletion rule in your compliance policy to meet GDPR requirements.',
   166→        effort: 'quick' as const,
   167→      })
   168→    }
   169→
   170→    // Check audit trail completeness
   171→    retentionMax += 25
   172→    const recentLogs = await prisma.aiAuditLog.findMany({
   173→      orderBy: { createdAt: 'desc' },
   174→      take: 50,
   175→    })
   176→    const logsWithError = recentLogs.filter((l) => l.error !== null)
   177→    const errorRate =
   178→      recentLogs.length > 0 ? logsWithError.length / recentLogs.length : 0
   179→    if (errorRate < 0.1) {
   180→      retentionScore += 25
   181→      retentionChecks.push({
   182→        id: 'audit-completeness',
   183→        name: 'Audit trail completeness',
   184→        status: recentLogs.length === 0 ? ('unknown' as const) : ('pass' as const),
   185→        score: 25,
   186→        maxScore: 25,
   187→        details:
   188→          recentLogs.length === 0
   189→            ? 'No recent audit logs to evaluate'
   190→            : `${Math.round(errorRate * 100)}% error rate in last 50 audit entries (healthy: <10%)`,
   191→      })
   192→    } else {
   193→      retentionScore += 10
   194→      retentionChecks.push({
   195→        id: 'audit-completeness',
   196→        name: 'Audit trail completeness',
   197→        status: 'warning' as const,
   198→        score: 10,
   199→        maxScore: 25,
   200→        details: `${Math.round(errorRate * 100)}% error rate in last 50 audit entries (threshold: 10%)`,
   201→      })
   202→      recommendations.push({
   203→        id: 'rec-audit-error-rate',
   204→        priority: 'medium' as const,
   205→        category: 'data_retention',
   206→        title: 'Reduce Audit Log Error Rate',
   207→        description: `Current error rate is ${Math.round(errorRate * 100)}%. Investigate and fix failing AI operations.`,
   208→        effort: 'moderate' as const,
   209→      })
   210→    }
   211→
   212→    categories.push({
   213→      id: 'data_retention',
   214→      name: 'Data Retention & Cleanup',
   215→      score: retentionScore,
   216→      maxScore: retentionMax,
   217→      percentage: retentionMax > 0 ? Math.round((retentionScore / retentionMax) * 100) : 0,
   218→      status:
   219→        retentionMax === 0
   220→          ? 'unknown'
   221→          : retentionScore / retentionMax >= 0.8
   222→            ? 'compliant'
   223→            : retentionScore / retentionMax >= 0.5
   224→              ? 'partial'
   225→              : 'non_compliant',
   226→      checks: retentionChecks,
   227→    })
   228→
   229→    // ─── 2. GDPR Compliance ─────────────────────────────
   230→    const gdprChecks: ComplianceCheck[] = []
   231→    let gdprScore = 0
   232→    let gdprMax = 0
   233→
   234→    gdprMax += 25
   235→    const gdprRule = compliancePolicy?.rules.find(
   236→      (r) => r.key === 'gdpr_compliant'
   237→    )
   238→    if (gdprRule && gdprRule.value === true) {
   239→      gdprScore += 25
   240→      gdprChecks.push({
   241→        id: 'gdpr-declared',
   242→        name: 'GDPR compliance declared',
   243→        status: 'pass' as const,
   244→        score: 25,
   245→        maxScore: 25,
   246→        details: 'GDPR compliance rule is active in policy configuration',
   247→      })
   248→    } else {
   249→      gdprChecks.push({
   250→        id: 'gdpr-declared',
   251→        name: 'GDPR compliance declared',
   252→        status: 'fail' as const,
   253→        score: 0,
   254→        maxScore: 25,
   255→        details: 'GDPR compliance rule is not enabled in compliance policy',
   256→      })
   257→      recommendations.push({
   258→        id: 'rec-gdpr-declare',
   259→        priority: 'high' as const,
   260→        category: 'gdpr',
   261→        title: 'Enable GDPR Compliance',
   262→        description: 'Set gdpr_compliant to true in your compliance policy to declare GDPR adherence.',
   263→        effort: 'quick' as const,
   264→      })
   265→    }
   266→
   267→    gdprMax += 25
   268→    const consentRule = policies
   269→      .find((p) => p.category === 'data_access' && p.isActive)
   270→      ?.rules.find((r) => r.key === 'require_consent')
   271→    if (consentRule && consentRule.value === true) {
   272→      gdprScore += 25
   273→      gdprChecks.push({
   274→        id: 'gdpr-consent',
   275→        name: 'Explicit consent required',
   276→        status: 'pass' as const,
   277→        score: 25,
   278→        maxScore: 25,
   279→        details: 'User consent is required before accessing sensitive data',
   280→      })
   281→    } else {
   282→      gdprChecks.push({
   283→        id: 'gdpr-consent',
   284→        name: 'Explicit consent required',
   285→        status: 'fail' as const,
   286→        score: 0,
   287→        maxScore: 25,
   288→        details: 'Consent requirement not enforced in data access policy',
   289→      })
   290→      recommendations.push({
   291→        id: 'rec-gdpr-consent',
   292→        priority: 'high' as const,
   293→        category: 'gdpr',
   294→        title: 'Require Explicit Consent',
   295→        description: 'Enable the require_consent rule in your data access policy for GDPR compliance.',
   296→        effort: 'quick' as const,
   297→      })
   298→    }
   299→
   300→    gdprMax += 25
   301→    const auditLoggingRule = policies
   302→      .find((p) => p.category === 'data_access' && p.isActive)
   303→      ?.rules.find((r) => r.key === 'audit_logging')
   304→    if (auditLoggingRule && auditLoggingRule.value === true) {
   305→      gdprScore += 25
   306→      gdprChecks.push({
   307→        id: 'gdpr-audit-logging',
   308→        name: 'Audit logging enabled',
   309→        status: 'pass' as const,
   310→        score: 25,
   311→        maxScore: 25,
   312→        details: 'All data access events are audit logged',
   313→      })
   314→    } else {
   315→      gdprChecks.push({
   316→        id: 'gdpr-audit-logging',
   317→        name: 'Audit logging enabled',
   318→        status: 'warning' as const,
   319→        score: 10,
   320→        maxScore: 25,
   321→        details: 'Audit logging may not be fully enabled for all data access events',
   322→      })
   323→    }
   324→
   325→    gdprMax += 25
   326→    // Check if personal data (visitorId) is properly handled in audit logs
   327→    const anonymizedLogsCount = await prisma.aiAuditLog.count({
   328→      where: { visitorId: null },
   329→    })
   330→    const identifiableLogsCount = totalLogsCount - anonymizedLogsCount
   331→    if (totalLogsCount === 0 || anonymizedLogsCount > 0) {
   332→      gdprScore += 25
   333→      gdprChecks.push({
   334→        id: 'gdpr-data-minimization',
   335→        name: 'Data minimization',
   336→        status: 'pass' as const,
   337→        score: 25,
   338→        maxScore: 25,
   339→        details: `${anonymizedLogsCount} of ${totalLogsCount} audit entries are anonymized (no visitor ID)`,
   340→      })
   341→    } else {
   342→      gdprScore += 10
   343→      gdprChecks.push({
   344→        id: 'gdpr-data-minimization',
   345→        name: 'Data minimization',
   346→        status: 'warning' as const,
   347→        score: 10,
   348→        maxScore: 25,
   349→        details: `All ${totalLogsCount} audit entries contain identifiable visitor IDs. Consider anonymization.`,
   350→      })
   351→      recommendations.push({
   352→        id: 'rec-gdpr-anonymize',
   353→        priority: 'low' as const,
   354→        category: 'gdpr',
   355→        title: 'Implement Data Anonymization',
   356→        description: 'Consider anonymizing older audit logs by removing visitor IDs while preserving analytics value.',
   357→        effort: 'significant' as const,
   358→      })
   359→    }
   360→
   361→    categories.push({
   362→      id: 'gdpr',
   363→      name: 'GDPR Compliance',
   364→      score: gdprScore,
   365→      maxScore: gdprMax,
   366→      percentage: gdprMax > 0 ? Math.round((gdprScore / gdprMax) * 100) : 0,
   367→      status:
   368→        gdprMax === 0
   369→          ? 'unknown'
   370→          : gdprScore / gdprMax >= 0.8
   371→            ? 'compliant'
   372→            : gdprScore / gdprMax >= 0.5
   373→              ? 'partial'
   374→              : 'non_compliant',
   375→      checks: gdprChecks,
   376→    })
   377→
   378→    // ─── 3. AI Bias Detection ───────────────────────────
   379→    const biasChecks: ComplianceCheck[] = []
   380→    let biasScore = 0
   381→    let biasMax = 0
   382→
   383→    // Check for model diversity in AI operations
   384→    biasMax += 25
   385→    const modelUsageRaw = await prisma.aiAuditLog.groupBy({
   386→      by: ['model'],
   387→      _count: { id: true },
   388→      orderBy: { _count: { id: 'desc' } },
   389→    })
   390→    const totalAuditQueries = modelUsageRaw.reduce(
   391→      (sum, m) => sum + m._count.id,
   392→      0
   393→    )
   394→    const topModelCount =
   395→      modelUsageRaw.length > 0 ? modelUsageRaw[0]._count.id : 0
   396→    const topModelPercentage =
   397→      totalAuditQueries > 0
   398→        ? topModelCount / totalAuditQueries
   399→        : 0
   400→
   401→    if (modelUsageRaw.length >= 2 && topModelPercentage < 0.9) {
   402→      biasScore += 25
   403→      biasChecks.push({
   404→        id: 'bias-model-diversity',
   405→        name: 'Model diversity',
   406→        status: totalAuditQueries === 0 ? ('unknown' as const) : ('pass' as const),
   407→        score: 25,
   408→        maxScore: 25,
   409→        details:
   410→          totalAuditQueries === 0
   411→            ? 'No AI queries to evaluate model diversity'
   412→            : `${modelUsageRaw.length} different models used. Top model at ${Math.round(topModelPercentage * 100)}% (threshold: 90%)`,
   413→      })
   414→    } else if (totalAuditQueries === 0) {
   415→      biasScore += 15
   416→      biasChecks.push({
   417→        id: 'bias-model-diversity',
   418→        name: 'Model diversity',
   419→        status: 'unknown' as const,
   420→        score: 15,
   421→        maxScore: 25,
   422→        details: 'No AI queries recorded yet — cannot evaluate model diversity',
   423→      })
   424→    } else {
   425→      biasScore += 10
   426→      biasChecks.push({
   427→        id: 'bias-model-diversity',
   428→        name: 'Model diversity',
   429→        status: 'warning' as const,
   430→        score: 10,
   431→        maxScore: 25,
   432→        details: `Only ${modelUsageRaw.length} model(s) used. Top model accounts for ${Math.round(topModelPercentage * 100)}% of queries.`,
   433→      })
   434→      recommendations.push({
   435→        id: 'rec-bias-model-diversity',
   436→        priority: 'medium' as const,
   437→        category: 'bias_detection',
   438→        title: 'Diversify AI Model Usage',
   439→        description: `One model dominates at ${Math.round(topModelPercentage * 100)}%. Consider testing alternative models for bias mitigation.`,
   440→        effort: 'moderate' as const,
   441→      })
   442→    }
   443→
   444→    // Check action type diversity
   445→    biasMax += 25
   446→    const actionTypesRaw = await prisma.aiAuditLog.groupBy({
   447→      by: ['action'],
   448→      _count: { id: true },
   449→    })
   450→    if (actionTypesRaw.length >= 3) {
   451→      biasScore += 25
   452→      biasChecks.push({
   453→        id: 'bias-action-diversity',
   454→        name: 'Action type coverage',
   455→        status: 'pass' as const,
   456→        score: 25,
   457→        maxScore: 25,
   458→        details: `${actionTypesRaw.length} different action types recorded: ${actionTypesRaw.map((a) => a.action).join(', ')}`,
   459→      })
   460→    } else {
   461→      biasScore += 10
   462→      biasChecks.push({
   463→        id: 'bias-action-diversity',
   464→        name: 'Action type coverage',
   465→        status: 'warning' as const,
   466→        score: 10,
   467→        maxScore: 25,
   468→        details: `Only ${actionTypesRaw.length} action type(s) recorded. Broader coverage improves bias detection.`,
   469→      })
   470→    }
   471→
   472→    // Check for regular monitoring
   473→    biasMax += 25
   474→    const now = new Date()
   475→    const thirtyDaysAgo = new Date(Date.now() - 30 * 24 * 60 * 60 * 1000)
   476→    const monthlyLogs = await prisma.aiAuditLog.count({
   477→      where: { createdAt: { gte: thirtyDaysAgo } },
   478→    })
   479→    if (monthlyLogs > 0) {
   480→      biasScore += 25
   481→      biasChecks.push({
   482→        id: 'bias-monitoring-active',
   483→        name: 'Active bias monitoring',
   484→        status: 'pass' as const,
   485→        score: 25,
   486→        maxScore: 25,
   487→        details: `${monthlyLogs} audit entries in the last 30 days — monitoring is active`,
   488→      })
   489→    } else {
   490→      biasChecks.push({
   491→        id: 'bias-monitoring-active',
   492→        name: 'Active bias monitoring',
   493→        status: 'warning' as const,
   494→        score: 5,
   495→        maxScore: 25,
   496→        details: 'No audit activity in the last 30 days — bias monitoring is inactive',
   497→      })
   498→      recommendations.push({
   499→        id: 'rec-bias-monitoring',
   500→        priority: 'medium' as const,
   501→        category: 'bias_detection',
   502→        title: 'Activate Bias Monitoring',
   503→        description: 'No AI activity detected in the last 30 days. Ensure monitoring pipelines are running.',
   504→        effort: 'moderate' as const,
   505→      })
   506→    }
   507→
   508→    // Check for token usage patterns (potential bias indicator)
   509→    biasMax += 25
   510→    const tokenAgg = await prisma.aiAuditLog.aggregate({
   511→      _sum: { tokensUsed: true },
   512→      where: { tokensUsed: { gt: 0 } },
   513→    })
   514→    const totalTokensUsed = tokenAgg._sum.tokensUsed || 0
   515→    if (totalTokensUsed > 1000) {
   516→      biasScore += 25
   517→      biasChecks.push({
   518→        id: 'bias-token-analysis',
   519→        name: 'Token usage tracking',
   520→        status: 'pass' as const,
   521→        score: 25,
   522→        maxScore: 25,
   523→        details: `Total tokens tracked: ${totalTokensUsed.toLocaleString()}. Sufficient data for bias analysis.`,
   524→      })
   525→    } else if (totalTokensUsed === 0) {
   526→      biasScore += 10
   527→      biasChecks.push({
   528→        id: 'bias-token-analysis',
   529→        name: 'Token usage tracking',
   530→        status: 'unknown' as const,
   531→        score: 10,
   532→        maxScore: 25,
   533→        details: 'No token usage data available for bias analysis',
   534→      })
   535→    } else {
   536→      biasScore += 15
   537→      biasChecks.push({
   538→        id: 'bias-token-analysis',
   539→        name: 'Token usage tracking',
   540→        status: 'warning' as const,
   541→        score: 15,
   542→        maxScore: 25,
   543→        details: `Only ${totalTokensUsed.toLocaleString()} tokens tracked. More data needed for robust bias analysis.`,
   544→      })
   545→    }
   546→
   547→    categories.push({
   548→      id: 'bias_detection',
   549→      name: 'AI Bias Detection',
   550→      score: biasScore,
   551→      maxScore: biasMax,
   552→      percentage: biasMax > 0 ? Math.round((biasScore / biasMax) * 100) : 0,
   553→      status:
   554→        biasMax === 0
   555→          ? 'unknown'
   556→          : biasScore / biasMax >= 0.8
   557→            ? 'compliant'
   558→            : biasScore / biasMax >= 0.5
   559→              ? 'partial'
   560→              : 'non_compliant',
   561→      checks: biasChecks,
   562→    })
   563→
   564→    // ─── 4. Audit Completeness ──────────────────────────
   565→    const auditChecks: ComplianceCheck[] = []
   566→    let auditScore = 0
   567→    let auditMax = 0
   568→
   569→    // Check automation audit coverage
   570→    auditMax += 25
   571→    const automationRuns = await prisma.aiAuditLog.count({
   572→      where: { action: 'automation_run' },
   573→    })
   574→    const automationRules = await prisma.automationRule.count({
   575→      where: { runCount: { gt: 0 } },
   576→    })
   577→    if (automationRuns >= automationRules && automationRules > 0) {
   578→      auditScore += 25
   579→      auditChecks.push({
   580→        id: 'audit-automation-coverage',
   581→        name: 'Automation run audit coverage',
   582→        status: 'pass' as const,
   583→        score: 25,
   584→        maxScore: 25,
   585→        details: `All ${automationRules} automation rules with runs have audit entries (${automationRuns} total)`,
   586→      })
   587→    } else if (automationRules === 0) {
   588→      auditScore += 15
   589→      auditChecks.push({
   590→        id: 'audit-automation-coverage',
   591→        name: 'Automation run audit coverage',
   592→        status: 'unknown' as const,
   593→        score: 15,
   594→        maxScore: 25,
   595→        details: 'No automation rules with execution history to audit',
   596→      })
   597→    } else {
   598→      auditScore += 10
   599→      auditChecks.push({
   600→        id: 'audit-automation-coverage',
   601→        name: 'Automation run audit coverage',
   602→        status: 'warning' as const,
   603→        score: 10,
   604→        maxScore: 25,
   605→        details: `${automationRules} rules with runs but only ${automationRuns} audit entries — possible gaps`,
   606→      })
   607→    }
   608→
   609→    // Check decision record audit coverage
   610→    auditMax += 25
   611→    const decisionRecords = await prisma.decisionRecord.count()
   612→    const decisionAuditLogs = await prisma.aiAuditLog.count({
   613→      where: {
   614→        action: 'ai_query',
   615→        details: { contains: 'decision_recorded' },
   616→      },
   617→    })
   618→    if (decisionRecords === 0) {
   619→      auditScore += 15
   620→      auditChecks.push({
   621→        id: 'audit-decision-coverage',
   622→        name: 'Decision record audit coverage',
   623→        status: 'unknown' as const,
   624→        score: 15,
   625→        maxScore: 25,
   626→        details: 'No decision records to audit',
   627→      })
   628→    } else if (decisionAuditLogs >= decisionRecords) {
   629→      auditScore += 25
   630→      auditChecks.push({
   631→        id: 'audit-decision-coverage',
   632→        name: 'Decision record audit coverage',
   633→        status: 'pass' as const,
   634→        score: 25,
   635→        maxScore: 25,
   636→        details: `All ${decisionRecords} decision records have corresponding audit entries`,
   637→      })
   638→    } else {
   639→      auditScore += 10
   640→      auditChecks.push({
   641→        id: 'audit-decision-coverage',
   642→        name: 'Decision record audit coverage',
   643→        status: 'warning' as const,
   644→        score: 10,
   645→        maxScore: 25,
   646→        details: `${decisionRecords} decisions but only ${decisionAuditLogs} audit entries`,
   647→      })
   648→    }
   649→
   650→    // Check audit log freshness
   651→    auditMax += 25
   652→    const latestLog = await prisma.aiAuditLog.findFirst({
   653→      orderBy: { createdAt: 'desc' },
   654→      select: { createdAt: true },
   655→    })
   656→    const hoursSinceLastLog = latestLog
   657→      ? (now.getTime() - latestLog.createdAt.getTime()) / (1000 * 60 * 60)
   658→      : Infinity
   659→    if (hoursSinceLastLog < 24) {
   660→      auditScore += 25
   661→      auditChecks.push({
   662→        id: 'audit-freshness',
   663→        name: 'Audit log freshness',
   664→        status: 'pass' as const,
   665→        score: 25,
   666→        maxScore: 25,
   667→        details: `Last audit entry: ${hoursSinceLastLog.toFixed(1)} hours ago`,
   668→      })
   669→    } else if (hoursSinceLastLog < 72) {
   670→      auditScore += 15
   671→      auditChecks.push({
   672→        id: 'audit-freshness',
   673→        name: 'Audit log freshness',
   674→        status: 'warning' as const,
   675→        score: 15,
   676→        maxScore: 25,
   677→        details: `Last audit entry: ${hoursSinceLastLog.toFixed(1)} hours ago (threshold: 24h)`,
   678→      })
   679→    } else if (latestLog) {
   680→      auditScore += 5
   681→      auditChecks.push({
   682→        id: 'audit-freshness',
   683→        name: 'Audit log freshness',
   684→        status: 'fail' as const,
   685→        score: 5,
   686→        maxScore: 25,
   687→        details: `Last audit entry: ${hoursSinceLastLog.toFixed(1)} hours ago — stale audit trail`,
   688→      })
   689→      recommendations.push({
   690→        id: 'rec-audit-freshness',
   691→        priority: 'high' as const,
   692→        category: 'audit_completeness',
   693→        title: 'Restore Audit Logging',
   694→        description: 'No audit entries for over 3 days. Verify AI services are logging correctly.',
   695→        effort: 'moderate' as const,
   696→      })
   697→    } else {
   698→      auditScore += 0
   699→      auditChecks.push({
   700→        id: 'audit-freshness',
   701→        name: 'Audit log freshness',
   702→        status: 'fail' as const,
   703→        score: 0,
   704→        maxScore: 25,
   705→        details: 'No audit logs exist',
   706→      })
   707→    }
   708→
   709→    // Check data access audit coverage
   710→    auditMax += 25
   711→    const dataAccessLogs = await prisma.aiAuditLog.count({
   712→      where: { action: 'data_access' },
   713→    })
   714→    const suggestionLogs = await prisma.aiAuditLog.count({
   715→      where: {
   716→        action: { in: ['suggestion_accepted', 'suggestion_dismissed'] },
   717→      },
   718→    })
   719→    const aiQueryLogs = await prisma.aiAuditLog.count({
   720→      where: { action: 'ai_query' },
   721→    })
   722→    if (dataAccessLogs > 0 || aiQueryLogs > 0) {
   723→      auditScore += 25
   724→      auditChecks.push({
   725→        id: 'audit-data-access',
   726→        name: 'Data access audit coverage',
   727→        status: 'pass' as const,
   728→        score: 25,
   729→        maxScore: 25,
   730→        details: `Data access: ${dataAccessLogs}, AI queries: ${aiQueryLogs}, Suggestions: ${suggestionLogs}`,
   731→      })
   732→    } else if (totalLogsCount > 0) {
   733→      auditScore += 15
   734→      auditChecks.push({
   735→        id: 'audit-data-access',
   736→        name: 'Data access audit coverage',
   737→        status: 'warning' as const,
   738→        score: 15,
   739→        maxScore: 25,
   740→        details: `${totalLogsCount} audit entries but no data_access or ai_query type entries found`,
   741→      })
   742→    } else {
   743→      auditScore += 5
   744→      auditChecks.push({
   745→        id: 'audit-data-access',
   746→        name: 'Data access audit coverage',
   747→        status: 'unknown' as const,
   748→        score: 5,
   749→        maxScore: 25,
   750→        details: 'No audit entries to evaluate data access coverage',
   751→      })
   752→    }
   753→
   754→    categories.push({
   755→      id: 'audit_completeness',
   756→      name: 'Audit Completeness',
   757→      score: auditScore,
   758→      maxScore: auditMax,
   759→      percentage: auditMax > 0 ? Math.round((auditScore / auditMax) * 100) : 0,
   760→      status:
   761→        auditMax === 0
   762→          ? 'unknown'
   763→          : auditScore / auditMax >= 0.8
   764→            ? 'compliant'
   765→            : auditScore / auditMax >= 0.5
   766→              ? 'partial'
   767→              : 'non_compliant',
   768→      checks: auditChecks,
   769→    })
   770→
   771→    // ─── Calculate Overall Score ────────────────────────
   772→    const totalWeight = categories.reduce((sum, c) => sum + c.maxScore, 0)
   773→    const totalAchieved = categories.reduce((sum, c) => sum + c.score, 0)
   774→    const overallScore =
   775→      totalWeight > 0 ? Math.round((totalAchieved / totalWeight) * 100) : 0
   776→
   777→    // Sort recommendations by priority
   778→    const priorityOrder = { high: 0, medium: 1, low: 2 } as const
   779→    recommendations.sort(
   780→      (a, b) => priorityOrder[a.priority] - priorityOrder[b.priority]
   781→    )
   782→
   783→    return NextResponse.json({
   784→      overallScore,
   785→      evaluatedAt: now.toISOString(),
   786→      summary: {
   787→        totalCategories: categories.length,
   788→        compliant: categories.filter((c) => c.status === 'compliant').length,
   789→        partial: categories.filter((c) => c.status === 'partial').length,
   790→        nonCompliant: categories.filter((c) => c.status === 'non_compliant').length,
   791→        unknown: categories.filter((c) => c.status === 'unknown').length,
   792→        totalRecommendations: recommendations.length,
   793→        highPriorityRecs: recommendations.filter((r) => r.priority === 'high')
   794→          .length,
   795→      },
   796→      categories,
   797→      recommendations,
   798→    })
   799→  } catch (error: unknown) {
   800→    return NextResponse.json(
   801→      { error: 'Internal server error' },
   802→      { status: 500 }
   803→    )
   804→  }
   805→}
