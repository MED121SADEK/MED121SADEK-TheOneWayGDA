   110→function formatDate(dateStr: string | null): string {
   111→  if (!dateStr) return '—'
   112→  const d = new Date(dateStr)
   113→  return d.toLocaleDateString('en-US', {
   114→    year: 'numeric',
   115→    month: 'short',
   116→    day: 'numeric',
   117→    hour: '2-digit',
   118→    minute: '2-digit',
   119→  })
   120→}

   122→function exportCSV(visitors: Visitor[], filename = 'visitors-export.csv') {
   123→  const headers = ['Name', 'Email', 'Type', 'Status', 'Country', 'Language', 'IP Address', 'Notes', 'Created At', 'Last Seen']
   124→  const rows = visitors.map(v => [
   125→    v.name || '',
   126→    v.email,
   127→    v.visitorType,
   128→    v.status,
   129→    v.country || '',
   130→    v.language || '',
   131→    v.ipAddress || '',
   132→    v.notes || '',
   133→    v.createdAt,
   134→    v.lastSeen || '',
   135→  ])
   136→  const csv = [headers, ...rows].map(row =>
   137→    row.map(cell => `"${String(cell).replace(/"/g, '""')}"`).join(',')
   138→  ).join('\n')
   139→  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' })
   140→  const url = URL.createObjectURL(blob)
   141→  const a = document.createElement('a')
   142→  a.href = url
   143→  a.download = filename
   144→  a.click()
   145→  URL.revokeObjectURL(url)
   146→}

  1036→                        {expandedId === visitor.id && (
  1037→                          <motion.div
  1038→                            initial={{ height: 0, opacity: 0 }}
  1039→                            animate={{ height: 'auto', opacity: 1 }}
  1040→                            exit={{ height: 0, opacity: 0 }}
  1041→                            transition={{ duration: 0.25 }}
  1042→                            className="overflow-hidden"
  1043→                          >
  1044→                            <div className="px-4 pb-4 pt-0">
  1045→                              <div className="border-t border-border/10 pt-3 grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
  1046→                                <DetailItem
  1047→                                  icon={<Clock className="h-3.5 w-3.5" />}
  1048→                                  label="Registered"
  1049→                                  value={formatDate(visitor.createdAt)}
  1050→                                />
  1051→                                <DetailItem
  1052→                                  icon={<Clock className="h-3.5 w-3.5" />}
  1053→                                  label="Last Seen"
  1054→                                  value={formatDate(visitor.lastSeen)}
  1055→                                />
  1056→                                <DetailItem
  1057→                                  icon={<MapPin className="h-3.5 w-3.5" />}
  1058→                                  label="IP Address"
  1059→                                  value={visitor.ipAddress || '—'}
  1060→                                />
  1061→                                <DetailItem
  1062→                                  icon={<MapPin className="h-3.5 w-3.5" />}
  1063→                                  label="Country"
  1064→                                  value={visitor.country || '—'}
  1065→                                />
  1066→                                <DetailItem
  1067→                                  icon={<Languages className="h-3.5 w-3.5" />}
  1068→                                  label="Language"
  1069→                                  value={visitor.language || '—'}
  1070→                                />
  1071→                                <DetailItem
  1072→                                  icon={<Globe className="h-3.5 w-3.5" />}
  1073→                                  label="Referrer / Path"
  1074→                                  value={visitor.path || '—'}
  1075→                                />
  1076→                                <DetailItem
  1077→                                  icon={<Mail className="h-3.5 w-3.5" />}
  1078→                                  label="Email"
  1079→                                  value={visitor.email}
  1080→                                  full={true}
  1081→                                />
  1082→                                {visitor.userAgent && (
  1083→                                  <DetailItem
  1084→                                    icon={<Users className="h-3.5 w-3.5" />}
  1085→                                    label="User Agent"
  1086→                                    value={visitor.userAgent}
  1087→                                    full={true}
  1088→                                  />
  1089→                                )}
  1090→                                {visitor.notes && (
  1091→                                  <DetailItem
  1092→                                    icon={<StickyNote className="h-3.5 w-3.5" />}
  1093→                                    label="Notes"
  1094→                                    value={visitor.notes}
  1095→                                    full={true}
  1096→                                  />
  1097→                                )}
  1098→                              </div>
  1099→                            </div>
  1100→                          </motion.div>
  1101→                        )}
  1102→                      </AnimatePresence>
  1103→                    </Card>
  1104→                  </motion.div>
  1105→                ))}
  1106→              </AnimatePresence>
  1107→
  1108→              {/* ── Pagination ──────────────────────────────────────────────── */}
  1109→              {pagination.totalPages > 1 && (
  1110→                <div className="flex items-center justify-between pt-4">
  1111→                  <p className="text-xs text-muted-foreground">
  1112→                    Showing {((pagination.page - 1) * pagination.limit) + 1}–{Math.min(pagination.page * pagination.limit, pagination.total)} of {pagination.total}
  1113→                  </p>
  1114→                  <div className="flex items-center gap-1">
  1115→                    <Button
  1116→                      variant="outline"
  1117→                      size="sm"
  1118→                      className="h-8 text-xs bg-muted/30 border-border/50"
  1119→                      disabled={pagination.page <= 1}
  1120→                      onClick={() => setPagination(p => ({ ...p, page: p.page - 1 }))}
  1121→                    >
  1122→                      Previous
  1123→                    </Button>
  1124→                    {/* Page numbers */}
  1125→                    {Array.from({ length: Math.min(5, pagination.totalPages) }, (_, i) => {
  1126→                      let pageNum: number
  1127→                      if (pagination.totalPages <= 5) {
  1128→                        pageNum = i + 1
  1129→                      } else if (pagination.page <= 3) {
  1130→                        pageNum = i + 1
  1131→                      } else if (pagination.page >= pagination.totalPages - 2) {
  1132→                        pageNum = pagination.totalPages - 4 + i
  1133→                      } else {
  1134→                        pageNum = pagination.page - 2 + i
  1135→                      }

  1137→                        <Button
  1138→                          key={pageNum}
  1139→                          variant={pagination.page === pageNum ? 'default' : 'outline'}

  1141→                          className={`h-8 w-8 p-0 text-xs ${pagination.page === pageNum ? 'bg-primary text-primary-foreground' : 'bg-muted/30 border-border/50'}`}
  1142→                          onClick={() => setPagination(p => ({ ...p, page: pageNum }))}
  1143→                        >
  1144→                          {pageNum}
  1145→                        </Button>
  1146→                      )
  1147→                    })}
  1148→                    {pagination.totalPages > 5 && pagination.page < pagination.totalPages - 2 && (
  1149→                      <span className="text-xs text-muted-foreground px-1">...</span>
  1150→                    )}
  1151→                    <Button
  1152→                      variant="outline"
  1153→                      size="sm"
  1154→                      className="h-8 text-xs bg-muted/30 border-border/50"
  1155→                      disabled={pagination.page >= pagination.totalPages}
  1156→                      onClick={() => setPagination(p => ({ ...p, page: p.page + 1 }))}
  1157→                    >
  1158→                      Next
  1159→                    </Button>
  1160→                  </div>
  1161→                </div>
  1162→              )}
  1163→            </motion.div>
  1164→          )}
  1165→        </motion.div>
  1166→      </main>
  1167→
  1168→      {/* ── Note Modal ─────────────────────────────────────────────────── */}
  1169→      <Dialog
  1170→        open={noteModal.open}
  1171→        onOpenChange={(open) => {
  1172→          if (!open) {
  1173→            setNoteModal({ open: false, visitor: null })
  1174→            setNoteText('')
  1175→          }
  1176→        }}
  1177→      >
  1178→        <DialogContent className="sm:max-w-md bg-card border-border/20">
  1179→          <DialogHeader>
  1180→            <DialogTitle className="flex items-center gap-2">
  1181→              <StickyNote className="h-5 w-5 text-primary" />
  1182→              Add Note
  1183→            </DialogTitle>
  1184→            <DialogDescription>
  1185→              {noteModal.visitor && (
  1186→                <>
  1187→                  For <span className="font-medium text-foreground">{noteModal.visitor.name || noteModal.visitor.email}</span>
  1188→                </>
  1189→              )}
  1190→            </DialogDescription>
  1191→          </DialogHeader>
  1192→          <Textarea
  1193→            value={noteText}
  1194→            onChange={e => setNoteText(e.target.value)}
  1195→            placeholder="Write a note about this visitor..."
  1196→            className="min-h-[120px] bg-muted/40 border-border/50 text-sm"
  1197→            autoFocus
  1198→          />
  1199→          <DialogFooter className="gap-2">
  1200→            <Button
  1201→              variant="outline"
  1202→              size="sm"
  1203→              className="bg-muted/40 border-border/50"
  1204→              onClick={() => {
  1205→                setNoteModal({ open: false, visitor: null })
  1206→                setNoteText('')
  1207→              }}
  1208→            >
  1209→              Cancel
  1210→            </Button>
  1211→            <Button
  1212→              size="sm"
  1213→              className="bg-primary hover:bg-primary/90 text-primary-foreground"
  1214→              onClick={handleSaveNote}
  1215→              disabled={noteSaving}
  1216→            >
  1217→              {noteSaving ? (
  1218→                <Loader2 className="h-4 w-4 animate-spin mr-1.5" />
  1219→              ) : (
  1220→                <Check className="h-4 w-4 mr-1.5" />
  1221→              )}
  1222→              Save Note
  1223→            </Button>
  1224→          </DialogFooter>
  1225→        </DialogContent>
  1226→      </Dialog>
  1227→    </div>
  1228→  )
  1229→}
