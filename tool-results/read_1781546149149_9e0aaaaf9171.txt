   128→function uid(): string {
   129→  return Date.now().toString(36) + Math.random().toString(36).slice(2)
   130→}

   132→function detectContext(pathname: string): PageContext {
   133→  // Exact match first
   134→  if (PAGE_CONTEXTS[pathname]) return PAGE_CONTEXTS[pathname]
   135→  // Prefix match
   136→  const matched = Object.keys(PAGE_CONTEXTS).find((key) => pathname.startsWith(key))
   137→  if (matched) return PAGE_CONTEXTS[matched]
   138→  return DEFAULT_CONTEXT
   139→}

   141→function formatTime(ts: string): string {
   142→  const d = new Date(ts)
   143→  return d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
   144→}

   146→function loadHistory(): ChatMessage[] {
   147→  if (typeof window === 'undefined') return []
   148→  try {
   149→    const raw = localStorage.getItem(STORAGE_KEY)
   150→    if (!raw) return []
   151→    const parsed = JSON.parse(raw) as ChatMessage[]
   152→    if (!Array.isArray(parsed)) return []
   153→    return parsed
   154→  } catch {
   155→    return []
   156→  }
   157→}
   158→
   159→function saveHistory(messages: ChatMessage[]) {
   160→  if (typeof window === 'undefined') return
   161→  try {
   162→    // Keep last 100 messages
   163→    const trimmed = messages.slice(-100)
   164→    localStorage.setItem(STORAGE_KEY, JSON.stringify(trimmed))
   165→  } catch {
   166→    // Storage full — silently fail
   167→  }
   168→}

   302→export default function AiCopilot() {
   303→  const { t } = useTranslation()
   304→  const store = useAppStore()
   305→
   306→  // ── State ──
   307→  const [isOpen, setIsOpen] = useState(false)
   308→  const [input, setInput] = useState('')
   309→  const [messages, setMessages] = useState<ChatMessage[]>([])
   310→  const [isTyping, setIsTyping] = useState(false)
   311→  const [streamingContent, setStreamingContent] = useState('')
   312→  const [isStreaming, setIsStreaming] = useState(false)
   313→  const [context, setContext] = useState<PageContext>(DEFAULT_CONTEXT)
   314→  const [hasNewInsight, setHasNewInsight] = useState(false)
   315→  const [showInsight, setShowInsight] = useState(true)
   316→
   317→  // ── Refs ──
   318→  const inputRef = useRef<HTMLInputElement>(null)
   319→  const scrollRef = useRef<HTMLDivElement>(null)
   320→  const messagesEndRef = useRef<HTMLDivElement>(null)
   321→
   322→  // ── Detect page context ──
   323→  useEffect(() => {
   324→    if (typeof window === 'undefined') return
   325→
   326→    const detect = () => {
   327→      const pathname = window.location.pathname
   328→      const ctx = detectContext(pathname)
   329→      setContext(ctx)
   330→    }
   331→
   332→    detect()
   333→    window.addEventListener('popstate', detect)
   334→    // Also detect on pushState/replaceState
   335→    const originalPush = history.pushState
   336→    const originalReplace = history.replaceState
   337→    history.pushState = function (...args) {
   338→      originalPush.apply(this, args)
   339→      detect()
   340→    }
   341→    history.replaceState = function (...args) {
   342→      originalReplace.apply(this, args)
   343→      detect()
   344→    }
   345→
   346→    return () => {
   347→      window.removeEventListener('popstate', detect)
   348→      history.pushState = originalPush
   349→      history.replaceState = originalReplace
   350→    }
   351→  }, [])
   352→
   353→  // ── Load chat history from localStorage ──
   354→  useEffect(() => {
   355→    const history = loadHistory()
   356→    if (history.length > 0) {
   357→      setMessages(history)
   358→    }
   359→  }, [])
   360→
   361→  // ── Persist messages ──
   362→  useEffect(() => {
   363→    if (messages.length > 0) {
   364→      saveHistory(messages)
   365→    }
   366→  }, [messages])
   367→
   368→  // ── Auto-scroll to bottom ──
   369→  useEffect(() => {
   370→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' })
   371→  }, [messages, streamingContent, isTyping])
   372→
   373→  // ── Focus input when panel opens ──
   374→  useEffect(() => {
   375→    if (isOpen) {
   376→      setTimeout(() => inputRef.current?.focus(), 300)
   377→    }
   378→  }, [isOpen])
   379→
   380→  // ── Show AI insight badge when context is workspace with data ──
   381→  useEffect(() => {
   382→    if (context.id === 'workspace' && store.data && Object.keys(store.data).length > 0) {
   383→      setHasNewInsight(true)
   384→    } else {
   385→      setHasNewInsight(false)
   386→    }
   387→    setShowInsight(true)
   388→  }, [context, store.data])
   389→
   390→  // ── Streaming simulation ──
   391→  const simulateStream = useCallback((fullContent: string) => {
   392→    setIsStreaming(true)
   393→    setStreamingContent('')
   394→    let index = 0
   395→    const speed = Math.max(8, Math.min(25, 1500 / fullContent.length)) // Adaptive speed
   396→
   397→    const interval = setInterval(() => {
   398→      if (index < fullContent.length) {
   399→        // Add characters in chunks for smoother feel
   400→        const chunk = fullContent.slice(index, index + 3)
   401→        setStreamingContent((prev) => prev + chunk)
   402→        index += 3
   403→      } else {
   404→        clearInterval(interval)
   405→        setIsStreaming(false)
   406→        setStreamingContent('')
   407→      }
   408→    }, speed)
   409→
   410→    return interval
   411→  }, [])
   412→
   413→  // ── Send message ──
   414→  const sendMessage = useCallback(
   415→    async (text: string) => {
   416→      if (!text.trim() || isTyping || isStreaming) return
   417→
   418→      const userMsg: ChatMessage = {
   419→        id: uid(),
   420→        role: 'user',
   421→        content: text.trim(),
   422→        timestamp: new Date().toISOString(),
   423→      }
   424→
   425→      setMessages((prev) => [...prev, userMsg])
   426→      setInput('')
   427→      setIsTyping(true)
   428→      setShowInsight(false)
   429→
   430→      // Build page context data for the API
   431→      const pageData: Record<string, unknown> = {}
   432→      pageData.pageLabel = context.label
   433→
   434→      if (context.id === 'workspace' && store.variables && store.variables.length > 0) {
   435→        const varCount = store.variables.length
   436→        const keys = Object.keys(store.data || {})
   437→        const rowCount = keys.length > 0
   438→          ? Math.max(...keys.map((k) => (store.data[k]?.length ?? 0)))
   439→          : 0
   440→        const types = store.variables.reduce((acc, v) => {
   441→          acc[v.type] = (acc[v.type] || 0) + 1
   442→          return acc
   443→        }, {} as Record<string, number>)
   444→
   445→        pageData.datasetInfo = {
   446→          variableCount: varCount,
   447→          rowCount,
   448→          variableTypes: types,
   449→          variableNames: store.variables.map((v) => v.name),
   450→        }
   451→        if (store.currentProject) pageData.projectName = store.currentProject.name
   452→      }
   453→
   454→      try {
   455→        // Call real AI backend
   456→        const response = await fetch('/api/ai/copilot', {
   457→          method: 'POST',
   458→          headers: { 'Content-Type': 'application/json' },
   459→          body: JSON.stringify({
   460→            messages: [...messages, { role: 'user', content: text.trim() }],
   461→            context: context.id,
   462→            pageData,
   463→          }),
   464→        })
   465→
   466→        if (response.ok) {
   467→          const data = await response.json()
   468→          const aiContent = data.message || 'No response generated.'
   469→
   470→          const aiMsg: ChatMessage = {
   471→            id: uid(),
   472→            role: 'ai',
   473→            content: aiContent,
   474→            timestamp: new Date().toISOString(),
   475→          }
   476→
   477→          setIsTyping(false)
   478→          setMessages((prev) => [...prev, aiMsg])
   479→          simulateStream(aiContent)
   480→        } else {
   481→          // API error — fall back to simulated response
   482→          throw new Error(`API returned ${response.status}`)
   483→        }
   484→      } catch {
   485→        // Fallback to local simulated response when API is unavailable
   486→        const aiContent = generateSimulatedResponse(text, context)
   487→        const aiMsg: ChatMessage = {
   488→          id: uid(),
   489→          role: 'ai',
   490→          content: aiContent,
   491→          timestamp: new Date().toISOString(),
   492→        }
   493→        setIsTyping(false)
   494→        setMessages((prev) => [...prev, aiMsg])
   495→        simulateStream(aiContent)
   496→      }
   497→    },
   498→    [isTyping, isStreaming, context, store.variables, store.data, store.currentProject, simulateStream, messages],
   499→  )
   500→
   501→  // ── Clear chat ──
   502→  const clearChat = useCallback(() => {
   503→    setMessages([])
   504→    localStorage.removeItem(STORAGE_KEY)
   505→    setShowInsight(true)
   506→  }, [])
   507→
   508→  // ── Quick action handler ──
   509→  const handleQuickAction = useCallback(
   510→    (action: QuickAction) => {
   511→      sendMessage(action.prompt)
   512→    },
   513→    [sendMessage],
   514→  )
   515→
   516→  // ── Form submit ──
   517→  const handleSubmit = (e: FormEvent) => {
   518→    e.preventDefault()
   519→    sendMessage(input)
   520→  }
   521→
   522→  // ── Keyboard shortcut ──
   523→  const handleKeyDown = (e: KeyboardEvent<HTMLInputElement>) => {
   524→    if (e.key === 'Enter' && !e.shiftKey) {
