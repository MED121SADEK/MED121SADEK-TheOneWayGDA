   135→   135→function makeRequest(url: string, options: RequestInit = {}): NextRequest {
   136→   136→  return new NextRequest(new URL(url, 'http://localhost:3000'), options)
   137→   137→}
   138→   138→function makeJsonRequest(url: string, body: unknown, options: RequestInit = {}): NextRequest {
   139→   139→  return new NextRequest(new URL(url, 'http://localhost:3000'), {
   140→   140→    ...options, method: 'POST',
   141→   141→    headers: { 'Content-Type': 'application/json', ...options.headers },
   142→   142→    body: JSON.stringify(body),
   143→   143→  })
   144→   144→}
   145→   145→function authedRequest(url: string, token = 'tok', options: RequestInit = {}) {
   146→   146→  return makeRequest(url, { ...options, headers: { Authorization: `Bearer ${token}`, ...options.headers } })
   147→   147→}
