   100→function detectColumnType(values: string[]): ColumnType {
   101→  if (values.length === 0) return "text"
   102→  const nonEmpty = values.filter((v) => v.trim() !== "")
   103→  if (nonEmpty.length === 0) return "text"
   104→
   105→  const sample = nonEmpty.slice(0, 50)
   106→
   107→  // Date check
   108→  const datePatterns = [
   109→    /^\d{4}[-/]\d{1,2}[-/]\d{1,2}$/,
   110→    /^\d{1,2}[-/]\d{1,2}[-/]\d{2,4}$/,
   111→    /^\d{4}-\d{2}-\d{2}T/,
   112→    /^\d{4}-\d{2}-\d{2}\s/,
   113→    /^(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)/i,
   114→  ]
   115→  const dateMatchCount = sample.filter((v) =>
   116→    datePatterns.some((p) => p.test(v.trim()))
   117→  ).length
   118→  if (sample.length >= 3 && dateMatchCount / sample.length > 0.7) return "date"
   119→
   120→  // Numeric check
   121→  const numericCount = sample.filter((v) => {
   122→    const trimmed = v.replace(/[$%,\s]/g, "")
   123→    return !isNaN(Number(trimmed)) && trimmed !== ""
   124→  }).length
   125→  if (sample.length >= 3 && numericCount / sample.length > 0.7) return "number"
   126→
   127→  // Category check: string with low unique values
   128→  const unique = new Set(sample.map((v) => v.trim().toLowerCase()))
   129→  if (
   130→    unique.size <= CATEGORY_THRESHOLD &&
   131→    unique.size < sample.length * 0.6 &&
   132→    sample.length >= 5
   133→  )
   134→    return "category"
   135→
   136→  return "text"
   137→}
   138→
   139→function getFileExtension(filename: string): string {
   140→  return filename.slice(filename.lastIndexOf(".")).toLowerCase()
   141→}
   142→
   143→function isExcelFile(filename: string): boolean {
   144→  const ext = getFileExtension(filename)
   145→  return ext === ".xlsx" || ext === ".xls"
   146→}

   148→// ── Component ───────────────────────────────────────────────────────────────
   149→
   150→export function SmartDataImport({
   151→  open,
   152→  onOpenChange,
   153→  onImport,
   154→}: SmartDataImportProps) {
   155→  // Phase management
   156→  const [phase, setPhase] = React.useState<Phase>("upload")
   157→  const [error, setError] = React.useState<string | null>(null)
   158→
   159→  // File state
   160→  const [fileName, setFileName] = React.useState<string>("")
   161→  const [dragOver, setDragOver] = React.useState(false)
   162→
   163→  // Raw parsed data (all rows)
   164→  const [rawHeaders, setRawHeaders] = React.useState<string[]>([])
   165→  const [rawRows, setRawRows] = React.useState<string[][]>([])
   166→
   167→  // Column configuration
   168→  const [columns, setColumns] = React.useState<ColumnConfig[]>([])
   169→
   170→  // Editing column name
   171→  const [editingColumnIndex, setEditingColumnIndex] = React.useState<
   172→    number | null
   173→  >(null)
   174→  const [editNameValue, setEditNameValue] = React.useState<string>("")
   175→
   176→  // File input ref
   177→  const fileInputRef = React.useRef<HTMLInputElement>(null)
   178→
   179→  // ── Reset ──
   180→
   181→  const resetAll = React.useCallback(() => {
   182→    setPhase("upload")
   183→    setError(null)
   184→    setFileName("")
   185→    setDragOver(false)
   186→    setRawHeaders([])
   187→    setRawRows([])
   188→    setColumns([])
   189→    setEditingColumnIndex(null)
   190→    setEditNameValue("")
   191→  }, [])
   192→
   193→  const handleClose = React.useCallback(
   194→    (nextOpen: boolean) => {
   195→      if (!nextOpen) resetAll()
   196→      onOpenChange(nextOpen)
   197→    },
   198→    [onOpenChange, resetAll]
   199→  )
   200→
   201→  // ── File Processing ──
   202→
   203→  const processFile = React.useCallback((file: File) => {
   204→    const ext = getFileExtension(file.name)
   205→
   206→    if (!ACCEPTED_EXTENSIONS.includes(ext)) {
   207→      setError(
   208→        `Unsupported file format "${ext}". Accepted: ${ACCEPTED_EXTENSIONS.join(", ")}`
   209→      )
   210→      return
   211→    }
   212→
   213→    setFileName(file.name)
   214→    setError(null)
   215→    setPhase("detecting")
   216→
   217→    if (isExcelFile(file.name)) {
   218→      // Use SheetJS for Excel files
   219→      const reader = new FileReader()
   220→      reader.onload = (e) => {
   221→        try {
   222→          const data = new Uint8Array(e.target?.result as ArrayBuffer)
   223→          const workbook = XLSX.read(data, { type: "array" })
   224→          const sheetName = workbook.SheetNames[0]
   225→          if (!sheetName) {
   226→            setError("The workbook has no sheets.")
   227→            setPhase("upload")
   228→            return
   229→          }
   230→          const sheet = workbook.Sheets[sheetName]
   231→          const jsonData = XLSX.utils.sheet_to_json<string[]>(sheet, {
   232→            header: 1,
   233→          })
   234→
   235→          if (jsonData.length === 0) {
   236→            setError("The file appears to be empty.")
   237→            setPhase("upload")
   238→            return
   239→          }
   240→
   241→          const headers = (jsonData[0] ?? []).map(String)
   242→          const rows = jsonData.slice(1).map((row) =>
   243→            row.map((cell) => (cell != null ? String(cell) : ""))
   244→          )
   245→
   246→          const cols = headers.map((name, i) => ({
   247→            name,
   248→            originalName: name,
   249→            type: detectColumnType(rows.map((r) => r[i] ?? "")),
   250→            included: true,
   251→            isMissingIndicator: false,
   252→          }))
   253→          setRawHeaders(headers)
   254→          setRawRows(rows)
   255→          setColumns(cols)
   256→          setPhase("preview")
   257→        } catch {
   258→          setError("Failed to parse the Excel file. It may be corrupted.")
   259→          setPhase("upload")
   260→        }
   261→      }
   262→      reader.onerror = () => {
   263→        setError("Failed to read the file.")
   264→        setPhase("upload")
   265→      }
   266→      reader.readAsArrayBuffer(file)
   267→    } else {
   268→      // Use PapaParse for CSV / TSV files
   269→      const reader = new FileReader()
   270→      reader.onload = (e) => {
   271→        const text = e.target?.result as string
   272→        Papa.parse(text, {
   273→          header: false,
   274→          skipEmptyLines: true,
   275→          complete: (result) => {
   276→            if (result.data.length === 0) {
   277→              setError("The file appears to be empty.")
   278→              setPhase("upload")
   279→              return
   280→            }
   281→
   282→            const dataArr = (result.data as unknown as any[][]) || []
   283→            const headers = (dataArr[0] ?? []).map(String)
   284→            const rows = dataArr
   285→              .slice(1)
   286→              .map((row: any[]) =>
   287→                row.map((cell) => (cell != null ? String(cell) : ""))
   288→              )
   289→
   290→            const cols = headers.map((name, i) => ({
   291→              name,
   292→              originalName: name,
   293→              type: detectColumnType(rows.map((r) => r[i] ?? "")),
   294→              included: true,
   295→              isMissingIndicator: false,
   296→            }))
   297→
   298→            setRawHeaders(headers)
   299→            setRawRows(rows)
   300→            setColumns(cols)
   301→            setPhase("preview")
   302→          },
   303→          error: () => {
   304→            setError("Failed to parse the CSV/TSV file.")
   305→            setPhase("upload")
   306→          },
   307→        })
   308→      }
   309→      reader.onerror = () => {
   310→        setError("Failed to read the file.")
   311→        setPhase("upload")
   312→      }
   313→      reader.readAsText(file)
   314→    }
   315→  }, [])
   316→
   317→  // ── Column Helpers ──
   318→
   319→  const handleRenameColumn = React.useCallback(
   320→    (index: number, newName: string) => {
   321→      setColumns((prev) =>
   322→        prev.map((col, i) =>
   323→          i === index ? { ...col, name: newName || col.originalName } : col
   324→        )
   325→      )
   326→    },
   327→    []
   328→  )
   329→
   330→  const handleDeleteColumn = React.useCallback((index: number) => {
   331→    setColumns((prev) => prev.filter((_, i) => i !== index))
   332→  }, [])
   333→
   334→  const handleToggleType = React.useCallback((index: number, type: ColumnType) => {
   335→    setColumns((prev) =>
   336→      prev.map((col, i) => (i === index ? { ...col, type } : col))
   337→    )
   338→  }, [])
   339→
   340→  const handleToggleIncluded = React.useCallback((index: number) => {
   341→    setColumns((prev) =>
   342→      prev.map((col, i) =>
   343→        i === index ? { ...col, included: !col.included } : col
   344→      )
   345→    )
   346→  }, [])
   347→
   348→  const handleToggleMissingIndicator = React.useCallback((index: number) => {
   349→    setColumns((prev) =>
   350→      prev.map((col, i) =>
   351→        i === index
   352→          ? { ...col, isMissingIndicator: !col.isMissingIndicator }
   353→          : col
   354→      )
   355→    )
   356→  }, [])
   357→
   358→  const handleAutoDetect = React.useCallback(() => {
   359→    setColumns((prev) =>
   360→      prev.map((col, i) => ({
   361→        ...col,
   362→        type: detectColumnType(rawRows.map((r) => r[i] ?? "")),
   363→      }))
   364→    )
   365→  }, [rawRows])
   366→
   367→  // ── File Drop / Upload Handlers ──
   368→
   369→  const handleDrop = React.useCallback(
   370→    (e: React.DragEvent) => {
   371→      e.preventDefault()
   372→      e.stopPropagation()
   373→      setDragOver(false)
   374→      const file = e.dataTransfer.files[0]
   375→      if (file) processFile(file)
   376→    },
   377→    [processFile]
   378→  )
   379→
   380→  const handleDragOver = React.useCallback((e: React.DragEvent) => {
   381→    e.preventDefault()
   382→    e.stopPropagation()
   383→    setDragOver(true)
   384→  }, [])
   385→
   386→  const handleDragLeave = React.useCallback((e: React.DragEvent) => {
   387→    e.preventDefault()
   388→    e.stopPropagation()
   389→    setDragOver(false)
   390→  }, [])
   391→
   392→  const handleFileInputChange = React.useCallback(
   393→    (e: React.ChangeEvent<HTMLInputElement>) => {
   394→      const file = e.target.files?.[0]
   395→      if (file) processFile(file)
   396→      // Reset input so same file can be re-selected
   397→      if (fileInputRef.current) fileInputRef.current.value = ""
   398→    },
   399→    [processFile]
   400→  )
   401→
   402→  // ── Import Action ──
   403→
   404→  const handleImport = React.useCallback(() => {
   405→    if (columns.length === 0) return
   406→
   407→    const includedColumns = columns.filter((c) => c.included)
   408→    const missingIndicators = columns
   409→      .filter((c) => c.isMissingIndicator)
   410→      .map((c) => c.name)
   411→
   412→    const data: Record<string, any[]> = {}
   413→    includedColumns.forEach((col) => {
   414→      data[col.name] = []
   415→    })
   416→
   417→    for (const row of rawRows) {
   418→      includedColumns.forEach((col) => {
   419→        const origIdx = rawHeaders.indexOf(col.originalName)
   420→        const raw = origIdx >= 0 ? row[origIdx] ?? "" : ""
   421→
   422→        // Check if value matches any missing indicator
   423→        if (missingIndicators.includes(raw.trim())) {
   424→          data[col.name].push(null)
   425→          return
   426→        }
   427→
   428→        if (col.type === "number") {
   429→          const cleaned = raw.replace(/[$%,\s]/g, "")
   430→          const num = Number(cleaned)
   431→          data[col.name].push(isNaN(num) ? null : num)
   432→        } else if (col.type === "date") {
   433→          const parsed = Date.parse(raw)
   434→          data[col.name].push(isNaN(parsed) ? raw : new Date(parsed).toISOString())
   435→        } else {
   436→          data[col.name].push(raw)
   437→        }
   438→      })
   439→    }
   440→
   441→    const variables = includedColumns.map((col) => ({
   442→      name: col.name,
   443→      type: col.type,
   444→    }))
   445→
   446→    onImport(data, variables)
   447→    handleClose(false)
   448→  }, [columns, rawHeaders, rawRows, onImport, handleClose])
   449→
   450→  // ── Derived ──
   451→
   452→  const includedCount = columns.filter((c) => c.included).length
   453→  const previewRows = rawRows.slice(0, MAX_PREVIEW_ROWS)
   454→
   455→  // ── Render: Upload Phase ──
   456→
   457→  const renderUpload = () => (
   458→    <div className="flex flex-col items-center justify-center gap-5 py-8">
   459→      <div
   460→        onDrop={handleDrop}
   461→        onDragOver={handleDragOver}
   462→        onDragLeave={handleDragLeave}
   463→        onClick={() => fileInputRef.current?.click()}
   464→        role="button"
   465→        tabIndex={0}
   466→        onKeyDown={(e) => {
   467→          if (e.key === "Enter" || e.key === " ") fileInputRef.current?.click()
   468→        }}
   469→        className={cn(
   470→          "relative w-full max-w-md cursor-pointer rounded-xl border-2 border-dashed p-8 text-center transition-all duration-200",
   471→          dragOver
   472→            ? "border-primary/60 bg-primary/5 scale-[1.02]"
   473→            : "border-border/60 hover:border-primary/30 hover:bg-muted/30",
   474→          error && "border-destructive/50"
   475→        )}
   476→      >
   477→        <input
   478→          ref={fileInputRef}
   479→          type="file"
   480→          accept={ACCEPTED_EXTENSIONS.join(",")}
   481→          onChange={handleFileInputChange}
   482→          className="hidden"
   483→          aria-label="Upload file"
   484→        />
   485→        <div
   486→          className={cn(
   487→            "mx-auto mb-3 flex size-12 items-center justify-center rounded-lg transition-colors",
   488→            dragOver ? "bg-primary/10" : "bg-muted/60"
   489→          )}
   490→        >
   491→          <Upload
   492→            className={cn(
   493→              "size-6 transition-colors",
   494→              dragOver ? "text-primary" : "text-muted-foreground"
   495→            )}
   496→          />
   497→        </div>
   498→        <p className="text-sm font-medium text-foreground">
   499→          {dragOver ? "Drop file here" : "Drag & drop or click to upload"}
   500→        </p>
   501→        <p className="mt-1 text-xs text-muted-foreground">
   502→          Supports{" "}
   503→          {ACCEPTED_EXTENSIONS.map((ext) => (
   504→            <Badge
   505→              key={ext}
   506→              variant="outline"
   507→              className="text-[10px] px-1 py-0 h-3.5 mx-0.5 font-mono"
   508→            >
   509→              {ext}
   510→            </Badge>
   511→          ))}
   512→        </p>
   513→      </div>
   514→
   515→      {error && (
   516→        <div className="flex items-start gap-2 max-w-md rounded-lg bg-destructive/10 border border-destructive/20 p-3">
   517→          <X className="size-4 text-destructive shrink-0 mt-0.5" />
   518→          <p className="text-xs text-destructive">{error}</p>
   519→        </div>
   520→      )}
   521→    </div>
   522→  )
   523→
   524→  // ── Render: Detecting Phase ──
   525→
   526→  const renderDetecting = () => (
   527→    <div className="flex flex-col items-center justify-center gap-4 py-12">
   528→      <div className="relative">
   529→        <div className="size-10 rounded-full border-2 border-primary/30 border-t-primary animate-spin" />
   530→        <FileSpreadsheet className="size-4 text-primary absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2" />
   531→      </div>
   532→      <div className="text-center">
   533→        <p className="text-sm font-medium text-foreground">
   534→          Analyzing{" "}
   535→          <span className="text-primary font-mono">{fileName}</span>
   536→        </p>
   537→        <p className="text-xs text-muted-foreground mt-1">
   538→          Detecting column types and structure...
   539→        </p>
   540→      </div>
   541→    </div>
   542→  )
   543→
   544→  // ── Render: Preview Phase ──
   545→
   546→  const renderPreview = () => (
   547→    <div className="space-y-4">
   548→      {/* Summary bar */}
   549→      <div className="flex items-center gap-3 px-1">
   550→        <div className="flex items-center gap-1.5">
   551→          <Table2 className="size-3.5 text-muted-foreground" />
   552→          <span className="text-xs text-muted-foreground">
   553→            <span className="text-foreground font-semibold">{rawRows.length}</span> rows
   554→          </span>
   555→          <span className="text-muted-foreground/40">×</span>
   556→          <span className="text-xs text-muted-foreground">
   557→            <span className="text-foreground font-semibold">{columns.length}</span>{" "}
   558→            columns
   559→          </span>
   560→        </div>
   561→        <span className="text-muted-foreground/30">|</span>
   562→        <span className="text-xs text-muted-foreground truncate flex-1">
   563→          <span className="text-muted-foreground/60">File:</span>{" "}
   564→          <span className="font-mono text-foreground/80">{fileName}</span>
   565→        </span>
   566→        <Button
   567→          variant="ghost"
   568→          size="sm"
   569→          className="h-7 text-xs gap-1.5 px-2 text-primary"
   570→          onClick={() => {
   571→            resetAll()
   572→          }}
   573→        >
   574→          <RotateCcw className="size-3" />
   575→          New file
   576→        </Button>
   577→      </div>
   578→
   579→      {/* Column badges row */}
   580→      <div className="flex flex-wrap gap-1.5 px-1">
   581→        {columns.map((col, i) => (
   582→          <div key={i} className="flex items-center gap-1 group">
   583→            <Badge
   584→              variant="outline"
   585→              className={cn(
   586→                "text-[10px] px-1.5 py-0 h-5 font-mono gap-1",
   587→                !col.included && "opacity-40 line-through",
   588→                TYPE_COLORS[col.type]
   589→              )}
   590→            >
   591→              <span
   592→                className={cn(
   593→                  "inline-block size-1.5 rounded-full",
   594→                  TYPE_DOT_COLORS[col.type]
   595→                )}
   596→              />
   597→              {col.name}
   598→            </Badge>
   599→          </div>
   600→        ))}
   601→      </div>
   602→
   603→      {/* Preview table */}
   604→      <ScrollArea className="max-h-72 rounded-lg border border-border/50">
   605→        <UITable>
   606→          <TableHeader>
   607→            <TableRow className="hover:bg-transparent border-border/50">
   608→              <TableHead className="text-[10px] text-muted-foreground/50 h-7 px-2.5 w-8 font-medium">
   609→                #
   610→              </TableHead>
   611→              {columns.map((col, i) => (
   612→                <TableHead

   680→

   684→          variant="ghost"
   685→          size="sm"

   737→              {columns.length}
   738→            </span>{" "}
   739→            columns
   740→          </span>
   741→        </div>
   742→        <Button
   743→          variant="ghost"
   744→          size="sm"
   745→          className="h-7 text-xs gap-1.5 px-2 text-primary"
   746→          onClick={handleAutoDetect}

   753→      {/* Column list */}

   769→                </span>

   771→                {/* Include toggle */}
   772→                <Switch

   778→

   781→                  <Input
   782→                    autoFocus
   783→                    value={editNameValue}

   791→                        handleRenameColumn(i, editNameValue)

   794→                      if (e.key === "Escape") setEditingColumnIndex(null)

   805→                      "text-xs font-mono truncate hover:text-primary transition-colors flex-1 min-w-0 text-left",
   806→                      !col.included
   807→                        ? "text-muted-foreground line-through"
   808→                        : "text-foreground"

   856→                      {TYPE_OPTIONS.map((t) => (

   858→                          <div className="flex items-center gap-1.5">
   859→                            <span
   860→                              className={cn(
   861→                                "inline-block size-1.5 rounded-full",
   862→                                TYPE_DOT_COLORS[t]
   863→                              )}
   864→                            />
   865→                            {TYPE_LABELS[t]}

   899→            size="sm"
   900→            className="h-8 text-xs gap-1.5"

   970→                          <Check className="size-3" />
   971→                        ) : (
   972→                          i + 1
   973→                        )}
   974→                      </div>

   977→                          "text-[10px] transition-colors hidden sm:inline",
   978→                          isActive

   980→                            : isCompleted
   981→                              ? "text-emerald-400/70"
   982→                              : "text-muted-foreground/40"

   984→                      >
   985→                        {labels[i]}

   991→                          "flex-1 h-px mx-1",

   993→                            ? "bg-emerald-500/30"
   994→                            : "bg-border/40"
   995→                        )}
   996→                      />
   997→                    )}

  1000→              }

  1002→          </div>
  1003→        </div>

  1005→        {/* Body */}
  1006→        <div className="flex-1 overflow-y-auto px-6 py-4">
  1007→          {phase === "upload" && renderUpload()}
  1008→          {phase === "detecting" && renderDetecting()}
  1009→          {phase === "preview" && renderPreview()}
  1010→          {phase === "settings" && renderSettings()}
  1011→        </div>
  1012→      </DialogContent>
  1013→    </Dialog>
  1014→  )
