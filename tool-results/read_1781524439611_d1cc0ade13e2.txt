    16→/* ─── Inline Statistical Engine ─── */
    17→function calcStats(values: number[]) {
    18→  const nums = values.filter(v => typeof v === 'number' && !isNaN(v))
    19→  if (nums.length === 0) return null
    20→  const n = nums.length
    21→  const s = nums.reduce((a, b) => a + b, 0)
    22→  const mean = s / n
    23→  const sorted = [...nums].sort((a, b) => a - b)
    24→  const median = n % 2 === 0 ? (sorted[n / 2 - 1] + sorted[n / 2]) / 2 : sorted[Math.floor(n / 2)]
    25→  const variance = nums.reduce((a, b) => a + (b - mean) ** 2, 0) / (n > 1 ? n - 1 : 1)
    26→  const stddev = Math.sqrt(variance)
    27→  const min = sorted[0]
    28→  const max = sorted[n - 1]
    29→  const freq: Record<number, number> = {}
    30→  nums.forEach(v => { freq[v] = (freq[v] || 0) + 1 })
    31→  const maxFreq = Math.max(...Object.values(freq))
    32→  const mode = nums.find(v => freq[v] === maxFreq)
    33→  const skewness = n > 2 ? (nums.reduce((a, b) => a + ((b - mean) / stddev) ** 3, 0) * n) / ((n - 1) * (n - 2)) : 0
    34→  const kurtosis = n > 3 ? (nums.reduce((a, b) => a + ((b - mean) / stddev) ** 4, 0) * n * (n + 1)) / ((n - 1) * (n - 2) * (n - 3)) - 3 * (n - 1) ** 2 / ((n - 2) * (n - 3)) : 0
    35→  const p25 = sorted[Math.floor(n * 0.25)]
    36→  const p50 = sorted[Math.floor(n * 0.5)]
    37→  const p75 = sorted[Math.floor(n * 0.75)]
    38→  return { n, sum: s, mean, median, mode, variance, stddev, min, max, range: max - min, skewness, kurtosis, p25, p50, p75 }
    39→}

    41→function calcCorrelation(x: number[], y: number[]): { r: number; n: number } | null {
    42→  const pairs: [number, number][] = []
    43→  for (let i = 0; i < Math.min(x.length, y.length); i++) {
    44→    if (typeof x[i] === 'number' && typeof y[i] === 'number' && !isNaN(x[i]) && !isNaN(y[i])) {
    45→      pairs.push([x[i], y[i]])
    46→    }
    47→  }
    48→  if (pairs.length < 3) return null
    49→  const n = pairs.length
    50→  const mx = pairs.reduce((a, p) => a + p[0], 0) / n
    51→  const my = pairs.reduce((a, p) => a + p[1], 0) / n
    52→  let num = 0, dx = 0, dy = 0
    53→  for (const [px, py] of pairs) {
    54→    num += (px - mx) * (py - my)
    55→    dx += (px - mx) ** 2
    56→    dy += (py - my) ** 2
    57→  }
    58→  const r = dx === 0 || dy === 0 ? 0 : num / Math.sqrt(dx * dy)
    59→  return { r, n }
    60→}

    62→function calcRegression(x: number[], y: number[]): { slope: number; intercept: number; r2: number; n: number } | null {
    63→  const pairs: [number, number][] = []
    64→  for (let i = 0; i < Math.min(x.length, y.length); i++) {
    65→    if (typeof x[i] === 'number' && typeof y[i] === 'number' && !isNaN(x[i]) && !isNaN(y[i])) {
    66→      pairs.push([x[i], y[i]])
    67→    }
    68→  }
    69→  if (pairs.length < 3) return null
    70→  const n = pairs.length
    71→  const mx = pairs.reduce((a, p) => a + p[0], 0) / n
    72→  const my = pairs.reduce((a, p) => a + p[1], 0) / n
    73→  let num = 0, den = 0
    74→  for (const [px, py] of pairs) {
    75→    num += (px - mx) * (py - my)
    76→    den += (px - mx) ** 2
    77→  }
    78→  if (den === 0) return null
    79→  const slope = num / den
    80→  const intercept = my - slope * mx
    81→  const ssRes = pairs.reduce((a, [px, py]) => a + (py - (slope * px + intercept)) ** 2, 0)
    82→  const ssTot = pairs.reduce((a, [, py]) => a + (py - my) ** 2, 0)
    83→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot
    84→  return { slope, intercept, r2, n }
    85→}

    87→function sum(arr: number[]): number {
    88→  return arr.reduce((a, b) => a + b, 0)
    89→}

   684→  const handleValidate = useCallback(() => {

   694→    let missingCells = 0

   696→    let typeErrors = 0
   697→    const emptyCols: string[] = []
   698→
   699→    for (const v of variables) {
   700→      const col = data[v.name] || []
   701→      let colMissing = 0
   702→      const values: number[] = []
   703→
   704→      for (let i = 0; i < col.length; i++) {
   705→        const val = col[i]
   706→        if (val === '' || val === null || val === undefined) {
   707→          colMissing++
   708→          missingCells++
   709→          issues.push({ row: i + 1, column: v.name, severity: 'warning', message: 'Missing value' })
   710→        } else if (v.type === 'numeric' && typeof val === 'string' && isNaN(parseFloat(val))) {
   711→          typeErrors++
   712→          issues.push({ row: i + 1, column: v.name, severity: 'error', message: `Non-numeric value "${val}" in numeric column` })
   713→        } else if (v.type === 'numeric') {
   714→          const num = typeof val === 'string' ? parseFloat(val) : val
   715→          if (typeof num === 'number' && !isNaN(num)) values.push(num)
   716→        }
   717→      }
   718→
   719→      if (colMissing === col.length && col.length > 0) {
   720→        emptyCols.push(v.name)
   721→        issues.push({ column: v.name, severity: 'error', message: 'Entire column is empty' })
   722→      }
   723→
   724→      // Check outliers using IQR method
   725→      if (values.length >= 4) {
   726→        const sorted = [...values].sort((a, b) => a - b)
   727→        const q1 = sorted[Math.floor(sorted.length * 0.25)]
   728→        const q3 = sorted[Math.floor(sorted.length * 0.75)]
   729→        const iqr = q3 - q1
   730→        const lower = q1 - 1.5 * iqr
   731→        const upper = q3 + 1.5 * iqr
   732→        for (let i = 0; i < col.length; i++) {
   733→          const val = col[i]
   734→          const num = typeof val === 'string' ? parseFloat(val) : val
   735→          if (typeof num === 'number' && !isNaN(num) && (num < lower || num > upper)) {
   736→            outlierCount++
   737→            if (issues.length < 50) {
   738→              issues.push({ row: i + 1, column: v.name, severity: 'info', message: `Potential outlier: ${num} (range: ${lower.toFixed(1)}\u2013${upper.toFixed(1)})` })
   739→            }
   740→          }
   741→        }
   742→      }
   743→    }
   744→
   745→    // Check duplicate rows
   746→    const rowCount = Math.max(0, ...Object.values(data).map(a => a.length))

   767→    // ── ALSO produce OutputPanel results so the user sees something ──

   774→
   775→    store.addOutput({
   776→      id: Date.now().toString(36),
   777→      title: 'Data Validation Report',
   778→      type: 'table',
   779→      content: {
   780→        headers: ['Metric', 'Value'],

   788→          ['Info (Outliers)', String(infos.length)],
   789→          ['Missing Cells', String(missingCells)],
   790→          ['Type Mismatches', String(typeErrors)],

   802→        title: `Validation Issues (${Math.min(30, issues.length)} of ${totalIssues})`,
   803→        type: 'table',
   804→        content: {
   805→          headers: ['Severity', 'Row', 'Column', 'Message'],

   809→            i.column,
   810→            i.message,
   811→          ]),
   812→        },
   813→        timestamp: new Date().toISOString(),
   814→      })
   815→    } else {
   816→      store.addOutput({
   817→        id: (Date.now() + 1).toString(36),
   818→        title: 'Validation Result',
   819→        type: 'text',
   820→        content: 'All checks passed. Your data looks clean with no issues detected.',
   821→        timestamp: new Date().toISOString(),
   822→      })
   823→    }
   824→
   825→    store.addSyntax(`DATA VALIDATION /VARIABLES=ALL`)
   826→      } finally {

   832→  /* ─── Data Cleaning (non-blocking) ─── */
   833→  const handleClean = useCallback(() => {

   838→    setTimeout(() => {

   844→    let filledCount = 0

   862→        const mean = nums.length > 0 ? nums.reduce((a, b) => a + b, 0) / nums.length : 0

   867→          }

   871→        const freq = new Map<string, number>()
   872→        nonEmpty.forEach(x => freq.set(String(x), (freq.get(String(x)) || 0) + 1))
   873→        let mode = ''
   874→        let maxFreq = 0
   875→        freq.forEach((count, val) => { if (count > maxFreq) { maxFreq = count; mode = val } })
   876→        for (let i = 0; i < col.length; i++) {
   877→          if (col[i] === '' || col[i] === null || col[i] === undefined) {
   878→            col[i] = mode
   879→            filledCount++
   880→          }
   881→        }
   882→      }
   883→
   884→      data[v.name] = col
   885→    }
   886→
   887→    // Remove duplicate rows
   888→    const rowCount = Math.max(0, ...Object.values(data).map(a => a.length))
   889→    const keepIndices: number[] = []
   890→    const seen = new Set<string>()
   891→    for (let i = 0; i < rowCount; i++) {
   892→      const key = variables.map(v => String(data[v.name]?.[i] ?? '')).join('|')
   893→      if (!seen.has(key)) {
   894→        seen.add(key)
   895→        keepIndices.push(i)
   896→      }
   897→    }
   898→
   899→    const removedRows = rowCount - keepIndices.length
   900→    if (keepIndices.length < rowCount) {
   901→      for (const v of variables) {
   902→        data[v.name] = keepIndices.map(i => data[v.name][i])
   903→      }
   904→    }
   905→

   910→    store.addOutput({

   912→      title: 'Data Cleaning Report',
   913→      type: 'table',
   914→      content: {
   915→        headers: ['Metric', 'Before', 'After', 'Change'],
   916→        rows: [
   917→          ['Total Rows', String(rowCount), String(newRowCount), removedRows > 0 ? `${removedRows} removed` : 'No change'],
   918→          ['Whitespace Trimmed', '-', String(trimmedCount), trimmedCount > 0 ? `${trimmedCount} cells` : 'None needed'],
   919→          ['Missing Values Filled', '-', String(filledCount), filledCount > 0 ? `${filledCount} cells` : 'None found'],
   920→          ['Duplicate Rows', '-', '-', removedRows > 0 ? `${removedRows} removed` : 'None found'],

   956→  /* ─── Data Transformations (non-blocking) ─── */
   957→  const handleTransformZScore = useCallback(() => {
   958→    if (isTransforming) return
   959→    setIsTransforming(true)
   960→    // Snapshot current state to avoid stale closures
   961→    const dataSnap: Record<string, any[]> = {}
   962→    for (const k of Object.keys(store.data)) dataSnap[k] = [...store.data[k]]
   963→    const selectedVars = [...store.selectedVariables]
   964→
   965→    setTimeout(() => {
   966→      try {
   967→        const data: Record<string, any[]> = { ...dataSnap }
   968→        const createdVars: string[] = []
   969→
   970→        for (const varName of selectedVars) {
   971→          const vals = (data[varName] || []).map((v: any) => typeof v === 'string' ? parseFloat(v) : v).filter((v: any): v is number => typeof v === 'number' && !isNaN(v))
   972→          if (vals.length < 2) continue
   973→          const m = vals.reduce((a, b) => a + b, 0) / vals.length
   974→          const s = Math.sqrt(vals.reduce((a, b) => a + (b - m) ** 2, 0) / (vals.length - 1))
   975→          if (s === 0) continue
   976→          const col = data[varName] || []
   977→          const newName = `z_${varName}`
   978→          data[newName] = col.map(v => {
   979→            const n = typeof v === 'string' ? parseFloat(v) : v
   980→            return typeof n === 'number' && !isNaN(n) ? parseFloat(((n - m) / s).toFixed(4)) : ''
   981→          })
   982→          store.addVariable({
   983→            id: Date.now().toString(36) + varName,
   984→            name: newName,
   985→            type: 'numeric',
   986→            label: `Z-score of ${varName}`,
   987→            width: 10, decimals: 4, missing: '', values: {},
   988→          })
   989→          createdVars.push(varName)
   990→        }
   991→
   992→        store.setData(data)
   993→        store.addOutput({
   994→          id: Date.now().toString(36),
   995→          title: 'Z-Score Transformation',
   996→          type: 'table',
   997→          content: {
   998→            headers: ['Original', 'Transformed', 'Formula'],
   999→            rows: createdVars.map(v => [v, `z_${v}`, '(x − mean) / SD']),
  1000→          },
  1001→          timestamp: new Date().toISOString(),
  1002→        })
  1003→        store.addSyntax(`COMPUTE z_*= (var - MEAN(var)) / SD(var)`)
  1004→      } finally {
  1005→        setIsTransforming(false)
  1006→      }
  1007→    }, 0)
  1008→  }, [store, isTransforming])

  1010→  const handleTransformNormalize = useCallback(() => {
  1011→    if (isTransforming) return
  1012→    setIsTransforming(true)

  1015→    const selectedVars = [...store.selectedVariables]
  1016→
  1017→    setTimeout(() => {
  1018→      try {
  1019→        const data: Record<string, any[]> = { ...dataSnap }
  1020→        const createdVars: string[] = []
  1021→
  1022→        for (const varName of selectedVars) {
  1023→          const vals = (data[varName] || []).map((v: any) => typeof v === 'string' ? parseFloat(v) : v).filter((v: any): v is number => typeof v === 'number' && !isNaN(v))
  1024→          if (vals.length < 2) continue
  1025→          const min = Math.min(...vals)
  1026→          const max = Math.max(...vals)
  1027→          const range = max - min
  1028→          if (range === 0) continue
  1029→          const col = data[varName] || []
  1030→          const newName = `norm_${varName}`
  1031→          data[newName] = col.map(v => {
  1032→            const n = typeof v === 'string' ? parseFloat(v) : v

  1034→          })
  1035→          store.addVariable({
  1036→            id: Date.now().toString(36) + varName,
  1037→            name: newName,
  1038→            type: 'numeric',
  1039→            label: `Normalized ${varName}`,
  1040→            width: 10, decimals: 4, missing: '', values: {},
  1041→          })
  1042→          createdVars.push(varName)
  1043→        }
  1044→
  1045→        store.setData(data)
  1046→        store.addOutput({
  1047→          id: Date.now().toString(36),
  1048→          title: 'Min-Max Normalization (0–1)',
  1049→          type: 'table',
  1050→          content: {
  1051→            headers: ['Original', 'Transformed', 'Formula'],
  1052→            rows: createdVars.map(v => [v, `norm_${v}`, '(x − min) / (max − min)']),
  1053→          },
  1054→          timestamp: new Date().toISOString(),
  1055→        })

  1063→  const handleTransformLog = useCallback(() => {

  1111→  const handleAutoProfile = useCallback(() => {
