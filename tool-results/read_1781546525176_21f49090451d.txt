   175→function rowCount(data: Record<string, any[]>): number {
   176→  const keys = Object.keys(data)

   178→  return Math.max(...keys.map(k => data[k]?.length ?? 0))

   181→// ── localStorage size monitoring ──
   182→const STORAGE_WARNING_THRESHOLD = 3 * 1024 * 1024 // 3MB — warn before 4MB hard limit
   183→let _lastStorageWarning = 0
   184→
   185→export function getStorageUsage(): { usedBytes: number; usedMB: number; percentage: number } {
   186→  if (typeof window === 'undefined') return { usedBytes: 0, usedMB: 0, percentage: 0 }
   187→  let total = 0
   188→  for (let i = 0; i < localStorage.length; i++) {
   189→    const key = localStorage.key(i)
   190→    if (key) total += (localStorage.getItem(key) || '').length
   191→  }
   192→  // Typical browser limit is 5-10MB; use 5MB as reference
   193→  const limit = 5 * 1024 * 1024
   194→  return { usedBytes: total, usedMB: Math.round(total / 1024 / 1024 * 100) / 100, percentage: Math.round(total / limit * 100) }
   195→}

   197→function checkStorageSize() {
   198→  if (typeof window === 'undefined') return
   199→  const { usedBytes } = getStorageUsage()
   200→  const now = Date.now()
   201→  // Throttle warnings to once per 60 seconds
   202→  if (usedBytes > STORAGE_WARNING_THRESHOLD && now - _lastStorageWarning > 60_000) {
   203→    _lastStorageWarning = now
   204→    const mb = (usedBytes / 1024 / 1024).toFixed(1)
   205→    console.warn(`[STORE] localStorage usage at ${mb}MB — consider exporting old projects and clearing history`)
   206→  }
   207→}
