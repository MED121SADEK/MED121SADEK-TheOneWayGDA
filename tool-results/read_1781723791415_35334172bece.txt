   121→   121→function makeRequest(url: string, options: RequestInit = {}): NextRequest {
   122→   122→  return new NextRequest(new URL(url, 'http://localhost:3000'), options)
   123→   123→}
   124→   124→function makeJsonRequest(url: string, body: unknown, options: RequestInit = {}): NextRequest {
   125→   125→  return new NextRequest(new URL(url, 'http://localhost:3000'), {
   126→   126→    ...options, method: 'POST',
   127→   127→    headers: { 'Content-Type': 'application/json', ...options.headers },
   128→   128→    body: JSON.stringify(body),
   129→   129→  })
   130→   130→}
   131→   131→function authedRequest(url: string, token: string, options: RequestInit = {}): NextRequest {
   132→   132→  return makeRequest(url, { ...options, headers: { Authorization: `Bearer ${token}`, ...options.headers } })
   133→   133→}
