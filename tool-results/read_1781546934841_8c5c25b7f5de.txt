    64→function getSession(): { email: string; name: string } | null {
    65→  if (typeof window === 'undefined') return null
    66→  try {
    67→    const s = localStorage.getItem('oneway-visitor-session')
    68→    if (!s) return null
    69→    return JSON.parse(s)
    70→  } catch { return null }
    71→}

    73→function timeAgo(date: string): string {
    74→  const now = Date.now()
    75→  const d = new Date(date).getTime()
    76→  const diff = now - d
    77→  const mins = Math.floor(diff / 60000)
    78→  if (mins < 1) return 'just now'
    79→  if (mins < 60) return `${mins}m ago`
    80→  const hrs = Math.floor(mins / 60)
    81→  if (hrs < 24) return `${hrs}h ago`
    82→  const days = Math.floor(hrs / 24)
    83→  if (days < 7) return `${days}d ago`
    84→  return new Date(date).toLocaleDateString()
    85→}

    87→function parseTags(tags: string | null | undefined): string[] {
    88→  if (!tags) return []

    92→function getPostTypeInfo(type: string): { label: string; icon: any; bg: string; border: string; text: string } {
    93→  switch (type) {
    94→    case 'auto':
    95→      return { label: 'Auto-Published', icon: Sparkles, bg: 'bg-blue-500/10', border: 'border-blue-500/30', text: 'text-blue-400' }
    96→    case 'digest':
    97→      return { label: 'Daily Digest', icon: FileText, bg: 'bg-violet-500/10', border: 'border-violet-500/30', text: 'text-violet-400' }
    98→    case 'user_highlight':
    99→      return { label: 'Community Pick', icon: Award, bg: 'bg-amber-500/10', border: 'border-amber-500/30', text: 'text-amber-400' }
   100→    case 'news':
   101→      return { label: 'AI News', icon: Newspaper, bg: 'bg-sky-500/10', border: 'border-sky-500/30', text: 'text-sky-400' }
   102→    default:
   103→      return { label: 'Community', icon: Users, bg: 'bg-emerald-500/10', border: 'border-emerald-500/30', text: 'text-emerald-400' }
   104→  }
   105→}

   107→function getBadgeColor(badgeType: string) {
   108→  switch (badgeType) {
   109→    case 'bot': return { border: 'rgba(99,102,241,0.4)', bg: 'rgba(99,102,241,0.1)', color: '#818cf8', label: 'Bot' }
   110→    case 'institution': return { border: 'rgba(168,85,247,0.4)', bg: 'rgba(168,85,247,0.1)', color: '#c084fc', label: 'Institution' }
   111→    case 'official': return { border: 'rgba(59,130,246,0.4)', bg: 'rgba(59,130,246,0.1)', color: '#60a5fa', label: 'Official' }
   112→    default: return { border: 'rgba(16,185,129,0.4)', bg: 'rgba(16,185,129,0.1)', color: '#34d399', label: 'Verified' }
   113→  }
   114→}
