    88→/* ─── Helpers ─── */
    89→function getSession(): { token: string; user: UserData } | null {
    90→  if (typeof window === 'undefined') return null
    91→  try {
    92→    const token = localStorage.getItem('oneway-auth-token')
    93→    const userStr = localStorage.getItem('oneway-user')
    94→    if (!token || !userStr) return null
    95→    return { token, user: JSON.parse(userStr) }
    96→  } catch {
    97→    return null
    98→  }
    99→}

   101→function timeAgo(date: string): string {
   102→  const diff = Date.now() - new Date(date).getTime()
   103→  const mins = Math.floor(diff / 60000)
   104→  if (mins < 1) return 'Just now'
   105→  if (mins < 60) return `${mins}m ago`
   106→  const hrs = Math.floor(mins / 60)
   107→  if (hrs < 24) return `${hrs}h ago`
   108→  const days = Math.floor(hrs / 24)
   109→  return `${days}d ago`
   110→}

   113→  switch (type) {

   115→    case 'project_created': return { icon: FolderOpen, color: 'text-blue-400 bg-blue-500/10', label: 'Project Created' }
   116→    case 'analysis_run': return { icon: BarChart3, color: 'text-purple-400 bg-purple-500/10', label: 'Analysis Run' }
   117→    case 'report_generated': return { icon: FileText, color: 'text-sky-400 bg-sky-500/10', label: 'Report Generated' }
   118→    case 'workflow_executed': return { icon: Workflow, color: 'text-orange-400 bg-orange-500/10', label: 'Workflow Executed' }

   121→    default: return { icon: Activity, color: 'text-muted-foreground bg-muted/30', label: type }
