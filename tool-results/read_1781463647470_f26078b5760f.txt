    11→    11→// ─── Live Data: Fetch real benchmark data from the database ───
    12→    12→async function fetchLiveBenchmarkData(): Promise<string> {
    13→    13→  try {
    14→    14→    const models = await prisma.aiModel.findMany({
    15→    15→      where: { isActive: true },
    16→    16→      include: {
    17→    17→        benchmarkScores: true,
    18→    18→        pricing: { where: { isActive: true }, take: 1 },
    19→    19→      },
    20→    20→      orderBy: { name: 'asc' },
    21→    21→    })
    22→    22→
    23→    23→    if (models.length === 0) return ''
    24→    24→
    25→    25→    const benchmarkNames = [...new Set(models.flatMap(m => m.benchmarkScores.map(b => b.benchmark)))]
    26→    26→    const pricingMap: Record<string, { input: number; output: number }> = {}
    27→    27→
    28→    28→    // Build benchmark comparison table
    29→    29→    const header = '| Model | ' + benchmarkNames.map(b => b).join(' | ') + ' | Price (in/out per 1M) |'
    30→    30→    const separator = '|---|' + benchmarkNames.map(() => '---:').join('|') + '|---:|'
    31→    31→
    32→    32→    const rows = models.map(model => {
    33→    33→      const scores = benchmarkNames.map(bench => {
    34→    34→        const score = model.benchmarkScores.find(s => s.benchmark === bench)
    35→    35→        return score ? `${score.score}/${score.maxScore}` : '-'
    36→    36→      })
    37→    37→      const price = model.pricing[0]
    38→    38→      const priceStr = price ? `$${price.inputPrice}/$${price.outputPrice}` : '-'
    39→    39→      pricingMap[model.name] = price ? { input: price.inputPrice, output: price.outputPrice } : { input: 0, output: 0 }
    40→    40→      return `| ${model.name} (${model.provider}) | ${scores.join(' | ')} | ${priceStr} |`
    41→    41→    }).join('\n')
    42→    42→
    43→    43→    // Find top model per benchmark
    44→    44→    const topPerBenchmark = benchmarkNames.map(bench => {
    45→    45→      const best = models
    46→    46→        .map(m => ({ name: m.name, score: m.benchmarkScores.find(s => s.benchmark === bench)?.score || 0 }))
    47→    47→        .sort((a, b) => b.score - a.score)[0]
    48→    48→      return `  - ${bench}: ${best.name} (${best.score})`
    49→    49→    }).join('\n')
    50→    50→
    51→    51→    return `
    52→    52→## LIVE LEADERBOARD DATA (from database, not hallucinated)
    53→    53→${header}
    54→    54→${separator}
    55→    55→${rows}
    56→    56→
    57→    57→### Top Models per Benchmark
    58→    58→${topPerBenchmark}
    59→    59→
    60→    60→### Pricing Reference
    61→    61→${Object.entries(pricingMap).map(([name, p]) => `  - ${name}: $${p.input}/$${p.output} per 1M tokens`).join('\n')}
    62→    62→
    63→    63→> CRITICAL: Always use these REAL numbers when comparing models. Never fabricate benchmark scores. If you don't have data for a specific model, say so explicitly.`
    64→    64→  } catch {
    65→    65→    return ''
    66→    66→  }
    67→    67→}

    69→    69→// ─── Conversation History: Fetch recent conversation for continuity ───
    70→    70→async function fetchConversationHistory(visitorId: string | null | undefined, currentSessionId?: string): Promise<string> {
    71→    71→  if (!visitorId) return ''
    72→    72→  try {
    73→    73→    const recentConversations = await prisma.aiConversation.findMany({
    74→    74→      where: { visitorId },
    75→    75→      orderBy: { updatedAt: 'desc' },
    76→    76→      take: 3,
    77→    77→      select: { id: true, sessionId: true, context: true, messages: true, updatedAt: true },
    78→    78→    })
    79→    79→
    80→    80→    if (recentConversations.length === 0) return ''
    81→    81→
    82→    82→    const parts: string[] = ['### Recent Conversations (for continuity)']
    83→    83→    for (const conv of recentConversations) {
    84→    84→      if (conv.id === currentSessionId) continue // Skip current session
    85→    85→      try {
    86→    86→        const msgs = JSON.parse(conv.messages) as Array<{ role: string; content: string }>
    87→    87→        const lastExchange = msgs.slice(-4) // Last 2 exchanges
    88→    88→        if (lastExchange.length > 0) {
    89→    89→          const summary = lastExchange.map(m => `${m.role}: ${m.content.slice(0, 150)}...`).join('\n  ')
    90→    90→          parts.push(`- [${conv.context}, ${conv.updatedAt.toISOString().slice(0, 10)}]:\n  ${summary}`)
    91→    91→        }
    92→    92→      } catch {
    93→    93→        // Skip malformed conversations
    94→    94→      }
    95→    95→    }
    96→    96→
    97→    97→    return parts.length > 1 ? '\n\n[CONVERSATION HISTORY]\n' + parts.join('\n') + '\nUse this to maintain conversation continuity — reference past topics naturally when relevant.' : ''
    98→    98→  } catch {
    99→    99→    return ''
   100→   100→  }
   101→   101→}

   480→   480→export async function POST(request: NextRequest) {
   481→   481→  const startTime = Date.now()
   482→   482→  let visitorId = request.headers.get('x-visitor-id')
   483→   483→  const ipAddress = request.headers.get('x-forwarded-for') || request.headers.get('x-real-ip') || null
   484→   484→  const userAgent = request.headers.get('user-agent') || null
   485→   485→
   486→   486→  try {
   487→   487→    const body = await request.json()
   488→   488→    const { messages, context = 'general', pageData, stream = false } = body
   489→   489→
   490→   490→    if (!messages || !Array.isArray(messages) || messages.length === 0) {
   491→   491→      return NextResponse.json({ error: 'Messages array is required' }, { status: 400 })
   492→   492→    }
   493→   493→
   494→   494→    const zai = await ZAI.create()
   495→   495→    const sessionId = body.sessionId || `session_${Date.now()}`
   496→   496→
   497→   497→    // ─── Parallel data fetch for maximum context ───
   498→   498→    const [baseSystemPrompt, liveData, memoryContext, conversationHistory] = await Promise.all([
   499→   499→      Promise.resolve(getSystemPrompt(context, pageData)),
   500→   500→      fetchLiveBenchmarkData(),
   501→   501→      fetchMemoryContext(visitorId),
   502→   502→      fetchConversationHistory(visitorId, sessionId),
   503→   503→    ])
   504→   504→
   505→   505→    // Assemble the full system prompt with all context layers
   506→   506→    const systemPrompt = baseSystemPrompt
   507→   507→      + (liveData ? '\n\n' + liveData : '')
   508→   508→      + memoryContext
   509→   509→      + conversationHistory
   510→   510→
   511→   511→    // Build user messages preserving conversation flow
   512→   512→    const userMessages = messages.map((m: { role: string; content: string }) => ({
   513→   513→      role: m.role as 'user' | 'assistant',
   514→   514→      content: m.content,
   515→   515→    }))
   516→   516→
   517→   517→    // ─── Streaming mode ───
   518→   518→    if (stream) {
   519→   519→      const encoder = new TextEncoder()
   520→   520→      const streamResponse = new ReadableStream({
   521→   521→        async start(controller) {
   522→   522→          try {
   523→   523→            const completion = await zai.chat.completions.create({
   524→   524→              messages: [
   525→   525→                { role: 'system', content: systemPrompt },
   526→   526→                ...userMessages,
   527→   527→              ],
   528→   528→              max_tokens: 8192,
   529→   529→              temperature: 0.7,
   530→   530→              stream: true as unknown as boolean,
   531→   531→            } as any)
   532→   532→
   533→   533→            if (completion && typeof completion === 'object') {
   534→   534→              const streamable = completion as any
   535→   535→              if (streamable.body && typeof streamable.body.getReader === 'function') {
   536→   536→                const reader = streamable.body.getReader()
   537→   537→                while (true) {
   538→   538→                  const { done, value } = await reader.read()
   539→   539→                  if (done) break
   540→   540→                  controller.enqueue(value)
   541→   541→                }
   542→   542→              } else if (Array.isArray(streamable.choices)) {
   543→   543→                const text = streamable.choices[0]?.message?.content || ''
   544→   544→                controller.enqueue(encoder.encode(`data: ${JSON.stringify({ choices: [{ delta: { content: text } }] })}\n\n`))
   545→   545→              }
   546→   546→            }
   547→   547→
   548→   548→            controller.enqueue(encoder.encode('data: [DONE]\n\n'))
   549→   549→            controller.close()
   550→   550→          } catch (error) {
   551→   551→            const errorMsg = error instanceof Error ? error.message : 'Stream error'
   552→   552→            controller.enqueue(encoder.encode(`data: ${JSON.stringify({ error: errorMsg })}\n\n`))
   553→   553→            controller.close()
   554→   554→          }
   555→   555→        },
   556→   556→      })
   557→   557→
   558→   558→      // Fire-and-forget: audit log + conversation save
   559→   559→      const lastUserMsg = messages[messages.length - 1]?.content || ''
   560→   560→      ;(async () => {
   561→   561→        try {
   562→   562→          await prisma.aiAuditLog.create({
   563→   563→            data: {
   564→   564→              visitorId: visitorId || null,
   565→   565→              action: 'ai_query',
   566→   566→              details: JSON.stringify({ context, messageCount: messages.length, mode: 'stream', version: 'v4', withLiveData: !!liveData, withHistory: !!conversationHistory }),
   567→   567→              inputData: JSON.stringify({ lastUserMessage: lastUserMsg.slice(0, 1000) }),
   568→   568→              durationMs: Date.now() - startTime,
   569→   569→              ipAddress,
   570→   570→              userAgent,
   571→   571→            },
   572→   572→          })
   573→   573→        } catch { /* silent */ }
   574→   574→      })()
   575→   575→
   576→   576→      return new Response(streamResponse, {
   577→   577→        headers: {
   578→   578→          'Content-Type': 'text/event-stream',
   579→   579→          'Cache-Control': 'no-cache',
   580→   580→          'Connection': 'keep-alive',
   581→   581→        },
   582→   582→      })
   583→   583→    }
   584→   584→
   585→   585→    // ─── Standard mode (deep response) ───
   586→   586→    const completion = await zai.chat.completions.create({
   587→   587→      messages: [
   588→   588→        { role: 'system', content: systemPrompt },
   589→   589→        ...userMessages,
   590→   590→      ],
   591→   591→      max_tokens: 8192,
   592→   592→      temperature: 0.7,
   593→   593→    })
   594→   594→
   595→   595→    const aiMessage = completion.choices?.[0]?.message?.content || 'I apologize, I could not generate a response.'
   596→   596→    const tokensUsed = completion.usage?.total_tokens || 0
   597→   597→    const durationMs = Date.now() - startTime
   598→   598→
   599→   599→    // Audit log
   600→   600→    try {
   601→   601→      await prisma.aiAuditLog.create({
   602→   602→        data: {
   603→   603→          visitorId: visitorId || null,
   604→   604→          action: 'ai_query',
   605→   605→          details: JSON.stringify({
   606→   606→            context, messageCount: messages.length, mode: 'standard', version: 'v4',
   607→   607→            withLiveData: !!liveData, withMemory: !!memoryContext, withHistory: !!conversationHistory,
   608→   608→          }),
   609→   609→          inputData: JSON.stringify({ lastUserMessage: messages[messages.length - 1]?.content?.slice(0, 1000) }),
   610→   610→          outputData: JSON.stringify({ responseLength: aiMessage.length, tokensUsed }),
   611→   611→          tokensUsed,
   612→   612→          durationMs,
   613→   613→          ipAddress,
   614→   614→          userAgent,
   615→   615→        },
   616→   616→      })
   617→   617→    } catch {
   618→   618→      // Audit log failure should not block the response
   619→   619→    }
   620→   620→
   621→   621→    // Save conversation with proper history
   622→   622→    try {
   623→   623→      await prisma.aiConversation.upsert({
   624→   624→        where: { id: sessionId },
   625→   625→        create: {
   626→   626→          id: sessionId,
   627→   627→          sessionId,
   628→   628→          visitorId: visitorId || null,
   629→   629→          context,
   630→   630→          messages: JSON.stringify([...messages, { role: 'assistant', content: aiMessage, timestamp: new Date().toISOString() }]),
   631→   631→        },
   632→   632→        update: {
   633→   633→          messages: JSON.stringify([...messages, { role: 'assistant', content: aiMessage, timestamp: new Date().toISOString() }]),
   634→   634→          updatedAt: new Date(),
   635→   635→        },
   636→   636→      })
   637→   637→    } catch {
   638→   638→      // Conversation save failure should not block the response
   639→   639→    }
   640→   640→
   641→   641→    return NextResponse.json({
   642→   642→      message: aiMessage,
   643→   643→      meta: {
   644→   644→        tokensUsed,
   645→   645→        durationMs,
   646→   646→        context,
   647→   647→        version: 'v4',
   648→   648→        contextLayers: {
   649→   649→          liveData: !!liveData,
   650→   650→          memory: !!memoryContext,
   651→   651→          conversationHistory: !!conversationHistory,
   652→   652→        },
   653→   653→      },
   654→   654→    })
   655→   655→  } catch (error: unknown) {
   656→   656→    const durationMs = Date.now() - startTime
   657→   657→    const errorMsg = error instanceof Error ? error.message : 'Unknown error'
   658→   658→
   659→   659→    try {
   660→   660→      await prisma.aiAuditLog.create({
   661→   661→        data: {
   662→   662→          visitorId: visitorId || null,
   663→   663→          action: 'ai_query',
   664→   664→          details: JSON.stringify({ context: 'general', error: true, version: 'v4' }),
   665→   665→          error: errorMsg,
   666→   666→          durationMs,
   667→   667→          ipAddress,
   668→   668→          userAgent,
   669→   669→        },
   670→   670→      })
   671→   671→    } catch { /* silent */ }
   672→   672→
   673→   673→    return NextResponse.json({ error: 'Failed to get AI response' }, { status: 500 })
   674→   674→  }
   675→   675→}

   688→   688→        createdAt: { gte: new Date(Date.now() - 24 * 60 * 60 * 1000) },
   689→   689→      },

   691→   691→      take: 5,
   692→   692→    })
