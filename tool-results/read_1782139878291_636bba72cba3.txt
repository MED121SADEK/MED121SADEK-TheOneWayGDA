    64→/* ─── Helpers ─── */
    65→function getSession(): { token: string; user: UserData } | null {
    66→  if (typeof window === 'undefined') return null
    67→  try {
    68→    const token = localStorage.getItem('oneway-auth-token')
    69→    const userStr = localStorage.getItem('oneway-user')
    70→    if (!token || !userStr) return null
    71→    return { token, user: JSON.parse(userStr) }
    72→  } catch { return null }
    73→}

    75→function timeAgo(date: string): string {
    76→  const diff = Date.now() - new Date(date).getTime()
    77→  const mins = Math.floor(diff / 60000)
    78→  if (mins < 1) return 'Just now'
    79→  if (mins < 60) return `${mins}m ago`
    80→  const hrs = Math.floor(mins / 60)
    81→  if (hrs < 24) return `${hrs}h ago`
    82→  const days = Math.floor(hrs / 24)
    83→  return `${days}d ago`
    84→}
