   106→   185→export function getStorageUsage(): { usedBytes: number; usedMB: number; percentage: number } {
   107→   186→  if (typeof window === 'undefined') return { usedBytes: 0, usedMB: 0, percentage: 0 }
   108→   187→  let total = 0
   109→   188→  for (let i = 0; i < localStorage.length; i++) {
   110→   189→    const key = localStorage.key(i)
   111→   190→    if (key) total += (localStorage.getItem(key) || '').length
   112→   191→  }
   113→   192→  // Typical browser limit is 5-10MB; use 5MB as reference
   114→   193→  const limit = 5 * 1024 * 1024
   115→   194→  return { usedBytes: total, usedMB: Math.round(total / 1024 / 1024 * 100) / 100, percentage: Math.round(total / limit * 100) }
   116→   195→}

   118→   197→function checkStorageSize() {
   119→   198→  if (typeof window === 'undefined') return
   120→   199→  const { usedBytes } = getStorageUsage()
   121→   200→  const now = Date.now()
   122→   201→  // Throttle warnings to once per 60 seconds
   123→   202→  if (usedBytes > STORAGE_WARNING_THRESHOLD && now - _lastStorageWarning > 60_000) {
   124→   203→    _lastStorageWarning = now
   125→   204→    const mb = (usedBytes / 1024 / 1024).toFixed(1)
   126→   205→    console.warn(`[STORE] localStorage usage at ${mb}MB — consider exporting old projects and clearing history`)
   127→   206→  }
