    33→/* ─── MAIN PAGE ─── */
    34→export default function CommunityPage() {
    35→  const { t, locale, setLocale, dir } = useTranslation()
    36→  const session = getSession()
    37→  const [posts, setPosts] = useState<Post[]>([])
    38→  const [loading, setLoading] = useState(true)
    39→  const [page, setPage] = useState(1)
    40→  const [hasMore, setHasMore] = useState(true)
    41→  const [activeTab, setActiveTab] = useState('all')
    42→  const [sortOrder, setSortOrder] = useState('latest')
    43→  const [searchQuery, setSearchQuery] = useState('')
    44→  const [searchInput, setSearchInput] = useState('')
    45→  const observerRef = useRef<HTMLDivElement>(null)
    46→
    47→  // Composer state
    48→  const [composerOpen, setComposerOpen] = useState(false)
    49→  const [composerTitle, setComposerTitle] = useState('')
    50→  const [composerContent, setComposerContent] = useState('')
    51→  const [composerImageUrl, setComposerImageUrl] = useState('')
    52→  const [composerLink, setComposerLink] = useState('')
    53→  const [composerSubmitting, setComposerSubmitting] = useState(false)
    54→
    55→  // Comment state
    56→  const [commentingPostId, setCommentingPostId] = useState<string | null>(null)
    57→  const [commentText, setCommentText] = useState('')
    58→  const [commentsMap, setCommentsMap] = useState<Record<string, Comment[]>>({})
    59→  const [commentsLoading, setCommentsLoading] = useState<string | null>(null)
    60→
    61→  // Interaction state (client-side cache)
    62→  const [likedPosts, setLikedPosts] = useState<Set<string>>(new Set())
    63→  const [savedPosts, setSavedPosts] = useState<Set<string>>(new Set())
    64→  const [repostedPosts, setRepostedPosts] = useState<Set<string>>(new Set())
    65→
    66→  // Share dialog
    67→  const [shareDialogPost, setShareDialogPost] = useState<Post | null>(null)
    68→  const [shareEmail, setShareEmail] = useState('')
    69→  const [copiedLink, setCopiedLink] = useState(false)
    70→
    71→  // News fetching
    72→  const [newsLoading, setNewsLoading] = useState(false)
    73→  const [newsFetchedAt, setNewsFetchedAt] = useState<string | null>(null)
    74→
    75→  // Category filter (AI | Research | Innovation)
    76→  const [categoryFilter, setCategoryFilter] = useState<string>('all')
    77→
    78→  // Topic following
    79→  const [followedTopics, setFollowedTopics] = useState<string[]>([])
    80→
    81→  // Verified researchers cache
    82→  const [verifiedMap, setVerifiedMap] = useState<Record<string, VerifiedInfo>>({})
    83→
    84→  /* ─── Fetch Verified Researchers ─── */
    85→  useEffect(() => {
    86→    const fetchVerified = async () => {
    87→      try {
    88→        const res = await fetch('/api/community/verified')
    89→        const data = await res.json()
    90→        if (data.researchers) {
    91→          const map: Record<string, VerifiedInfo> = {}
    92→          data.researchers.forEach((r: VerifiedInfo) => { map[r.email] = r })
    93→          setVerifiedMap(map)
    94→        }
    95→      } catch { /* silent */ }
    96→    }
    97→    fetchVerified()
    98→  }, [])
    99→
   100→  /* ─── Fetch Posts ─── */
   101→  const fetchPosts = useCallback(async (pageNum: number, append: boolean = false) => {
   102→    try {
   103→      const params = new URLSearchParams({ page: String(pageNum), limit: '15', sort: sortOrder })
   104→      if (activeTab !== 'saved') params.set('type', activeTab === 'all' ? 'all' : activeTab)
   105→      if (searchQuery) params.set('search', searchQuery)
   106→      const res = await fetch(`/api/community/posts?${params}`)
   107→      const data = await res.json()
   108→      if (append) {
   109→        setPosts(prev => [...prev, ...data.posts])
   110→      } else {
   111→        setPosts(data.posts)
   112→      }
   113→      setHasMore(data.pagination.page < data.pagination.pages)
   114→    } catch (err) {
   115→      console.error('Fetch posts error:', err)
   116→    }
   117→  }, [activeTab, sortOrder, searchQuery])
   118→
   119→  useEffect(() => {
   120→    setLoading(true)
   121→    setPage(1)
   122→    fetchPosts(1, false).finally(() => setLoading(false))
   123→  }, [activeTab, sortOrder, searchQuery, fetchPosts])
   124→
   125→  // Infinite scroll
   126→  useEffect(() => {
   127→    if (!observerRef.current || !hasMore) return
   128→    const observer = new IntersectionObserver(
   129→      (entries) => {
   130→        if (entries[0].isIntersecting && hasMore && !loading) {
   131→          const nextPage = page + 1
   132→          setPage(nextPage)
   133→          fetchPosts(nextPage, true)
   134→        }
   135→      },
   136→      { threshold: 0.5 }
   137→    )
   138→    observer.observe(observerRef.current)
   139→    return () => observer.disconnect()
   140→  }, [hasMore, loading, page, fetchPosts])
   141→
   142→  // Fetch user interactions for visible posts
   143→  useEffect(() => {
   144→    if (!session || posts.length === 0) return
   145→    const fetchInteractions = async () => {
   146→      try {
   147→        const results = await Promise.all(
   148→          posts.slice(0, 20).map(async (post) => {
   149→            const res = await fetch(`/api/community/posts/${post.id}/interact?visitorId=${session.email}`)
   150→            return { postId: post.id, ...(await res.json()) }
   151→          })
   152→        )
   153→        const liked = new Set<string>()
   154→        const saved = new Set<string>()
   155→        const reposted = new Set<string>()
   156→        results.forEach(r => {
   157→          if (r.liked) liked.add(r.postId)
   158→          if (r.saved) saved.add(r.postId)
   159→          if (r.reposted) reposted.add(r.postId)
   160→        })
   161→        setLikedPosts(liked)
   162→        setSavedPosts(saved)
   163→        setRepostedPosts(reposted)
   164→      } catch { /* silent */ }
   165→    }
   166→    fetchInteractions()
   167→  }, [posts, session])
   168→
   169→  /* ─── Actions ─── */
   170→  const handleLike = async (postId: string) => {
   171→    if (!session) return
   172→    const isLiked = likedPosts.has(postId)
   173→    const action = isLiked ? 'unlike' : 'like'
   174→    setLikedPosts(prev => { const n = new Set(prev); isLiked ? n.delete(postId) : n.add(postId); return n })
   175→    setPosts(prev => prev.map(p => p.id === postId ? { ...p, likes: p.likes + (isLiked ? -1 : 1) } : p))
   176→    try {
   177→      await fetch(`/api/community/posts/${postId}/interact`, {
   178→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   179→        body: JSON.stringify({ visitorId: session.email, action }),
   180→      })
   181→    } catch { /* revert silently */ }
   182→  }
   183→
   184→  const handleSave = async (postId: string) => {
   185→    if (!session) return
   186→    const isSaved = savedPosts.has(postId)
   187→    const action = isSaved ? 'unsave' : 'save'
   188→    setSavedPosts(prev => { const n = new Set(prev); isSaved ? n.delete(postId) : n.add(postId); return n })
   189→    setPosts(prev => prev.map(p => p.id === postId ? { ...p, saves: p.saves + (isSaved ? -1 : 1) } : p))
   190→    try {
   191→      await fetch(`/api/community/posts/${postId}/interact`, {
   192→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   193→        body: JSON.stringify({ visitorId: session.email, action }),
   194→      })
   195→    } catch { /* revert silently */ }
   196→  }
   197→
   198→  const handleRepost = async (postId: string) => {
   199→    if (!session || repostedPosts.has(postId)) return
   200→    setRepostedPosts(prev => { const n = new Set(prev); n.add(postId); return n })
   201→    setPosts(prev => prev.map(p => p.id === postId ? { ...p, reposts: p.reposts + 1 } : p))
   202→    try {
   203→      await fetch(`/api/community/posts/${postId}/interact`, {
   204→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   205→        body: JSON.stringify({ visitorId: session.email, action: 'repost' }),
   206→      })
   207→    } catch { /* revert silently */ }
   208→  }
   209→
   210→  const handleDelete = async (postId: string, author: string) => {
   211→    if (!session || session.email !== author) return
   212→    setPosts(prev => prev.filter(p => p.id !== postId))
   213→    try { await fetch(`/api/community/posts/${postId}?author=${session.email}`, { method: 'DELETE' }) } catch { /* silent */ }
   214→  }
   215→
   216→  /* ─── Comments ─── */
   217→  const toggleComments = async (postId: string) => {
   218→    if (commentingPostId === postId) { setCommentingPostId(null); return }
   219→    setCommentingPostId(postId)
   220→    if (!commentsMap[postId]) {
   221→      setCommentsLoading(postId)
   222→      try {
   223→        const res = await fetch(`/api/community/posts/${postId}/comments`)
   224→        const data = await res.json()
   225→        setCommentsMap(prev => ({ ...prev, [postId]: data.comments }))
   226→      } catch { /* silent */ }
   227→      setCommentsLoading(null)
   228→    }
   229→  }
   230→
   231→  const submitComment = async (postId: string) => {
   232→    if (!session || !commentText.trim()) return
   233→    const optimisticComment: Comment = {
   234→      id: 'temp-' + Date.now(), postId, author: session.email,
   235→      authorName: session.name, content: commentText.trim(), createdAt: new Date().toISOString(),
   236→    }
   237→    setCommentsMap(prev => ({ ...prev, [postId]: [...(prev[postId] || []), optimisticComment] }))
   238→    setPosts(prev => prev.map(p => p.id === postId ? { ...p, comments: p.comments + 1 } : p))
   239→    setCommentText('')
   240→    try {
   241→      await fetch(`/api/community/posts/${postId}/comments`, {
   242→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   243→        body: JSON.stringify({ author: session.email, authorName: session.name, content: commentText.trim() }),
   244→      })
   245→    } catch { /* silent */ }
   246→  }
   247→
   248→  /* ─── Composer ─── */
   249→  const handleSubmitPost = async () => {
   250→    if (!session || !composerTitle.trim() || !composerContent.trim()) return
   251→    setComposerSubmitting(true)
   252→    try {
   253→      const tags = ['Community']
   254→      if (composerLink) tags.push('Link')
   255→      const res = await fetch('/api/community/posts', {
   256→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   257→        body: JSON.stringify({
   258→          title: composerTitle.trim(), content: composerContent.trim(),
   259→          author: session.email, authorName: session.name,
   260→          imageUrl: composerImageUrl.trim() || null, sourceUrl: composerLink.trim() || null, tags,
   261→        }),
   262→      })
   263→      const data = await res.json()
   264→      if (res.ok && data.post) {
   265→        setPosts(prev => [data.post, ...prev])
   266→        setComposerTitle(''); setComposerContent(''); setComposerImageUrl(''); setComposerLink('')
   267→        setComposerOpen(false)
   268→      }
   269→    } catch { /* silent */ }
   270→    setComposerSubmitting(false)
   271→  }
   272→
   273→  /* ─── News Refresh ─── */
   274→  const handleRefreshNews = async () => {
   275→    setNewsLoading(true)
   276→    try {
   277→      const res = await fetch('/api/community/news')
   278→      const data = await res.json()
   279→      setNewsFetchedAt(data.fetchedAt)
   280→      setPage(1); await fetchPosts(1, false)
   281→    } catch { /* silent */ }
   282→    setNewsLoading(false)
   283→  }
   284→
   285→  /* ─── Fetch Topic Follows ─── */
   286→  useEffect(() => {
   287→    if (!session) return
   288→    const fetchTopics = async () => {
   289→      try {
   290→        const res = await fetch(`/api/community/topics?visitorId=${session.email}`)
   291→        const data = await res.json()
   292→        if (data.topics) setFollowedTopics(data.topics)
   293→      } catch { /* silent */ }
   294→    }
   295→    fetchTopics()
   296→  }, [session])
   297→
   298→  /* ─── Seed Portal on First Visit ─── */
   299→  useEffect(() => {
   300→    const seeded = localStorage.getItem('oneway-community-seeded')
   301→    if (!seeded) {
   302→      fetch('/api/community/seed', { method: 'POST' }).then(() => {
   303→        localStorage.setItem('oneway-community-seeded', 'true')
   304→        setPage(1); fetchPosts(1, false)
   305→      }).catch(() => { /* silent */ })
   306→    }
   307→  }, [])
   308→
   309→  /* ─── Share ─── */
   310→  const handleCopyLink = (post: Post) => {
   311→    navigator.clipboard.writeText(`${window.location.origin}/community/${post.id}`)
   312→    setCopiedLink(true)
   313→    setTimeout(() => setCopiedLink(false), 2000)
   314→  }
   315→  const handleShareEmail = () => {
   316→    if (!shareDialogPost || !shareEmail.trim()) return
   317→    const subject = encodeURIComponent(`Check this out: ${shareDialogPost.title}`)
   318→    const body = encodeURIComponent(`I thought you'd find this interesting:\n\n${shareDialogPost.title}\n\n${shareDialogPost.sourceUrl || ''}\n\n- Shared from TheOneWayGDA`)
   319→    window.open(`mailto:${shareEmail.trim()}?subject=${subject}&body=${body}`)
   320→    setShareDialogPost(null); setShareEmail('')
   321→  }
   322→
   323→  /* ─── Derived Data ─── */
   324→  const filteredPosts = (() => {
   325→    let result = activeTab === 'saved' ? posts.filter(p => savedPosts.has(p.id)) : posts
   326→    if (categoryFilter !== 'all') {
   327→      result = result.filter(p => parseTags(p.tags).includes(categoryFilter))
   328→    }
   329→    return result
   330→  })()
   331→  const pinnedDigest = posts.find(p => p.type === 'digest' && p.featured)
   332→  const communityPicks = posts.filter(p => p.type === 'user_highlight' && p.featured)
   333→  const topStories = posts
   334→    .filter(p => !['digest', 'user_highlight'].includes(p.type) && (p.featured || p.likes >= 3))
   335→    .sort((a, b) => b.likes - a.likes).slice(0, 5)
   336→  const totalStats = posts.reduce((acc, p) => ({
   337→    likes: acc.likes + (p.likes || 0), comments: acc.comments + (p.comments || 0),
   338→    reposts: acc.reposts + (p.reposts || 0), saves: acc.saves + (p.saves || 0),
   339→  }), { likes: 0, comments: 0, reposts: 0, saves: 0 })
   340→  const feedPosts = filteredPosts.filter(p => !(p.type === 'digest' && p.featured))
   341→
   342→  /* ─── RENDER ─── */
   343→  return (
   344→    <div className="min-h-screen flex flex-col mesh-gradient noise-overlay" dir={dir}>
   345→      {/* NAV */}
   346→      <nav className="sticky top-0 z-50 nav-premium">
   347→        <div className="max-w-5xl mx-auto flex items-center justify-between px-4 sm:px-6 h-14 gap-3">
   348→          <div className="flex items-center gap-3">
   349→            <Link href="/" className="flex items-center gap-2 text-muted-foreground hover:text-foreground transition-colors">
   350→              <ArrowLeft className="size-4" />
   351→            </Link>
   352→            <Image src="/images/logo.png" alt="TheOneWayGDA" width={28} height={28} className="rounded-lg" />
   353→            <span className="text-lg font-bold gradient-text-premium">{t('community.navTitle') || 'AI News & Community'}</span>
   354→          </div>
   355→          <div className="flex items-center gap-2">
   356→            <Button variant="outline" size="sm" onClick={handleRefreshNews} disabled={newsLoading} className="gap-1.5 text-xs">
   357→              <RefreshCw className={`size-3.5 ${newsLoading ? 'animate-spin' : ''}`} />
   358→              {t('community.refreshNews') || 'Fetch News'}
   359→            </Button>
   360→            <Button size="sm" onClick={() => setComposerOpen(true)} className="gap-1.5 rounded-full">
   361→              <Plus className="size-3.5" />
   362→              {t('community.newPost') || 'Post'}
   363→            </Button>
   364→            <Select value={locale} onValueChange={(v) => setLocale(v as Locale)}>
   365→              <SelectTrigger className="h-8 w-24 text-xs">
   366→                <Globe className="size-3 mr-0.5" />
   367→                <SelectValue />
   368→              </SelectTrigger>
   369→              <SelectContent>
   370→                {localeNames.map(l => (
   371→                  <SelectItem key={l} value={l} className="text-xs">{t(`lang.${l}`)}</SelectItem>
   372→                ))}
   373→              </SelectContent>
   374→            </Select>
   375→          </div>
   376→        </div>
   377→      </nav>
   378→
   379→      <div className="flex-1 max-w-5xl mx-auto w-full px-4 sm:px-6 py-6">
   380→        {/* Tabs & Filters */}
   381→        <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3 mb-6">
   382→          <Tabs value={activeTab} onValueChange={setActiveTab}>
   383→            <TabsList className="grid grid-cols-4 w-full sm:w-auto">
   384→              <TabsTrigger value="all" className="gap-1.5 text-xs"><Flame className="size-3.5" />{t('community.all') || 'All'}</TabsTrigger>
   385→              <TabsTrigger value="news" className="gap-1.5 text-xs"><Newspaper className="size-3.5" />{t('community.news') || 'AI News'}</TabsTrigger>
   386→              <TabsTrigger value="community" className="gap-1.5 text-xs"><Users className="size-3.5" />{t('community.community') || 'Community'}</TabsTrigger>
   387→              <TabsTrigger value="saved" className="gap-1.5 text-xs"><Bookmark className="size-3.5" />{t('community.saved') || 'Saved'}</TabsTrigger>
   388→            </TabsList>
   389→          </Tabs>
   390→          <div className="flex items-center gap-2 w-full sm:w-auto">
   391→            <div className="relative flex-1 sm:flex-initial">
   392→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5 text-muted-foreground" />
   393→              <Input
   394→                value={searchInput} onChange={(e) => setSearchInput(e.target.value)}
   395→                onKeyDown={(e) => e.key === 'Enter' && setSearchQuery(searchInput)}
   396→                placeholder={t('community.searchPlaceholder') || 'Search posts...'}
   397→                className="pl-8 h-8 text-xs w-full sm:w-48"
   398→              />
   399→            </div>
   400→            <Select value={sortOrder} onValueChange={setSortOrder}>
   401→              <SelectTrigger className="h-8 w-28 text-xs"><SelectValue /></SelectTrigger>
   402→              <SelectContent>
   403→                <SelectItem value="latest" className="text-xs"><Clock className="size-3 mr-1 inline" />{t('community.latest') || 'Latest'}</SelectItem>
   404→                <SelectItem value="popular" className="text-xs"><Zap className="size-3 mr-1 inline" />{t('community.popular') || 'Popular'}</SelectItem>
   405→              </SelectContent>
   406→            </Select>
   407→          </div>
   408→        </div>
   409→
   410→        {/* News fetched banner */}
   411→        {newsFetchedAt && (
   412→          <motion.div initial={{ opacity: 0, y: -5 }} animate={{ opacity: 1, y: 0 }}
   413→            className="mb-4 flex items-center gap-2 text-xs text-muted-foreground bg-emerald-500/5 border border-emerald-500/20 rounded-lg px-3 py-2">
   414→            <Sparkles className="size-3.5 text-emerald-500" />
   415→            <span>{t('community.newsUpdated') || 'AI News updated'}: {new Date(newsFetchedAt).toLocaleString()}</span>
   416→          </motion.div>
   417→        )}
   418→
   419→        {/* Category Filter */}
   420→        {activeTab !== 'saved' && (
   421→          <div className="flex gap-1.5 mb-4 overflow-x-auto pb-1">
   422→            {[
   423→              { key: 'all', label: 'All Topics', icon: Flame, color: '' },
   424→              { key: 'AI', label: 'AI', icon: Brain, color: 'bg-blue-500/10 text-blue-400 border-blue-500/30' },
   425→              { key: 'Research', label: 'Research', icon: Beaker, color: 'bg-purple-500/10 text-purple-400 border-purple-500/30' },
   426→              { key: 'Innovation', label: 'Innovation', icon: Lightbulb, color: 'bg-amber-500/10 text-amber-400 border-amber-500/30' },
   427→            ].map(cat => (
   428→              <button key={cat.key} onClick={() => setCategoryFilter(cat.key)}
   429→                className={`flex items-center gap-1 px-3 py-1.5 rounded-lg text-xs font-medium whitespace-nowrap transition-all border ${
   430→                  categoryFilter === cat.key
   431→                    ? cat.color || 'bg-primary/10 text-primary border-primary/30'
   432→                    : 'bg-muted/20 text-muted-foreground hover:text-foreground border-transparent'
   433→                }`}>
   434→                <cat.icon className="size-3" />
   435→                {cat.label}
   436→                {cat.key !== 'all' && followedTopics.includes(cat.key) && <Zap className="size-2.5 text-primary" />}
   437→              </button>
   438→            ))}
   439→          </div>
   440→        )}
   441→
   442→        {/* Live engagement bar */}
   443→        {activeTab === 'all' && !searchQuery && (
   444→          <div className="flex items-center gap-4 mb-4 px-3 py-2 rounded-lg bg-card/40 border border-border/20 text-[10px] text-muted-foreground">
   445→            <span className="flex items-center gap-1"><Activity className="size-3 text-emerald-500" /> <span className="font-medium text-foreground">{posts.length}</span> posts</span>
   446→            <span className="flex items-center gap-1"><Heart className="size-3 text-rose-400" /> <span className="font-medium text-foreground">{totalStats.likes}</span></span>
   447→            <span className="flex items-center gap-1"><MessageCircle className="size-3 text-blue-400" /> <span className="font-medium text-foreground">{totalStats.comments}</span></span>
