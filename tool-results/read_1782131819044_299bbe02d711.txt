   157→function timeAgo(dateStr: string): string {
   158→  const diff = Date.now() - new Date(dateStr).getTime();
   159→  const mins = Math.floor(diff / 60000);
   160→  if (mins < 1) return 'just now';
   161→  if (mins < 60) return `${mins}m ago`;
   162→  const hrs = Math.floor(mins / 60);
   163→  if (hrs < 24) return `${hrs}h ago`;
   164→  const days = Math.floor(hrs / 24);
   165→  return `${days}d ago`;
   166→}

   190→function getVoterId(): string {
   191→  if (typeof window === 'undefined') return 'anonymous';
   192→  let voterId = localStorage.getItem('arena-voter-id');
   193→  if (!voterId) {
   194→    voterId = `voter_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
   195→    localStorage.setItem('arena-voter-id', voterId);
   196→  }
   197→  return voterId;
   198→}
