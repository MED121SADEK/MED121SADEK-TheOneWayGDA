   161→/* ─── Panel 2: Data Import ─── */
   162→export function ImportPanel(h: HandlerHook) {
   163→  const fileInputRef = useRef<HTMLInputElement>(null)
   164→  const [dragOver, setDragOver] = useState(false)
   165→  const [importError, setImportError] = useState('')
   166→  const [importSuccess, setImportSuccess] = useState('')
   167→
   168→  const MAX_FILE_SIZE = 50 * 1024 * 1024 // 50MB
   169→  const ACCEPTED_EXTENSIONS = ['.csv', '.xlsx', '.xls', '.json', '.tsv']
   170→
   171→  const validateAndProcessFile = (file: File) => {
   172→    setImportError('')
   173→    setImportSuccess('')
   174→
   175→    // Validate file size
   176→    if (file.size > MAX_FILE_SIZE) {
   177→      setImportError(`File too large (${(file.size / 1024 / 1024).toFixed(1)}MB). Maximum is 50MB.`)
   178→      return
   179→    }
   180→    if (file.size === 0) {
   181→      setImportError('File is empty.')
   182→      return
   183→    }
   184→
   185→    // Validate extension
   186→    const ext = '.' + file.name.split('.').pop()?.toLowerCase()
   187→    if (!ACCEPTED_EXTENSIONS.includes(ext)) {
   188→      setImportError(`Unsupported file type "${ext}". Accepted: ${ACCEPTED_EXTENSIONS.join(', ')}`)
   189→      return
   190→    }
   191→
   192→    try {
   193→      h.store.importFile(file)
   194→      setImportSuccess(`Imported "${file.name}" (${(file.size / 1024).toFixed(1)}KB)`)
   195→    } catch (err) {
   196→      setImportError(`Import failed: ${err instanceof Error ? err.message : 'Unknown error'}`)
   197→    }
   198→  }
   199→
   200→  const handleDrop = (e: React.DragEvent) => {
   201→    e.preventDefault()
   202→    e.stopPropagation()
   203→    setDragOver(false)
   204→    const files = Array.from(e.dataTransfer.files)
   205→    if (files.length === 0) return
   206→    validateAndProcessFile(files[0])
   207→  }
   208→
   209→  const handleDragOver = (e: React.DragEvent) => {
   210→    e.preventDefault()
   211→    e.stopPropagation()
   212→    setDragOver(true)
   213→  }
   214→
   215→  const handleDragLeave = (e: React.DragEvent) => {
   216→    e.preventDefault()
   217→    e.stopPropagation()
   218→    setDragOver(false)
   219→  }
   220→
   221→  return (
   222→    <div className="p-4 space-y-4 overflow-y-auto h-full scrollbar-thin" style={{ contain: 'layout style paint' }}>
   223→      <div
   224→        className={cn(
   225→          'border-2 border-dashed rounded-xl p-8 text-center cursor-pointer transition-all',
   226→          dragOver ? 'border-teal-400 bg-teal-500/10 scale-[1.01]' : 'border-border hover:border-teal-400/50 hover:bg-teal-500/5'
   227→        )}
   228→        onClick={() => fileInputRef.current?.click()}
   229→        onDrop={handleDrop}
   230→        onDragOver={handleDragOver}
   231→        onDragLeave={handleDragLeave}
   232→      >
   233→        <input
   234→          ref={fileInputRef}
   235→          type="file"
   236→          accept=".csv,.xlsx,.xls,.json,.tsv"
   237→          className="hidden"
   238→          onChange={(e) => {
   239→            const file = e.target.files?.[0]
   240→            if (file) validateAndProcessFile(file)
   241→            // Reset input so same file can be re-selected
   242→            if (fileInputRef.current) fileInputRef.current.value = ''
   243→          }}
   244→        />
   245→        {dragOver ? (
   246→          <>
   247→            <Download className="w-8 h-8 mx-auto mb-3 text-teal-400" />
   248→            <p className="text-sm font-medium text-teal-400">Drop your file here</p>
   249→          </>
   250→        ) : (
   251→          <>
   252→            <Upload className="w-8 h-8 mx-auto mb-3 text-teal-400" />
   253→            <p className="text-sm font-medium text-foreground">{h.t('import.dragDrop')}</p>
   254→            <p className="text-xs text-muted-foreground mt-1">{h.t('import.supportedFormats')}</p>
   255→            <p className="text-[10px] text-muted-foreground/60 mt-1">Max 50MB</p>
   256→          </>
   257→        )}
   258→      </div>
   259→
   260→      {importError && (
   261→        <div className="flex items-start gap-2 bg-red-500/10 border border-red-500/20 rounded-lg p-2.5 text-[10px] text-red-400">
   262→          <X className="w-3 h-3 mt-0.5 flex-shrink-0" />
   263→          <div>
   264→            <p className="font-medium">Import Error</p>
   265→            <p>{importError}</p>
   266→          </div>
   267→          <button onClick={() => setImportError('')} className="ml-auto flex-shrink-0 hover:text-red-300">
   268→            <X className="w-3 h-3" />
   269→          </button>
   270→        </div>
   271→      )}
   272→      {importSuccess && (
   273→        <div className="flex items-center gap-2 bg-emerald-500/10 border border-emerald-500/20 rounded-lg p-2.5 text-[10px] text-emerald-400">
   274→          <Check className="w-3 h-3" />
   275→          <span>{importSuccess}</span>
   276→        </div>
   277→      )}
   278→
   279→      <Separator />
   280→
   281→      <div className="space-y-2">
   282→        <p className="text-xs font-semibold text-foreground">Paste CSV Data</p>
   283→        <Textarea
   284→          value={h.importText}
   285→          onChange={(e) => h.setImportText(e.target.value)}
   286→          placeholder="Name,Age,Score&#10;Alice,25,89&#10;Bob,30,92"
   287→          className="text-xs min-h-24 font-mono"
   288→        />
   289→        <Button size="sm" className="w-full bg-teal-600 hover:bg-teal-700 text-white text-xs" onClick={h.handleImportCSV} disabled={!h.importText.trim()}>
   290→          <Database className="w-3.5 h-3.5 mr-1.5" />
   291→          Import CSV
   292→        </Button>
   293→      </div>
   294→
   295→      <Separator />
   296→
   297→      <div className="space-y-2">
   298→        <p className="text-xs font-semibold text-foreground">Quick Templates</p>
   299→        <div className="grid grid-cols-2 gap-2">
   300→          {[
   301→            { name: 'Sample Survey', data: 'Respondent,Age,Gender,Satisfaction,Income\n1,25,Male,4,45000\n2,30,Female,5,52000\n3,22,Male,3,38000\n4,35,Female,4,61000\n5,28,Male,5,48000\n6,40,Female,3,72000\n7,33,Male,4,55000\n8,27,Female,5,42000' },
   302→            { name: 'Student Grades', data: 'Student,Math,Science,English,History\nAlice,85,92,88,76\nBob,78,85,90,82\nCarol,92,88,76,95\nDave,65,72,88,70\nEve,88,95,82,90' },
   303→            { name: 'Sales Data', data: 'Month,Revenue,Profit,Customers\nJan,45000,12000,320\nFeb,52000,15000,380\nMar,48000,11000,350\nApr,61000,18000,420\nMay,55000,14000,390' },
   304→          ].map((tpl) => (

   409→/* ─── Panel 4: Analysis Center (Self-Contained) ─── */
   410→export function AnalysisPanel(h: HandlerHook) {
   411→  const [lastRun, setLastRun] = useState<string | null>(null)
   412→  const [lastRunTime, setLastRunTime] = useState<number>(0)
   413→  const [inlineMode, setInlineMode] = useState<string | null>(null) // for dialog-based tools
   414→  const [dialogVars, setDialogVars] = useState<{ var1: string; var2: string }>({ var1: '', var2: '' })
   415→
   416→  const hasData = h.store.variables.length > 0
   417→  const hasSelection = h.store.selectedVariables.length > 0
   418→  const selCount = h.store.selectedVariables.length
   419→
   420→  const runTool = useCallback((name: string, action: () => void) => {
   421→    setLastRun(name)
   422→    setLastRunTime(Date.now())
   423→    action()
   424→  }, [])
   425→
   426→  const selectAllNumeric = () => {
   427→    const numeric = h.store.variables.filter(v => v.type === 'numeric').map(v => v.name)
   428→    h.store.setSelectedVariables(numeric)
   429→  }
   430→
   431→  const selectAll = () => {
   432→    h.store.setSelectedVariables(h.store.variables.map(v => v.name))
   433→  }
   434→
   435→  const runInlineAnalysis = useCallback((type: string) => {
   436→    if (!dialogVars.var1 || !dialogVars.var2) return
   437→    if (type === 'crosstabs') {
   438→      h.setCrosstabRowVar(dialogVars.var1)
   439→      h.setCrosstabColVar(dialogVars.var2)
   440→      h.handleRunCrosstabs()
   441→    } else if (type === 'ttest') {
   442→      h.setTtestGroupVar(dialogVars.var1)
   443→      h.setTtestValueVar(dialogVars.var2)
   444→      h.handleRunTTest()
   445→    } else if (type === 'anova') {
   446→      h.setAnovaGroupVar(dialogVars.var1)
   447→      h.setAnovaValueVar(dialogVars.var2)

   753→function OutputTable({ item }: { item: OutputItem }) {
   754→  if (item.type !== 'table' || !item.content) return null
   755→  const { headers = [], rows = [] } = item.content as { headers?: string[]; rows?: string[][] }
   756→  return (
   757→    <div className="overflow-x-auto border rounded-lg">
   758→      <table className="w-full text-[10px] border-collapse">
   759→        <thead className="sticky top-0">
   760→          <tr className="bg-muted/60">
   761→            {headers.map((h, i) => (
   762→              <th key={i} className="border px-2 py-1.5 text-left font-semibold text-foreground whitespace-nowrap">{h}</th>
   763→            ))}
   764→          </tr>
   765→        </thead>
   766→        <tbody>
   767→          {rows.map((row, ri) => (
   768→            <tr key={ri} className="hover:bg-muted/20">
   769→              {row.map((cell, ci) => (
   770→                <td key={ci} className="border px-2 py-1 text-muted-foreground whitespace-nowrap">{cell}</td>
   771→              ))}
   772→            </tr>
   773→          ))}
   774→        </tbody>
   775→      </table>
   776→    </div>
   777→  )

   780→function OutputChart({ item }: { item: OutputItem }) {
   781→  if (item.type !== 'chart' || !item.content) return null
   782→  const c = item.content as any
   783→  const chartId = `chart-${item.id}`
   784→
   785→  const renderChart = () => {
   786→    switch (c.chartType) {
   787→      case 'scatter':
   788→        return <ChartScatter data={c.data} title={c.title} />
   789→      case 'histogram':
   790→        return <ChartHistogram values={c.values} title={c.title} />
   791→      case 'pie':
   792→        return <ChartPie data={c.data} title={c.title} />
   793→      case 'bar':
   794→        return <ChartBar data={c.data} title={c.title} bars={[]} />
   795→      case 'boxplot':
   796→        return <ChartBoxPlot groups={c.groups} title={c.title} />
   797→      case 'line':
   798→        return <ChartLine data={c.data || []} lines={c.lines || []} title={c.title} />
   799→      default:
   800→        return <p className="text-xs text-muted-foreground">Unknown chart type</p>
   801→    }
   802→  }
   803→
   804→  return (
   805→    <Card className="border-border/50">
   806→      <CardContent className="p-3">
   807→        <div className="flex items-center justify-between mb-2">
   808→          <span className="text-[10px] font-semibold truncate">{item.title}</span>
   809→          <Button
   810→            variant="ghost"
   811→            size="sm"
   812→            className="h-5 w-5 p-0 text-muted-foreground hover:text-foreground"
   813→            onClick={() => exportChartAsImage(chartId as unknown as React.RefObject<HTMLDivElement>)}
   814→          >
   815→            <Download className="w-3 h-3" />
   816→          </Button>
   817→        </div>
   818→        <div id={chartId} className="w-full">
   819→          {renderChart()}
   820→        </div>
   821→      </CardContent>
   822→    </Card>
   823→  )
   824→}

   934→/* ─── Panel 7: Scan & OCR ─── */
   935→export function ScanPanel(h: HandlerHook) {
   936→  const [dragOver, setDragOver] = useState(false)
   937→  const [scanError, setScanError] = useState('')
   938→
   939→  const MAX_SCAN_SIZE = 20 * 1024 * 1024 // 20MB
   940→  const SCAN_ACCEPTED = ['.pdf', '.png', '.jpg', '.jpeg', '.webp', '.bmp', '.tiff']
   941→
   942→  const validateScanFile = (file: File): string | null => {
   943→    if (file.size > MAX_SCAN_SIZE) return `File too large (${(file.size / 1024 / 1024).toFixed(1)}MB). Max 20MB.`
   944→    if (file.size === 0) return 'File is empty.'
   945→    const ext = '.' + file.name.split('.').pop()?.toLowerCase()
   946→    if (!SCAN_ACCEPTED.includes(ext)) return `Unsupported type. Accepted: ${SCAN_ACCEPTED.join(', ')}`
   947→    return null
   948→  }
   949→
   950→  const processScanFile = (file: File) => {
   951→    setScanError('')
   952→    const err = validateScanFile(file)
   953→    if (err) { setScanError(err); return }
   954→    h.setScanFile(file)
   955→    const reader = new FileReader()
   956→    reader.onload = (ev) => h.setScanPreview(ev.target?.result as string)
   957→    reader.onerror = () => setScanError('Failed to read file.')
   958→    reader.readAsDataURL(file)
   959→  }
   960→
   961→  const handleScanDrop = (e: React.DragEvent) => {
   962→    e.preventDefault(); e.stopPropagation(); setDragOver(false)
   963→    const files = Array.from(e.dataTransfer.files)
   964→    if (files.length > 0) processScanFile(files[0])
   965→  }
   966→
   967→  const handleScanDragOver = (e: React.DragEvent) => { e.preventDefault(); e.stopPropagation(); setDragOver(true) }
   968→  const handleScanDragLeave = (e: React.DragEvent) => { e.preventDefault(); e.stopPropagation(); setDragOver(false) }
   969→
   970→  const handleScanSubmit = async () => {
   971→    if (!h.scanFile) return
   972→    setScanError('')
   973→    h.store.setScanState('processing')
   974→
   975→    const formData = new FormData()
   976→    formData.append('file', h.scanFile)
   977→
   978→    if (h.scanFile.name.toLowerCase().endsWith('.pdf')) {
   979→      formData.append('templateHint', 'PDF document')
   980→    }
   981→
   982→    try {
   983→      const res = await fetch('/api/scan', { method: 'POST', body: formData })
   984→      const data = await res.json()
   985→
   986→      if (!res.ok) {
   987→        setScanError(data.error || `Scan failed (HTTP ${res.status})`)
   988→        h.store.setScanState('error')
   989→        return
   990→      }
   991→
   992→      if (data.fields && data.fields.length > 0) {
   993→        h.store.setScanResults(data as any)
   994→      } else if (data.rawText && data.rawText.length > 10) {
   995→        h.store.setScanResults({
   996→          ...data,
   997→          fields: [{ label: 'Raw Text', value: data.rawText.slice(0, 500), confidence: 0.5, type: 'string' }],
   998→          summary: data.summary || 'Document text extracted. Review the raw text below.',
   999→        } as any)
  1000→      } else {
  1001→        setScanError('Could not extract structured data from this document. The content may be unclear, handwritten, or image-based. Try uploading a clearer scan or a different file format.')
  1002→        h.store.setScanState('error')
  1003→      }
  1004→    } catch (err) {
  1005→      setScanError(`Network error: ${err instanceof Error ? err.message : 'Connection failed'}`)
  1006→      h.store.setScanState('error')
  1007→    }
  1008→  }
  1009→
  1010→  return (
  1011→    <div className="p-4 space-y-4 overflow-y-auto h-full scrollbar-thin" style={{ contain: 'layout style paint' }}>
  1012→      <div
  1013→        className={cn(
  1014→          'border-2 border-dashed rounded-xl p-8 text-center cursor-pointer transition-all',
  1015→          h.scanFile ? 'border-cyan-400 bg-cyan-50/10' : dragOver ? 'border-cyan-400 bg-cyan-500/10 scale-[1.01]' : 'border-border hover:border-cyan-400/50 hover:bg-cyan-500/5'
  1016→        )}
  1017→        onClick={() => h.fileInputRef.current?.click()}
  1018→        onDrop={handleScanDrop}
  1019→        onDragOver={handleScanDragOver}
  1020→        onDragLeave={handleScanDragLeave}
  1021→      >
  1022→        <input
  1023→          ref={h.fileInputRef}
  1024→          type="file"
  1025→          accept=".pdf,.png,.jpg,.jpeg,.webp,.bmp,.tiff"
  1026→          className="hidden"
  1027→          onChange={(e) => {
  1028→            const f = e.target.files?.[0]
  1029→            if (f) processScanFile(f)
  1030→            if (h.fileInputRef.current) h.fileInputRef.current.value = ''
  1031→          }}
  1032→        />
  1033→        {h.store.scanState === 'processing' ? (
  1034→          <div className="flex flex-col items-center gap-2">
  1035→            <div className="w-8 h-8 border-2 border-cyan-500 border-t-transparent rounded-full animate-spin" />
  1036→            <p className="text-sm text-muted-foreground">Analyzing document with AI...</p>
  1037→            <p className="text-[10px] text-muted-foreground/60">This may take 10-30 seconds for complex documents</p>
  1038→          </div>
  1039→        ) : h.scanPreview ? (
  1040→          <div className="flex flex-col items-center gap-2">
  1041→            {/* eslint-disable-next-line @next/next/no-img-element */}
  1042→            <img src={h.scanPreview} alt="Preview" className="max-h-40 rounded-lg shadow-sm" />
  1043→            <p className="text-[10px] text-muted-foreground">{h.scanFile?.name} ({((h.scanFile?.size || 0) / 1024).toFixed(1)}KB)</p>
  1044→          </div>
  1045→        ) : (
  1046→          <>
  1047→            <ScanLine className="w-8 h-8 mx-auto mb-2 text-cyan-400" />
  1048→            <p className="text-sm font-medium">{h.t('scan.dragDrop')}</p>
  1049→            <p className="text-[10px] text-muted-foreground mt-1">{h.t('scan.supportedFormats')}</p>
  1050→            <p className="text-[10px] text-muted-foreground/60 mt-1">Max 20MB - Drag and drop or click</p>
  1051→          </>
  1052→        )}
  1053→      </div>
  1054→
  1055→      {scanError && (
  1056→        <div className="flex items-start gap-2 bg-red-500/10 border border-red-500/20 rounded-lg p-2.5 text-[10px] text-red-400">
  1057→          <X className="w-3 h-3 mt-0.5 flex-shrink-0" />
  1058→          <div className="flex-1">
  1059→            <p className="font-medium">Scan Error</p>
  1060→            <p>{scanError}</p>
  1061→          </div>
  1062→          <button onClick={() => setScanError('')} className="ml-auto flex-shrink-0 hover:text-red-300">
  1063→            <X className="w-3 h-3" />
  1064→          </button>
  1065→        </div>
  1066→      )}
  1067→
  1068→      <div className="flex gap-2">
  1069→        <Button size="sm" className="flex-1 bg-cyan-600 hover:bg-cyan-700 text-white text-xs" disabled={!h.scanFile || h.store.scanState === 'processing'} onClick={handleScanSubmit}>
  1070→          {h.store.scanState === 'processing' ? 'Analyzing...' : h.t('scan.title')}
  1071→        </Button>
  1072→        {h.store.scanState === 'done' && h.store.scanResults && (
  1073→          <Button size="sm" variant="outline" className="flex-1 text-xs" onClick={() => { h.store.importScanResults(h.store.scanResults!); h.setScanDialogOpen(false) }}>
  1074→            <Check className="w-3.5 h-3.5 mr-1" />{h.t('scan.approve')}
  1075→          </Button>
  1076→        )}
  1077→      </div>
  1078→
  1079→      {/* Raw text fallback when fields are limited */}
  1080→      {h.store.scanState === 'done' && h.store.scanResults && h.store.scanResults.rawText && h.store.scanResults.fields.length <= 2 && (
  1081→        <div className="space-y-1.5">
  1082→          <p className="text-[10px] font-semibold text-amber-400">Extracted Text (raw)</p>
  1083→          <div className="max-h-32 overflow-y-auto bg-muted/20 rounded-lg p-2 text-[10px] text-muted-foreground whitespace-pre-wrap">
  1084→            {h.store.scanResults.rawText.slice(0, 2000)}
  1085→          </div>
  1086→          <p className="text-[9px] text-muted-foreground/60">AI had difficulty parsing structured fields from this document. You can copy the raw text or re-upload a clearer version.</p>
  1087→        </div>
  1088→      )}
  1089→
  1090→      {/* Batch Scan */}
  1091→      <div className="flex items-center gap-2">
  1092→        <Button size="sm" variant="outline" className="text-[10px] h-7" onClick={() => h.batchInputRef.current?.click()}>
  1093→          <FolderOpen className="w-3 h-3 mr-1" />{h.t('scan.batch')}
  1094→        </Button>
  1095→        <input ref={h.batchInputRef} type="file" accept=".pdf,.png,.jpg,.jpeg,.webp" multiple className="hidden" onChange={async (e) => {
  1096→          const files = Array.from(e.target.files || [])
  1097→          for (const f of files) {
  1098→            const err = validateScanFile(f)
  1099→            if (err) continue
  1100→            h.store.addToBatchQueue({ id: Date.now().toString(36) + Math.random().toString(36).slice(2), name: f.name, status: 'pending' })
  1101→          }
  1102→          if (h.batchInputRef.current) h.batchInputRef.current.value = ''
  1103→        }} />
  1104→        {h.store.batchQueue.length > 0 && <span className="text-[10px] text-muted-foreground">{h.store.batchQueue.length} files queued</span>}
  1105→      </div>
  1106→
  1107→      {/* Results */}
  1108→      {h.store.scanState === 'done' && h.store.scanResults && h.store.scanResults.fields.length > 0 && (
  1109→        <div className="space-y-2">
  1110→          <p className="text-[10px] font-semibold">{h.t('scan.extracted')} ({h.store.scanResults.fields.length})</p>
  1111→          <div className="space-y-1.5 max-h-48 overflow-y-auto">
  1112→            {h.store.scanResults.fields.map((f: any, i: number) => (
  1113→              <div key={i} className="flex items-center gap-2 bg-muted/20 rounded-lg p-2 text-[10px]">
  1114→                <Badge variant="outline" className="text-[8px] w-12 justify-center shrink-0">{f.type || 'text'}</Badge>
  1115→                <span className="font-medium w-20 truncate">{f.label}</span>
  1116→                <input className="flex-1 text-[10px] bg-background border rounded px-1.5 py-0.5 min-w-0" value={h.editedFields[f.label] ?? f.value} onChange={e => h.setEditedFields(p => ({ ...p, [f.label]: e.target.value }))} />
  1117→                <Badge variant={f.confidence > 0.8 ? 'default' : 'secondary'} className={cn('text-[8px]', f.confidence > 0.8 ? 'bg-emerald-500 text-white' : 'bg-amber-500 text-white')}>
  1118→                  {Math.round((f.confidence || 0.5) * 100)}%
  1119→                </Badge>
  1120→              </div>
  1121→            ))}
  1122→          </div>
  1123→        </div>
  1124→      )}
  1125→
  1126→      {h.store.scanHistory.length > 0 && (
  1127→        <div>
  1128→          <p className="text-[10px] font-semibold mb-1.5">{h.t('scan.history')}</p>
  1129→          <div className="space-y-1 max-h-24 overflow-y-auto">
  1130→            {h.store.scanHistory.slice(-5).reverse().map((s: any, i: number) => (
  1131→              <div key={i} className="text-[10px] text-muted-foreground bg-muted/20 rounded p-1.5">{s.summary || `${s.fields.length} fields, ${s.tables.length} tables`}</div>
  1132→            ))}
  1133→          </div>
  1134→        </div>
  1135→      )}
  1136→    </div>
  1137→  )
