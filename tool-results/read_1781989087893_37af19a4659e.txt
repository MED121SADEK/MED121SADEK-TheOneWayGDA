    73→export function generateActionToken(userId: string, action: 'approve' | 'reject' | 'reset' | 'sub_approve' | 'sub_reject'): string {
    74→  const payload = `${userId}:${action}:${Date.now()}`
    75→  const signature = createHmac('sha256', EMAIL_SECRET).update(payload).digest('hex')
    76→  return Buffer.from(`${payload}:${signature}`).toString('base64url')
    77→}

    79→export function verifyActionToken(token: string): { userId: string; action: 'approve' | 'reject' | 'reset'; valid: boolean } {
    80→  try {
    81→    const decoded = Buffer.from(token, 'base64url').toString('utf-8')
    82→    const [payload, signature] = decoded.split(':')
    83→
    84→    // Reconstruct and verify signature
    85→    const parts = payload.split(':')
    86→    if (parts.length < 3) return { userId: '', action: 'approve', valid: false }
    87→
    88→    const userId = parts[0]
    89→    const action = parts[1] as 'approve' | 'reject' | 'reset'
    90→    const timestamp = parseInt(parts[2])
    91→
    92→    // Check token expiry (24 hours)
    93→    if (Date.now() - timestamp > 24 * 60 * 60 * 1000) {
    94→      return { userId, action, valid: false }
    95→    }
    96→
    97→    // Verify HMAC signature
    98→    const expectedPayload = `${userId}:${action}:${timestamp}`
    99→    const expectedSig = createHmac('sha256', EMAIL_SECRET).update(expectedPayload).digest('hex')
   100→
   101→    // Find the signature part (everything after the last : of payload)
   102→    const sigIndex = decoded.lastIndexOf(':') + 1
   103→    const actualSig = decoded.slice(sigIndex)
   104→
   105→    if (actualSig === expectedSig) {
   106→      return { userId, action, valid: true }
   107→    }
   108→    return { userId, action, valid: false }
   109→  } catch {
   110→    return { userId: '', action: 'approve', valid: false }
   111→  }
   112→}
