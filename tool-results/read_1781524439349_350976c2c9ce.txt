    73→function getSession(): { email: string; name: string } | null {
    74→  if (typeof window === 'undefined') return null
    75→  try {
    76→    const s = localStorage.getItem('oneway-visitor-session')
    77→    if (!s) return null
    78→    return JSON.parse(s)
    79→  } catch { return null }
    80→}

    82→function timeAgo(date: string): string {
    83→  const now = Date.now()
    84→  const d = new Date(date).getTime()
    85→  const diff = now - d
    86→  const mins = Math.floor(diff / 60000)
    87→  if (mins < 1) return 'just now'
    88→  if (mins < 60) return `${mins}m ago`
    89→  const hrs = Math.floor(mins / 60)
    90→  if (hrs < 24) return `${hrs}h ago`
    91→  const days = Math.floor(hrs / 24)
    92→  if (days < 7) return `${days}d ago`
    93→  return new Date(date).toLocaleDateString()
    94→}

    96→function parseTags(tags: string | null | undefined): string[] {
    97→  if (!tags) return []

   101→function truncate(text: string, len: number): string {
   102→  if (text.length <= len) return text
   103→  return text.slice(0, len) + '...'

   106→/* ─── Post Type Helper ─── */
   107→function getPostTypeInfo(type: string): { label: string; icon: any; bg: string; border: string; text: string } {
   108→  switch (type) {
   109→    case 'auto':
   110→      return { label: 'Auto', icon: Sparkles, bg: 'bg-blue-500/10', border: 'border-blue-500/30', text: 'text-blue-400' }
   111→    case 'digest':
   112→      return { label: 'Daily Digest', icon: FileText, bg: 'bg-violet-500/10', border: 'border-violet-500/30', text: 'text-violet-400' }
   113→    case 'user_highlight':
   114→      return { label: 'Community Picks', icon: Award, bg: 'bg-amber-500/10', border: 'border-amber-500/30', text: 'text-amber-400' }
   115→    case 'news':
   116→      return { label: 'AI News', icon: Newspaper, bg: 'bg-sky-500/10', border: 'border-sky-500/30', text: 'text-sky-400' }
   117→    default:
   118→      return { label: 'Community', icon: Users, bg: 'bg-emerald-500/10', border: 'border-emerald-500/30', text: 'text-emerald-400' }
   119→  }
   120→}
